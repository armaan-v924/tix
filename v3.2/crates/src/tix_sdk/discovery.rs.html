1//! Ticket discovery: which ticket does a command operate on?
2//!
3//! Owned by `tix-cli` and written here first; promoted to `tix-sdk` (#96) so
4//! plugins resolve tickets identically to the canonical frontend. Explicitly
5//! *not* an engine concern — `tix-engine` takes resolved paths as given
6//! ([ticket discovery](https://tix.armaanv.dev/latest/concepts/ticket-discovery/)).
7//!
8//! Two paths in:
9//!
10//! - **Discovery** ([`discover_ticket_root`]): walk upward from the logical
11//!   cwd testing for the *file* `.tix/ticket.toml`; nearest ancestor wins.
12//! - **Override** ([`resolve_override`]): a `--ticket <path | id>` argument,
13//!   disambiguated by shape ([`TicketRef`]). Both forms are assertions, not
14//!   starting points — a near-miss (e.g. a subdirectory of a ticket) errors
15//!   rather than re-walking.
16
17use crate::error::SdkError;
18use std::path::{Path, PathBuf};
19use tix_engine::TixError;
20use tracing::debug;
21
22/// A `--ticket` argument: one flag, two forms, disambiguated by shape
23/// ([ticket discovery](https://tix.armaanv.dev/latest/concepts/ticket-discovery/)).
24///
25/// - **Path** — the argument contains a path separator, is absolute, or is
26///   `.`/`..`. Asserts *this path is the ticket root*.
27/// - **Id** — any bare name, resolved against the configured tickets
28///   directory.
29///
30/// A bare name is always an id; a ticket directory in cwd must be written
31/// `./NAME`. See [`resolve_override`] for the resolution semantics.
32#[derive(Debug, Clone, PartialEq, Eq)]
33pub enum TicketRef {
34    /// An asserted ticket-root path (`./NAME`, `some/dir`, `/abs/path`, `.`, `..`).
35    Path(PathBuf),
36    /// A bare ticket id, resolved as `tickets_directory.join(id)`.
37    Id(String),
38}
39
40impl std::str::FromStr for TicketRef {
41    type Err = String;
42
43    fn from_str(s: &str) -> Result<Self, Self::Err> {
44        let path = Path::new(s);
45        let is_path_shape = s == "." || s == ".." || path.is_absolute() || s.contains(['/', '\\']);
46        if is_path_shape {
47            Ok(TicketRef::Path(path.to_path_buf()))
48        } else {
49            Ok(TicketRef::Id(s.to_string()))
50        }
51    }
52}
53
54/// Returns true if `path` is a ticket root: a directory containing the file
55/// `.tix/ticket.toml`.
56///
57/// The predicate is the **file**, not the bare `.tix/` directory — projects
58/// live above tickets with their own `.tix/`, so testing for the directory
59/// would halt an upward walk at a project and misreport it as a ticket
60/// ([ticket discovery](https://tix.armaanv.dev/latest/concepts/ticket-discovery/)).
61pub fn is_ticket_root(path: &Path) -> bool {
62    path.join(".tix").join("ticket.toml").is_file()
63}
64
65/// The logical current working directory — `$PWD` as the shell reports it.
66///
67/// Ticket directories contain worktrees and people symlink into them, so
68/// logical vs. physical paths can resolve *different tickets* from the same
69/// `cd`. Discovery MUST NOT canonicalize: we walk what the user
70/// sees in their prompt, matching git's behavior. `$PWD` is trusted only when
71/// it is absolute and still names the same directory as the process cwd;
72/// otherwise the process cwd is the fallback.
73pub fn logical_cwd() -> Result<PathBuf, SdkError> {
74    let process_cwd =
75        std::env::current_dir().map_err(|e| SdkError::Engine(TixError::IoError(e)))?;
76    if let Some(pwd) = std::env::var_os("PWD") {
77        let pwd = PathBuf::from(pwd);
78        if pwd.is_absolute() && same_directory(&pwd, &process_cwd) {
79            return Ok(pwd);
80        }
81    }
82    Ok(process_cwd)
83}

85/// Walks upward from the logical cwd and returns the nearest ticket root.
86///
87/// Equivalent to [`discover_ticket_root_from`] starting at [`logical_cwd`].
88/// Returns `Ok(None)` when no ancestor is a ticket root — commands that
89/// require ticket context should turn that into a clear error; there is
90/// deliberately **no** fallback to the tickets directory.
91pub fn discover_ticket_root() -> Result<Option<PathBuf>, SdkError> {
92    Ok(discover_ticket_root_from(&logical_cwd()?))
93}
94
95/// Walks upward from `start`, testing each directory for `.tix/ticket.toml`.
96///
97/// - Nearest ancestor wins.
98/// - Ceiling is the filesystem root, stopping early at a device boundary
99///   (matching git's `GIT_DISCOVERY_ACROSS_FILESYSTEM` default).
100/// - The walk is **not** bounded by the configured tickets directory — that
101///   coupling was rejected because a moved or symlinked ticket would fail
102///   with no visible cause.
103pub fn discover_ticket_root_from(start: &Path) -> Option<PathBuf> {
104    let found = ancestors_on_device(start).find(|dir| is_ticket_root(dir))?;
105    debug!(path = %found.display(), "ticket root found");
106    Some(found.to_path_buf())
107}
108
109/// What one upward walk found: the nearest ticket, and the nearest project
110/// above it.
111///
112/// Either may be absent independently — a loose ticket has no project, and
113/// a cwd at project level (above every ticket) has a project but no ticket.
114#[derive(Debug, Clone, Default, PartialEq, Eq)]
115pub struct Discovered {
116    /// The nearest ticket root at or above the start.
117    pub ticket: Option<PathBuf>,
118    /// The nearest project root above the ticket (or at or above the start,
119    /// when there is no ticket).
120    pub project: Option<PathBuf>,
121}
122
123/// Walks upward from the logical cwd, collecting ticket and project.
124///
125/// Equivalent to [`discover_from`] starting at [`logical_cwd`].
126pub fn discover() -> Result<Discovered, SdkError> {
127    Ok(discover_from(&logical_cwd()?))
128}
129
130/// Walks upward from `start` once, collecting the nearest ticket root
131/// (`.tix/ticket.toml`) and the nearest project root
132/// ([`.tix/project.toml`](crate::project::is_project_root)) above it.
133///
134/// Same walk as [`discover_ticket_root_from`] — logical path, device-boundary
135/// ceiling, **not** bounded by the configured tickets directory — except that
136/// it carries on past the ticket to find the project. The project marker is
137/// what makes this possible without config: from inside
138/// `<tickets>/acme/JIRA-1/api` nothing but the marker says `acme` is a
139/// project rather than any other ancestor.
140///
141/// A directory that is a ticket root is only ever the ticket, even if it
142/// also carries a project marker; and the walk ends at the first project, so
143/// a ticket *above* a project is never reported.
144pub fn discover_from(start: &Path) -> Discovered {
145    let mut found = Discovered::default();
146    for dir in ancestors_on_device(start) {
147        if found.ticket.is_none() && is_ticket_root(dir) {
148            debug!(path = %dir.display(), "ticket root found");
149            found.ticket = Some(dir.to_path_buf());
150        } else if crate::project::is_project_root(dir) {
151            debug!(path = %dir.display(), "project root found");
152            found.project = Some(dir.to_path_buf());
153            break;
154        }
155    }
156    found
157}
158
159/// `start` and its ancestors, nearest first, ending at the filesystem root
160/// or just before the first directory on a different device (matching git's
161/// `GIT_DISCOVERY_ACROSS_FILESYSTEM` default).
162///
163/// The boundary only stops the walk when both devices are known and differ;
164/// an unreadable directory does not end it.
165fn ancestors_on_device(start: &Path) -> impl Iterator<Item = &Path> {
166    let start_device = device_of(start);
167    start.ancestors().take_while(move |dir| {
168        let crossed = matches!((start_device, device_of(dir)), (Some(a), Some(b)) if a != b);
169        if crossed {
170            debug!(path = %dir.display(), "stopping discovery at device boundary");
171        }
172        !crossed
173    })
174}

176/// Resolves an explicit `--ticket` override, skipping the discovery walk.
177///
178/// Disambiguated by shape at parse time (see [`TicketRef`]):
179///
180/// - **Path form** (`./NAME`, `some/dir`, `/abs/path`, `.`, `..`): asserts
181///   *this path is the ticket root*. The cwd is left alone.
182/// - **Id form** (any bare name): resolved as `tickets_directory.join(id)`,
183///   v2 parity. A ticket directory in cwd must be written `./NAME` — a bare
184///   name is always an id.
185///
186/// Both forms are assertions: a path that is not a ticket root errors rather
187/// than being used as a starting point for a new walk.
188///
189/// As a safety net (not a bound), a resolved ticket outside
190/// `tickets_directory` is logged at debug and accepted.
191///
192/// # Errors
193///
194/// [`TixError::TicketNotFound`] if the asserted path is not a ticket root.
195pub fn resolve_override(ticket: &TicketRef, tickets_directory: &Path) -> Result<PathBuf, SdkError> {
196    let (root, described) = match ticket {
197        TicketRef::Path(path) => (path.clone(), format!("path '{}'", path.display())),
198        TicketRef::Id(id) => (tickets_directory.join(id), format!("ticket id '{id}'")),
199    };
200
201    if !is_ticket_root(&root) {
202        return Err(SdkError::Engine(TixError::TicketNotFound(format!(
203            "{described} is not a ticket root (no .tix/ticket.toml at {})",
204            root.display()
205        ))));
206    }
207
208    if !root.starts_with(tickets_directory) {
209        debug!(
210            path = %root.display(),
211            tickets_directory = %tickets_directory.display(),
212            "resolved ticket lies outside the tickets directory"
213        );
214    }
215
216    Ok(root)
217}

219/// Resolves the ticket a command operates on: the `--ticket` override when
220/// given, the discovery walk otherwise.
221///
222/// Returns `Ok(None)` only when there is no override and the walk found
223/// nothing. Commands that require ticket context should error clearly on
224/// `None`; commands that merely prefer it (or create tickets) may proceed.
225pub fn resolve_ticket_root(
226    ticket: Option<&TicketRef>,
227    tickets_directory: &Path,
228) -> Result<Option<PathBuf>, SdkError> {
229    match ticket {
230        Some(ticket_ref) => resolve_override(ticket_ref, tickets_directory).map(Some),
231        None => discover_ticket_root(),
232    }
233}

235/// Returns true when two paths name the same directory (same device+inode on
236/// Unix; canonical-path equality elsewhere).
237#[cfg(unix)]
238pub(crate) fn same_directory(a: &Path, b: &Path) -> bool {
239    use std::os::unix::fs::MetadataExt;
240    match (std::fs::metadata(a), std::fs::metadata(b)) {
241        (Ok(ma), Ok(mb)) => ma.dev() == mb.dev() && ma.ino() == mb.ino(),
242        _ => false,
243    }
244}
245
246#[cfg(not(unix))]
247pub(crate) fn same_directory(a: &Path, b: &Path) -> bool {
248    match (std::fs::canonicalize(a), std::fs::canonicalize(b)) {
249        (Ok(ca), Ok(cb)) => ca == cb,
250        _ => false,
251    }
252}

254/// The device id `path` lives on, when it can be determined.
255#[cfg(unix)]
256fn device_of(path: &Path) -> Option<u64> {
257    use std::os::unix::fs::MetadataExt;
258    std::fs::metadata(path).ok().map(|m| m.dev())
259}
260
261#[cfg(not(unix))]
262fn device_of(_path: &Path) -> Option<u64> {
263    // Windows has no cheap device-id equivalent here; the walk simply runs to
264    // the filesystem root.
265    None
266}
