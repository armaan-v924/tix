1//! `tix config get` — read value(s) from anywhere in the config document.
2
3use crate::tix::config::ConfigPath;
4use crate::tix::utils::OutputType;
5use tix_sdk::SdkError;
6use tix_sdk::document::TixDocument;
7
8/// Read a value from the config document
9#[derive(clap::Args, Debug)]
10pub struct Args {
11    /// The `<section>[.<key>]` path(s) to read (e.g. `defaults.branch_prefix`)
12    #[arg(required = true)]
13    pub key: Vec<ConfigPath>,
14}
15
16/// Reads each requested path out of the document and prints what is there.
17///
18/// A single value prints bare — a string without its quotes, everything
19/// else as the TOML literal `tix config set` would take back, so a list key
20/// round-trips through the two commands. Anything else (several paths, or
21/// one naming a whole table) prints as TOML at its real position in the
22/// document, `[defaults]` header and all, which is also what makes the JSON
23/// shape mirror the file rather than flattening it.
24///
25/// A path that is valid but unset errors: absent and empty are different,
26/// and a script reading a key that was never written should hear about it.
27pub fn run(app: &crate::tix::utils::App, args: Args) -> Result<(), SdkError> {
28    let document = TixDocument::load(&app.context.config_path)?;
29
30    let mut found = Vec::with_capacity(args.key.len());
31    for path in &args.key {
32        let mut item = document.doc().as_item();
33        for segment in path.segments() {
34            item = item.get(segment).ok_or_else(|| {
35                SdkError::Message(format!(
36                    "'{path}' is not set (config: {})",
37                    app.context.config_path.display()
38                ))
39            })?;
40        }
41        found.push((path, item.clone()));
42    }
43
44    // A lone value is the only output that is not a document fragment.
45    if let ([(_, item)], OutputType::Default) = (found.as_slice(), app.output)
46        && let Some(text) = bare_value(item)
47    {
48        println!("{text}");
49        return Ok(());
50    }
51
52    let extract = rebuild(&found)?;
53    match app.output {
54        // A table cloned out of the document brings its own blank-line
55        // prefix; the extract starts at its first line.
56        OutputType::Default | OutputType::Toml => {
57            print!("{}", extract.to_string().trim_start_matches('\n'))
58        }
59        OutputType::Json => {
60            let value: toml::Value = toml::from_str(&extract.to_string())?;
61            let json = serde_json::to_string_pretty(&value)
62                .map_err(|e| SdkError::Message(format!("json conversion failed: {e}")))?;
63            println!("{json}");
64        }
65    }
66    Ok(())
67}
68
69/// A value's natural text: a string's own characters, unquoted, and every
70/// other value's TOML literal. `None` for a table, which is a piece of
71/// document rather than a value and has no bare form.
72fn bare_value(item: &toml_edit::Item) -> Option<String> {
73    match item.as_value()? {
74        toml_edit::Value::String(text) => Some(text.value().to_string()),
75        value => Some(value.to_string().trim().to_string()),
76    }
77}
78
79/// Rebuilds the requested items into a document of their own, each at the
80/// path it was read from.
81///
82/// Reconstructing the position is what keeps the output honest: a table
83/// printed on its own would carry sub-headers relative to nothing, and a
84/// value would print a dotted key TOML reads back as a different shape.
85/// Rebuilt this way, the output is a valid fragment of the config file, and
86/// the JSON conversion comes for free by parsing it.
87///
88/// # Errors
89///
90/// [`SdkError::Message`] if two requested paths disagree about the document
91/// shape (`defaults` and `defaults.branch_prefix.x`, say) — the same
92/// non-table diagnostic any nested write raises.
93fn rebuild(found: &[(&ConfigPath, toml_edit::Item)]) -> Result<TixDocument, SdkError> {
94    let mut extract = TixDocument::empty();
95    for (path, item) in found {
96        let segments = path.segments();
97        let (leaf, parents) = segments.split_last().expect("non-empty by construction");
98        extract.table_at(parents)?.insert(leaf, item.clone());
99    }
100    Ok(extract)
101}
