1//! `tix config set` — write one key anywhere in the config document,
2//! through the format-preserving document layer.
3
4use crate::tix::config::{ConfigKeyPath, validate_section};
5use tix_sdk::SdkError;
6use tix_sdk::document::with_write;
7
8/// Set a value in the config document
9#[derive(clap::Args, Debug)]
10pub struct Args {
11    /// The `<section>.<key>` path to set (e.g. `defaults.branch_prefix`)
12    pub key: ConfigKeyPath,
13
14    /// The new value (parsed as a TOML value when possible, else a string)
15    pub value: String,
16}
17
18/// Sets `<section>.<key>` and persists atomically.
19///
20/// The edit is a targeted path traversal over the format-preserving DOM —
21/// plugin tables, comments, and formatting elsewhere in the file survive
22/// byte-identical (applies to the CLI itself, not just plugin deltas).
23/// Missing levels along the path materialize as real headered tables via
24/// [`tix_sdk::document::TixDocument::table_at`], never as the collapsed
25/// inline form bare indexing would produce (#146).
26///
27/// Before writing, the edited section is re-deserialized into the type that
28/// owns it ([`validate_section`]); a value that type rejects fails the whole
29/// write, and nothing lands on disk ([`with_write`] discards on error). A
30/// `[<plugin>]` table has no such type and is written as given.
31///
32/// The whole cycle runs under the exclusive advisory lock (#67), so
33/// concurrent `tix config set` invocations serialize rather than clobber.
34pub fn run(app: &crate::tix::utils::App, args: Args) -> Result<(), SdkError> {
35    with_write(&app.context.config_path, |document| {
36        // `1` stays an integer, `true` a bool, `["a", "b"]` an array — the
37        // shape a list-valued key needs until add/remove verbs exist.
38        // Anything that isn't valid TOML (a bare path, say) is a string.
39        let value: toml_edit::Value = args
40            .value
41            .parse()
42            .unwrap_or_else(|_| toml_edit::Value::from(args.value.as_str()));
43        document.table_at(&args.key.table_path())?[args.key.leaf()] = toml_edit::Item::Value(value);
44
45        validate_section(document, args.key.section())
46    })?;
47    println!(
48        "{} = {} ({})",
49        args.key,
50        args.value,
51        app.context.config_path.display()
52    );
53    Ok(())
54}
