39/// Which shape the target argument took.
40///
41/// The two forms agree on everything except where missing directories may be
42/// invented — see [`create_ticket_root`].
43#[derive(Debug, Clone, Copy, PartialEq, Eq)]
44enum Form {
45    /// A bare key, placed under the configured tickets directory.
46    Key,
47    /// An explicit path, placed exactly where it points.
48    Path,
49}
50
51/// Where a ticket will be created, and the key its document will record.
52#[derive(Debug, PartialEq, Eq)]
53struct Target {
54    /// The directory to create: `<tickets_directory>/<key>` for the key
55    /// form, the resolved argument for the path form.
56    root: PathBuf,
57    /// The ticket key — the argument itself, the directory name, or `--key`.
58    key: String,
59    /// The shape the argument took.
60    form: Form,
61}
62
63/// Resolves the creation target: where the ticket directory goes, and what
64/// key it records.
65///
66/// Shape decides, exactly as `--ticket` is disambiguated ([`TicketRef`],
67/// `design/spec.md` §4) — one rule for finding tickets and creating them:
68///
69/// - **Key form** (a bare name): created at `tickets_directory.join(key)`,
70///   v2 parity, and the key is the argument.
71/// - **Path form** (contains a separator, is absolute, or is `.`/`..`):
72///   created at that path, wherever it points. The key defaults to the
73///   directory name, which is the only thing the key form could ever have
74///   given it; `--key` overrides when the directory should be named for
75///   humans and the ticket for the tracker.
76///
77/// Relative paths resolve against `cwd`, which callers pass as the
78/// **logical** cwd: never canonicalized, so a ticket created through a
79/// symlink keeps the identity the user sees (the discovery rule, applied at
80/// creation).
81///
82/// # Errors
83///
84/// [`SdkError::Message`] when the key is empty or unusable as a directory
85/// name, when a path names no directory to create (`/`, `..` — the argument
86/// must name the ticket directory itself, not the place it goes), or when
87/// `--key` accompanies the key form, where it could only contradict the
88/// argument that already names the ticket.
89fn resolve_target(
90    target: &TicketRef,
91    key_override: Option<&str>,
92    tickets_directory: &Path,
93    cwd: &Path,
94) -> Result<Target, SdkError> {
95    match target {
96        TicketRef::Id(key) => {
97            if let Some(override_key) = key_override {
98                return Err(SdkError::Message(format!(
99                    "--key '{override_key}' conflicts with '{key}', which already names the ticket — \
100                     pass a path (e.g. './{key}') to give the directory a name of its own"
101                )));
102            }
103            let key = validated_key(key)?;
104            Ok(Target {
105                root: tickets_directory.join(&key),
106                key,
107                form: Form::Key,
108            })
109        }
110        TicketRef::Path(path) => {
111            let root = if path.is_absolute() {
112                drop_current_dir_components(path)
113            } else {
114                drop_current_dir_components(&cwd.join(path))
115            };
116            let key = match key_override {
117                Some(key) => validated_key(key)?,
118                None => {
119                    let name = root.file_name().ok_or_else(|| {
120                        SdkError::Message(format!(
121                            "'{}' does not name a directory to create — give the path the ticket \
122                             itself should occupy, e.g. './client-work/JIRA-123'",
123                            path.display()
124                        ))
125                    })?;
126                    validated_key(&name.to_string_lossy())?
127                }
128            };
129            Ok(Target {
130                root,
131                key,
132                form: Form::Path,
133            })
134        }
135    }
136}

138/// Drops `.` components, so `./client-work/JIRA-1` prints and is recorded as
139/// the location it actually names.
140///
141/// Only `.` — it resolves to the directory it sits in whatever that directory
142/// is, so removing it cannot change which path is meant. `..` is deliberately
143/// left in place: collapsing it lexically would resolve *through* a symlink
144/// and pick a different directory than the shell would (`design/spec.md` §4).
145fn drop_current_dir_components(path: &Path) -> PathBuf {
146    path.components()
147        .filter(|component| !matches!(component, std::path::Component::CurDir))
148        .collect()
149}
150
151/// Returns `key` if it can serve as both a ticket key and a directory name.
152///
153/// The key form cannot produce most of these (a name with a separator parses
154/// as a path, not a key), but `--key` takes arbitrary text and the check is
155/// what keeps it from writing a ticket that no directory can hold.
156fn validated_key(key: &str) -> Result<String, SdkError> {
157    if key.is_empty() {
158        return Err(SdkError::Message(
159            "the ticket key is empty — it becomes a directory name".to_string(),
160        ));
161    }
162    if key.contains(['/', '\\']) {
163        return Err(SdkError::Message(format!(
164            "'{key}' is not a valid ticket key — it becomes a directory name, so '/' and '\\' are rejected"
165        )));
166    }
167    if key == "." || key == ".." {
168        return Err(SdkError::Message(format!(
169            "'{key}' is not a valid ticket key — it becomes a directory name"
170        )));
171    }
172    Ok(key.to_string())
173}

175/// Creates the ticket directory, refusing to overwrite one and — in the path
176/// form — refusing to invent its parent.
177///
178/// The forms differ deliberately. `tickets_directory` is configured and
179/// intentional, so the key form creates it when it is missing. An arbitrary
180/// path is neither: a typo (`./cleint-work/JIRA-1`) would otherwise grow a
181/// tree of worktrees somewhere the user never meant, and unwinding that costs
182/// more than the `mkdir -p` the error asks for. Creating parents is the
183/// loosening we could still make later; requiring them is not.
184fn create_ticket_root(target: &Target) -> Result<(), SdkError> {
185    if target.root.exists() {
186        return Err(SdkError::Message(format!(
187            "ticket '{}' already exists at {}",
188            target.key,
189            target.root.display()
190        )));
191    }
192    match target.form {
193        Form::Key => std::fs::create_dir_all(&target.root).map_err(SdkError::from),
194        Form::Path => {
195            let parent = target.root.parent().ok_or_else(|| {
196                SdkError::Message(format!(
197                    "{} has no parent directory to create the ticket in",
198                    target.root.display()
199                ))
200            })?;
201            if !parent.is_dir() {
202                return Err(SdkError::Message(format!(
203                    "{} does not exist — create it first (`mkdir -p`), or pass a bare key to \
204                     create the ticket under the tickets directory",
205                    parent.display()
206                )));
207            }
208            std::fs::create_dir(&target.root).map_err(SdkError::from)
209        }
210    }
211}

213/// Creates a ticket workspace with a worktree per selected repo.
214///
215/// The target argument is disambiguated by shape (`resolve_target`): a bare
216/// key lands under `tickets_directory`, a path lands where it points. Both
217/// forms are otherwise identical — same seeds, same derivation, same
218/// document.
219///
220/// Runs **without ticket context** by design — this is the command that
221/// creates it. `[defaults]` is read **once**, here, and the derived values
222/// are written into the ticket document; later changes to `[defaults]` never
223/// touch this ticket (see
224/// [creation-time seeds](https://tix.armaanv.dev/latest/concepts/seeds/)).
225///
226/// Repo selection: explicit aliases win, `--all` takes every registered
227/// repo, and neither falls back to the `defaults.repositories` seed list.
228///
229/// Partial failure leaves successfully created worktrees in place and
230/// records only them in `.tix/ticket.toml`; the command then errors naming
231/// the repos that failed.
232pub fn run(app: &crate::tix::utils::App, args: Args) -> Result<(), SdkError> {
233    let document = TixDocument::load(&app.context.config_path)?;
234    let cli: CliConfig = document.section("cli")?.ok_or_else(|| {
235        SdkError::Message("global config has no [cli] section — run `tix config init`".to_string())
236    })?;
237    let engine: EngineConfig = document.section_or_default("engine")?;
238    let defaults: Defaults = document.section_or_default("defaults")?;
239
240    let target = resolve_target(
241        &args.target,
242        args.key.as_deref(),
243        &cli.tickets_directory,
244        &tix_sdk::discovery::logical_cwd()?,
245    )?;
246
247    // --- select repositories ---
248    let selected: Vec<String> = if args.all {
249        let mut aliases: Vec<String> = engine.configured_repositories.keys().cloned().collect();
250        aliases.sort();
251        aliases
252    } else if !args.repo_aliases.is_empty() {
253        args.repo_aliases
254            .iter()
255            .map(|alias| alias.0.clone())
256            .collect()
257    } else {
258        defaults.repositories.clone()
259    };
260    for alias in &selected {
261        if !engine.configured_repositories.contains_key(alias) {
262            let mut known: Vec<&str> = engine
263                .configured_repositories
264                .keys()
265                .map(String::as_str)
266                .collect();
267            known.sort();
268            return Err(SdkError::Engine(TixError::RepoNotFound(format!(
269                "'{alias}' is not a registered repository (registered: {})",
270                if known.is_empty() {
271                    "none".to_string()
272                } else {
273                    known.join(", ")
274                }
275            ))));
276        }
277    }
278
279    // --- seed reads: derivation happens once, now ---
280    let branch = derive_branch_name(
281        defaults.branch_prefix.as_deref(),
282        &target.key,
283        args.description.as_deref(),
284    );
285    let description = args.description.clone().unwrap_or_default();
286
287    create_ticket_root(&target)?;
288    let ticket_root = &target.root;
289
290    // --- create worktrees; collect successes and failures ---
291    let mut worktrees: HashMap<String, WorktreeConfig> = HashMap::new();
292    let mut failures: Vec<(String, TixError)> = Vec::new();
293    for alias in &selected {
294        let repo_config = engine.configured_repositories[alias].clone();
295        let result = repo_config
296            .ensure(alias)
297            .and_then(|repo| repo.create_worktree(alias, &branch, &ticket_root.join(alias), false));
298        match result {
299            Ok(worktree) => {
300                info!(alias = %alias, branch = %worktree.branch, "worktree created");
301                worktrees.insert(
302                    alias.clone(),
303                    WorktreeConfig {
304                        repo: alias.clone(),
305                        branch: worktree.branch,
306                    },
307                );
308            }
309            Err(e) => {
310                error!(alias = %alias, error = %e, "failed to create worktree");
311                failures.push((alias.clone(), e));
312            }
313        }
314    }
315
316    // --- write the ticket document, recording only what actually exists ---
317    let ticket = TicketConfig {
318        key: target.key.clone(),
319        description,
320        worktrees,
321    };
322    let mut ticket_document = TixDocument::empty();
323    ticket_document.set_section("ticket", &ticket)?;
324    ticket_document.save(&ticket_root.join(".tix").join("ticket.toml"))?;
325
326    println!("{}", ticket_root.display());
327
328    if !failures.is_empty() {
329        let names: Vec<&str> = failures.iter().map(|(alias, _)| alias.as_str()).collect();
330        return Err(SdkError::Message(format!(
331            "ticket created, but {} of {} worktrees failed: {} — fix and `tix ticket add` them",
332            failures.len(),
333            selected.len(),
334            names.join(", ")
335        )));
336    }
337    Ok(())
338}

345    /// The tickets directory and cwd used by the resolution tests, distinct
346    /// so a target landing in the wrong one is visible.
347    const TICKETS: &str = "/home/u/tickets";
348    const CWD: &str = "/home/u/work";
349
350    fn resolve(arg: &str, key: Option<&str>) -> Result<Target, SdkError> {
351        let target: TicketRef = arg.parse().unwrap();
352        resolve_target(&target, key, Path::new(TICKETS), Path::new(CWD))
353    }

363        assert_eq!(target.key, "JIRA-123");
