3/// The `[defaults]` section of the global config — **seeds, not defaults**.
4///
5/// Values here are read **once** by `tix ticket setup` at ticket creation and
6/// written into the ticket document. Changing a global value later affects
7/// only new tickets; there is no runtime override/fallback resolver anywhere
8/// in tix. Anything that shapes git state must be a seed — topology on disk
9/// cannot be retroactively rewritten by config
10/// (see [creation-time seeds](https://tix.armaanv.dev/latest/concepts/seeds/)).
11///
12/// Every field is optional in the document: a partial (or absent) `[defaults]`
13/// section is normal, and [`Default`] gives the all-empty value for the
14/// `section_or_default` read path.
15///
16/// The field set adopts v2's provisionally and may expand or retract as
17/// `tix ticket setup` takes shape.
18///
19/// # Examples
20///
21/// A partial `[defaults]` section parses; unset fields stay `None`/empty:
22///
23/// ```
24/// # use tix_engine::Defaults;
25/// let defaults: Defaults = toml::from_str(
26///     r#"
27///     branch_prefix = "feature"
28///     repositories = ["backend", "frontend"]
29///     "#,
30/// )
31/// .unwrap();
32///
33/// assert_eq!(defaults.branch_prefix.as_deref(), Some("feature"));
34/// assert_eq!(defaults.repositories, vec!["backend", "frontend"]);
35/// assert!(defaults.github_base_url.is_none());
36/// assert!(defaults.default_repository_owner.is_none());
37/// ```
38///
39/// An empty document is the same as no section at all:
40///
41/// ```
42/// # use tix_engine::Defaults;
43/// let defaults: Defaults = toml::from_str("").unwrap();
44/// assert_eq!(defaults, Defaults::default());
45/// ```
46#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
47#[derive(Serialize, Deserialize, Clone, Debug, Default, PartialEq)]
48#[serde(deny_unknown_fields)]
49pub struct Defaults {
50    /// Seed for branch name derivation at `tix ticket setup`:
51    /// `<prefix>/<key>-<sanitized-description>`.
52    #[serde(default, skip_serializing_if = "Option::is_none")]
53    pub branch_prefix: Option<String>,
54    /// Seed for remote URL construction.
55    #[serde(default, skip_serializing_if = "Option::is_none")]
56    pub github_base_url: Option<String>,
57    /// Seed for remote URL construction: the owner used when a repository is
58    /// given without one.
59    #[serde(default, skip_serializing_if = "Option::is_none")]
60    pub default_repository_owner: Option<String>,
61    /// Which registered repositories a new ticket includes.
62    ///
63    /// Seeds a *new* ticket. The ticket document's worktree map records what
64    /// a ticket actually has; changing this leaves existing tickets alone.
65    #[serde(default, skip_serializing_if = "Vec::is_empty")]
66    pub repositories: Vec<String>,
67}

69#[cfg(test)]
70mod tests {
71    use super::*;
72
73    fn sample_defaults() -> Defaults {
74        Defaults {
75            branch_prefix: Some("feature".to_string()),
76            github_base_url: Some("https://github.mycompany.com".to_string()),
77            default_repository_owner: Some("my-org".to_string()),
78            repositories: vec!["backend".to_string(), "frontend".to_string()],
79        }
80    }
81
82    /// A fully populated `[defaults]` section deserializes correctly.
83    #[test]
84    fn test_deserialize_full_section() {
85        let toml = r#"
86branch_prefix = "feature"
87github_base_url = "https://github.mycompany.com"
88default_repository_owner = "my-org"
89repositories = ["backend", "frontend"]
90"#;
91        let defaults: Defaults = toml::from_str(toml).unwrap();
92        assert_eq!(defaults, sample_defaults());
93    }
94
95    /// Serializing and deserializing preserves all data.
96    #[test]
97    fn test_round_trip() {
98        let defaults = sample_defaults();
99        let restored: Defaults = toml::from_str(&toml::to_string(&defaults).unwrap()).unwrap();
100        assert_eq!(restored, defaults);
101    }
102
103    /// Unset optional fields serialize to nothing rather than explicit nulls,
104    /// so a default value round-trips through an empty document.
105    #[test]
106    fn test_default_serializes_empty() {
107        assert_eq!(toml::to_string(&Defaults::default()).unwrap(), "");
108    }
109
110    /// An empty document parses to the all-empty `Default` value.
111    #[test]
112    fn test_empty_section() {
113        let defaults: Defaults = toml::from_str("").unwrap();
114        assert_eq!(defaults, Defaults::default());
115    }
116
117    /// Unknown fields in the `[defaults]` section are rejected.
118    #[test]
119    fn test_rejects_unknown_fields() {
120        let toml = r#"
121branch_prefix = "feature"
122base_branch = "not-a-seed-we-know"
123"#;
124        assert!(toml::from_str::<Defaults>(toml).is_err());
125    }
126}
