4/// The recorded state of one worktree in a ticket document — the value type of
5/// [`TicketConfig::worktrees`](crate::TicketConfig::worktrees).
6///
7/// The map key (the worktree directory name under the ticket root) lives in
8/// [`TicketConfig`](crate::TicketConfig); this struct carries what that
9/// directory *is*: which repository it belongs to and which branch it has
10/// checked out. Repos in one ticket may sit on different branches — worktrees
11/// share a branch *prefix*, not a branch.
12///
13/// # Examples
14///
15/// As TOML, an entry appears as a sub-table of `worktrees` named after its
16/// directory:
17///
18/// ```
19/// # use tix_engine::WorktreeConfig;
20/// let entry: WorktreeConfig = toml::from_str(
21///     r#"
22///     repo = "backend"
23///     branch = "feature/JIRA-123-fix-login"
24///     "#,
25/// )
26/// .unwrap();
27/// assert_eq!(entry.repo, "backend");
28/// assert_eq!(entry.branch, "feature/JIRA-123-fix-login");
29/// ```
30#[derive(Serialize, Deserialize, Clone, Debug, PartialEq)]
31#[serde(deny_unknown_fields)]
32pub struct WorktreeConfig {
33    /// The alias of the repository this worktree belongs to.
34    pub repo: String,
35    /// The branch checked out in this worktree.
36    pub branch: String,
37}

39/// A live git worktree associated with a repository and ticket branch.
40///
41/// Produced by [`TicketConfig::resolve`](crate::TicketConfig::resolve) (from
42/// recorded state verified against disk) or by
43/// [`Repository::create_worktree`](crate::Repository::create_worktree) (from a
44/// worktree it just created). Holding one means the worktree existed at the
45/// time it was produced.
46#[derive(Serialize, Debug, Clone, PartialEq)]
47pub struct Worktree {
48    /// The worktree directory name under the ticket root — the key of this
49    /// worktree's entry in [`TicketConfig::worktrees`](crate::TicketConfig::worktrees).
50    pub name: String,
51    /// The alias of the repository.
52    pub repo_alias: String,
53    /// The path to the worktree directory.
54    pub path: PathBuf,
55    /// The branch of the worktree.
56    pub branch: String,
57}
