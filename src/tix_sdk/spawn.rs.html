1//! Nested `tix` invocation helper (`design/spec.md` §5.5).
2//!
3//! A plugin may shell out to `tix`, but a nested host re-discovers from cwd
4//! — which the plugin may have changed — so nested `tix` can silently
5//! resolve a *different ticket*. Correct (cwd is the interface) but a
6//! footgun for authors who mean "same ticket". The supported path is
7//! skew-proof by construction: [`tix_command`] pins the current ticket.
8
9use std::path::Path;
10use std::process::Command;
11
12/// Builds a `Command` for a nested `tix` invocation pinned to `ticket_root`.
13///
14/// The returned command already carries `--ticket <path>`, so it resolves
15/// the same ticket regardless of what cwd has become. Callers add their
16/// subcommand and args and spawn as usual.
17///
18/// The fork-bomb guard (`TIX_DEPTH`) is the *host's* job on dispatch —
19/// nothing to set here.
20///
21/// # Examples
22///
23/// ```no_run
24/// # use std::path::Path;
25/// let mut command = tix_sdk::spawn::tix_command(Path::new("/home/user/tickets/JIRA-1"));
26/// command.args(["ticket", "info"]);
27/// let status = command.status().expect("tix not on PATH?");
28/// ```
29pub fn tix_command(ticket_root: &Path) -> Command {
30    let mut command = Command::new("tix");
31    command.arg("--ticket").arg(ticket_root);
32    command
33}
34
35#[cfg(test)]
36mod tests {
37    use super::*;
38
39    /// The pin rides at the front of the arg list.
40    #[test]
41    fn test_pins_ticket() {
42        let command = tix_command(Path::new("/tickets/JIRA-1"));
43        let args: Vec<_> = command
44            .get_args()
45            .map(|a| a.to_string_lossy().to_string())
46            .collect();
47        assert_eq!(args, vec!["--ticket", "/tickets/JIRA-1"]);
48        assert_eq!(command.get_program(), "tix");
49    }
50}
