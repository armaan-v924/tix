1//! The SDK's error type.
2//!
3//! Document parsing and serialization are SDK concerns (`design/spec.md`
4//! §2.2) — the `ParseError`/`SerializationError` variants that used to live
5//! on the engine's [`TixError`] migrate here, so `tix-engine` carries no
6//! runtime `toml` dependency.
7
8use std::fmt;
9use tix_engine::TixError;
10
11/// An error from the SDK's context-and-consistency layer, or from the
12/// engine underneath it.
13pub enum SdkError {
14    /// An engine operation failed.
15    Engine(TixError),
16    /// A TOML document could not be parsed.
17    Parse(toml::de::Error),
18    /// A value could not be serialized to TOML.
19    Serialization(toml::ser::Error),
20    /// The config file was not found at the resolved path.
21    ///
22    /// Migrated from the engine: config location is an SDK concern, and the
23    /// engine never touches config paths.
24    ConfigNotFound(String),
25    /// A plugin misused the invocation contract (malformed delta, broken
26    /// revalidation) — a bug in the plugin, not a tix failure, and nothing
27    /// was written.
28    PluginImplementation(String),
29    /// A freeform SDK-level error message.
30    Message(String),
31}
32
33impl fmt::Display for SdkError {
34    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
35        match self {
36            SdkError::Engine(e) => fmt::Display::fmt(e, f),
37            SdkError::Parse(e) => write!(f, "parse error: {}", e),
38            SdkError::Serialization(e) => write!(f, "serialization error: {}", e),
39            SdkError::ConfigNotFound(s) => write!(f, "config not found: {}", s),
40            SdkError::PluginImplementation(s) => write!(
41                f,
42                "plugin implementation error: {} — this is a bug in the plugin; nothing was written",
43                s
44            ),
45            SdkError::Message(s) => write!(f, "{}", s),
46        }
47    }
48}
49
50impl fmt::Debug for SdkError {
51    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
52        fmt::Display::fmt(self, f)
53    }
54}
55
56impl std::error::Error for SdkError {}
57
58impl From<TixError> for SdkError {
59    fn from(e: TixError) -> Self {
60        SdkError::Engine(e)
61    }
62}
63
64impl From<toml::de::Error> for SdkError {
65    fn from(e: toml::de::Error) -> Self {
66        SdkError::Parse(e)
67    }
68}
69
70impl From<toml::ser::Error> for SdkError {
71    fn from(e: toml::ser::Error) -> Self {
72        SdkError::Serialization(e)
73    }
74}
75
76impl From<std::io::Error> for SdkError {
77    fn from(e: std::io::Error) -> Self {
78        SdkError::Engine(TixError::IoError(e))
79    }
80}
81
82impl From<String> for SdkError {
83    fn from(s: String) -> Self {
84        SdkError::Message(s)
85    }
86}
87
88impl From<&str> for SdkError {
89    fn from(s: &str) -> Self {
90        SdkError::Message(s.to_string())
91    }
92}
