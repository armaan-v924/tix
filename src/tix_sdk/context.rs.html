1//! Frontend context: paths resolved once at startup and handed to every
2//! subcommand.
3//!
4//! Config path resolution is stage 1 of the read path (`design/spec.md`
5//! §3.3) and belongs to the frontend/SDK — the engine never reads env vars
6//! or flags. Written in `tix-cli` first (#52); promoted here (#96) so plugins
7//! resolve identically.
8//!
9//! The default location is deliberately isolated in [`default_config_path`]
10//! so it can be revisited without touching resolution or call sites.
11
12use crate::error::SdkError;
13use std::ffi::OsString;
14use std::path::PathBuf;
15use tracing::debug;
16
17/// The environment variable overriding the global config location, beaten
18/// only by the `--config` flag.
19pub const CONFIG_PATH_ENV: &str = "TIX_CONFIG_PATH";
20
21/// Paths every subcommand receives, resolved once at startup.
22#[derive(Debug, Clone, PartialEq)]
23pub struct Context {
24    /// The resolved global config path. Resolution picks the *location*; the
25    /// file may not exist yet (`tix config init` creates it).
26    pub config_path: PathBuf,
27}
28
29impl Context {
30    /// Builds the context from the parsed top-level arguments.
31    ///
32    /// # Errors
33    ///
34    /// [`SdkError::ConfigNotFound`] if no flag or env var is set and the
35    /// platform config directory cannot be determined.
36    pub fn resolve(config_flag: Option<PathBuf>) -> Result<Self, SdkError> {
37        Ok(Self {
38            config_path: resolve_config_path(config_flag)?,
39        })
40    }
41
42    /// The config path, verified to exist on disk.
43    ///
44    /// Subcommands that require config call this; `tix config init` and
45    /// `tix cli completions` use [`Self::config_path`] directly.
46    ///
47    /// # Errors
48    ///
49    /// [`SdkError::ConfigNotFound`] with the resolved path and a pointer at
50    /// `tix config init` when the file does not exist.
51    pub fn require_config_path(&self) -> Result<&PathBuf, SdkError> {
52        if !self.config_path.is_file() {
53            return Err(SdkError::ConfigNotFound(format!(
54                "no config file at {} — run `tix config init` to create one",
55                self.config_path.display()
56            )));
57        }
58        Ok(&self.config_path)
59    }
60}
61
62/// Resolves the global config path with the precedence
63/// `--config` flag > `TIX_CONFIG_PATH` > platform default.
64///
65/// Resolution picks a location; it does not require the file to exist —
66/// existence is enforced per-subcommand via [`Context::require_config_path`],
67/// since `tix config init` must be able to run *before* the file exists.
68///
69/// # Errors
70///
71/// [`SdkError::ConfigNotFound`] if neither flag nor env var is set and the
72/// platform config directory cannot be determined.
73pub fn resolve_config_path(flag: Option<PathBuf>) -> Result<PathBuf, SdkError> {
74    resolve_from(
75        flag,
76        std::env::var_os(CONFIG_PATH_ENV),
77        default_config_path(),
78    )
79}
80
81/// The platform default config location: `dirs::config_dir()/tix/config.toml`
82/// (`~/Library/Application Support/tix/config.toml` on macOS,
83/// `$XDG_CONFIG_HOME/tix/config.toml` on Linux).
84///
85/// The single place the default lives — change it here, nowhere else.
86pub fn default_config_path() -> Option<PathBuf> {
87    dirs::config_dir().map(|dir| dir.join("tix").join("config.toml"))
88}
89
90/// Pure precedence logic, separated from ambient env access for testability.
91///
92/// An env var that is set but empty is ignored — `TIX_CONFIG_PATH= tix ...`
93/// falls through to the default rather than resolving to an empty path.
94fn resolve_from(
95    flag: Option<PathBuf>,
96    env: Option<OsString>,
97    default: Option<PathBuf>,
98) -> Result<PathBuf, SdkError> {
99    if let Some(path) = flag {
100        debug!(path = %path.display(), "config path from --config flag");
101        return Ok(path);
102    }
103    if let Some(value) = env
104        && !value.is_empty()
105    {
106        let path = PathBuf::from(value);
107        debug!(path = %path.display(), "config path from {CONFIG_PATH_ENV}");
108        return Ok(path);
109    }
110    default.ok_or_else(|| {
111        SdkError::ConfigNotFound(
112            "cannot determine the platform config directory; \
113             pass --config or set TIX_CONFIG_PATH"
114                .to_string(),
115        )
116    })
117}

119#[cfg(test)]
120mod tests {
121    use super::*;
122
123    fn flag(s: &str) -> Option<PathBuf> {
124        Some(PathBuf::from(s))
125    }
126
127    /// The `--config` flag beats the env var and the default.
128    #[test]
129    fn test_flag_wins() {
130        let resolved = resolve_from(
131            flag("/from/flag.toml"),
132            Some(OsString::from("/from/env.toml")),
133            Some(PathBuf::from("/from/default.toml")),
134        )
135        .unwrap();
136        assert_eq!(resolved, PathBuf::from("/from/flag.toml"));
137    }
138
139    /// The env var beats the default when no flag is given.
140    #[test]
141    fn test_env_beats_default() {
142        let resolved = resolve_from(
143            None,
144            Some(OsString::from("/from/env.toml")),
145            Some(PathBuf::from("/from/default.toml")),
146        )
147        .unwrap();
148        assert_eq!(resolved, PathBuf::from("/from/env.toml"));
149    }
150
151    /// An empty env var is ignored rather than resolving to an empty path.
152    #[test]
153    fn test_empty_env_ignored() {
154        let resolved = resolve_from(
155            None,
156            Some(OsString::new()),
157            Some(PathBuf::from("/from/default.toml")),
158        )
159        .unwrap();
160        assert_eq!(resolved, PathBuf::from("/from/default.toml"));
161    }
162
163    /// With neither flag nor env var, the platform default applies.
164    #[test]
165    fn test_default_applies() {
166        let resolved = resolve_from(None, None, Some(PathBuf::from("/from/default.toml"))).unwrap();
167        assert_eq!(resolved, PathBuf::from("/from/default.toml"));
168    }
169
170    /// No flag, no env var, no determinable platform directory is an error.
171    #[test]
172    fn test_no_source_errors() {
173        assert!(matches!(
174            resolve_from(None, None, None),
175            Err(SdkError::ConfigNotFound(_))
176        ));
177    }
178
179    /// The platform default ends in tix/config.toml.
180    #[test]
181    fn test_default_shape() {
182        if let Some(default) = default_config_path() {
183            assert!(default.ends_with("tix/config.toml"));
184        }
185    }
186
187    /// require_config_path errors for a missing file and passes for a
188    /// present one.
189    #[test]
190    fn test_require_config_path() {
191        let dir = tempfile::tempdir().unwrap();
192        let path = dir.path().join("config.toml");
193        let ctx = Context {
194            config_path: path.clone(),
195        };
196
197        assert!(matches!(
198            ctx.require_config_path(),
199            Err(SdkError::ConfigNotFound(_))
200        ));
201
202        std::fs::write(&path, "").unwrap();
203        assert_eq!(ctx.require_config_path().unwrap(), &path);
204    }
205}
