1//! `tix ticket info` — display a ticket's metadata and worktree table.
2
3use crate::tix::ticket::{TicketSharedArgs, load_ticket_config, require_ticket_root};
4use crate::tix::utils::OutputType;
5use std::path::{Path, PathBuf};
6use tix_sdk::{SdkError, TicketConfig};
7
8/// Show a ticket's metadata and worktrees
9#[derive(clap::Args, Debug)]
10pub struct Args {
11    #[command(flatten)]
12    pub shared: TicketSharedArgs,
13
14    /// Output format
15    #[arg(short, long)]
16    pub output: Option<OutputType>,
17}
18
19/// One row of the worktree table: recorded state plus its on-disk status.
20struct WorktreeRow {
21    name: String,
22    repo: String,
23    branch: String,
24    path: PathBuf,
25    status: WorktreeStatus,
26}
27
28/// Whether a recorded worktree actually resolves on disk.
29enum WorktreeStatus {
30    Ok,
31    MissingDirectory,
32    NotAGitRepository,
33}
34
35impl WorktreeStatus {
36    fn of(path: &Path) -> Self {
37        if !path.is_dir() {
38            WorktreeStatus::MissingDirectory
39        } else if !tix_sdk::opens_as_git_repository(path) {
40            WorktreeStatus::NotAGitRepository
41        } else {
42            WorktreeStatus::Ok
43        }
44    }
45
46    fn marker(&self) -> &'static str {
47        match self {
48            WorktreeStatus::Ok => "",
49            WorktreeStatus::MissingDirectory => "  ⚠ missing on disk",
50            WorktreeStatus::NotAGitRepository => "  ⚠ not a git repository",
51        }
52    }
53
54    fn label(&self) -> &'static str {
55        match self {
56            WorktreeStatus::Ok => "ok",
57            WorktreeStatus::MissingDirectory => "missing",
58            WorktreeStatus::NotAGitRepository => "not-a-git-repository",
59        }
60    }
61}
62
63/// Prints key, description, path, and the per-repo worktree table.
64///
65/// Pure frontend over the recorded `[ticket]` section: each worktree is
66/// checked individually so one that fails to resolve renders with a warning
67/// marker instead of failing the whole command — which is also why this does
68/// not go through the all-or-nothing [`TicketConfig::resolve`].
69///
70/// There is deliberately no single ticket branch to print: branches are
71/// per-worktree (spec §3.2).
72pub fn run(app: &crate::tix::utils::App, args: Args) -> Result<(), SdkError> {
73    let root = require_ticket_root(&app.context, args.shared.ticket.as_ref())?;
74    let ticket: TicketConfig = load_ticket_config(&root)?;
75
76    // Grouped by repo, then by name within a repo.
77    let mut rows: Vec<WorktreeRow> = ticket
78        .worktrees
79        .iter()
80        .map(|(name, entry)| {
81            let path = root.join(name);
82            WorktreeRow {
83                name: name.clone(),
84                repo: entry.repo.clone(),
85                branch: entry.branch.clone(),
86                status: WorktreeStatus::of(&path),
87                path,
88            }
89        })
90        .collect();
91    rows.sort_by(|a, b| a.repo.cmp(&b.repo).then(a.name.cmp(&b.name)));
92
93    match app.output {
94        OutputType::Default => {
95            if ticket.description.is_empty() {
96                println!("{}", ticket.key);
97            } else {
98                println!("{}  {}", ticket.key, ticket.description);
99            }
100            println!("{}", root.display());
101            if rows.is_empty() {
102                println!("\nno worktrees");
103            } else {
104                let name_width = rows.iter().map(|r| r.name.len()).max().unwrap_or(0);
105                let repo_width = rows.iter().map(|r| r.repo.len()).max().unwrap_or(0);
106                println!();
107                for row in &rows {
108                    println!(
109                        "{:name_width$}  {:repo_width$}  {}{}",
110                        row.name,
111                        row.repo,
112                        row.branch,
113                        row.status.marker()
114                    );
115                }
116            }
117        }
118        OutputType::Json => {
119            let worktrees: Vec<serde_json::Value> = rows
120                .iter()
121                .map(|row| {
122                    serde_json::json!({
123                        "name": row.name,
124                        "repo": row.repo,
125                        "branch": row.branch,
126                        "path": row.path,
127                        "status": row.status.label(),
128                    })
129                })
130                .collect();
131            let value = serde_json::json!({
132                "key": ticket.key,
133                "description": ticket.description,
134                "path": root,
135                "worktrees": worktrees,
136            });
137            println!("{}", serde_json::to_string_pretty(&value).unwrap());
138        }
139        OutputType::Toml => {
140            println!("path = {:?}\n", root.display().to_string());
141            print!("{}", toml::to_string(&ticket)?);
142        }
143    }
144    Ok(())
145}
