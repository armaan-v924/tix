1//! `tix ticket setup` — create a new ticket workspace with worktrees.
2
3use crate::tix::config::CliConfig;
4use crate::tix::repo::RepoAlias;
5use crate::tix::ticket::derive_branch_name;
6use std::collections::HashMap;
7use std::path::PathBuf;
8use tix_sdk::document::TixDocument;
9use tix_sdk::{Defaults, EngineConfig, SdkError, TicketConfig, TixError, WorktreeConfig};
10use tracing::{error, info};
11
12/// Create a new ticket workspace with worktrees
13#[derive(clap::Args, Debug)]
14pub struct Args {
15    /// The ticket key, e.g. JIRA-123 (a name, not a path)
16    pub key: String,
17
18    /// Human-readable description; feeds branch name derivation
19    #[arg(short, long)]
20    pub description: Option<String>,
21
22    /// Include every registered repository
23    #[arg(short, long, group = "repos")]
24    pub all: bool,
25
26    /// Repositories to include (defaults to [defaults].repositories)
27    #[arg(group = "repos")]
28    pub repo_aliases: Vec<RepoAlias>,
29}
30
31/// Creates `<tickets_directory>/<key>/` with a worktree per selected repo.
32///
33/// Runs **without ticket context** by design — this is the command that
34/// creates it. `[defaults]` is read **once**, here, and the derived values
35/// are written into the ticket document; later changes to `[defaults]` never
36/// touch this ticket (seeds, not defaults — spec §3.4).
37///
38/// Repo selection: explicit aliases win, `--all` takes every registered
39/// repo, and neither falls back to the `defaults.repositories` seed list.
40///
41/// Partial failure leaves successfully created worktrees in place and
42/// records only them in `.tix/ticket.toml`; the command then errors naming
43/// the repos that failed.
44pub fn run(app: &crate::tix::utils::App, args: Args) -> Result<(), SdkError> {
45    if args.key.is_empty() || args.key.contains(['/', '\\']) {
46        return Err(SdkError::Message(format!(
47            "'{}' is not a valid ticket key — keys are names, not paths (path-form creation is #114)",
48            args.key
49        )));
50    }
51
52    let document = TixDocument::load(&app.context.config_path)?;
53    let cli: CliConfig = document.section("cli")?.ok_or_else(|| {
54        SdkError::Message("global config has no [cli] section — run `tix config init`".to_string())
55    })?;
56    let engine: EngineConfig = document.section_or_default("engine")?;
57    let defaults: Defaults = document.section_or_default("defaults")?;
58
59    // --- select repositories ---
60    let selected: Vec<String> = if args.all {
61        let mut aliases: Vec<String> = engine.configured_repositories.keys().cloned().collect();
62        aliases.sort();
63        aliases
64    } else if !args.repo_aliases.is_empty() {
65        args.repo_aliases
66            .iter()
67            .map(|alias| alias.0.clone())
68            .collect()
69    } else {
70        defaults.repositories.clone()
71    };
72    for alias in &selected {
73        if !engine.configured_repositories.contains_key(alias) {
74            let mut known: Vec<&str> = engine
75                .configured_repositories
76                .keys()
77                .map(String::as_str)
78                .collect();
79            known.sort();
80            return Err(SdkError::Engine(TixError::RepoNotFound(format!(
81                "'{alias}' is not a registered repository (registered: {})",
82                if known.is_empty() {
83                    "none".to_string()
84                } else {
85                    known.join(", ")
86                }
87            ))));
88        }
89    }
90
91    // --- seed reads: derivation happens once, now ---
92    let branch = derive_branch_name(
93        defaults.branch_prefix.as_deref(),
94        &args.key,
95        args.description.as_deref(),
96    );
97    let description = args.description.clone().unwrap_or_default();
98
99    let ticket_root: PathBuf = cli.tickets_directory.join(&args.key);
100    if ticket_root.exists() {
101        return Err(SdkError::Message(format!(
102            "ticket '{}' already exists at {}",
103            args.key,
104            ticket_root.display()
105        )));
106    }
107    std::fs::create_dir_all(&ticket_root).map_err(SdkError::from)?;
108
109    // --- create worktrees; collect successes and failures ---
110    let mut worktrees: HashMap<String, WorktreeConfig> = HashMap::new();
111    let mut failures: Vec<(String, TixError)> = Vec::new();
112    for alias in &selected {
113        let repo_config = engine.configured_repositories[alias].clone();
114        let result = repo_config
115            .ensure(alias)
116            .and_then(|repo| repo.create_worktree(alias, &branch, &ticket_root.join(alias), false));
117        match result {
118            Ok(worktree) => {
119                info!(alias = %alias, branch = %worktree.branch, "worktree created");
120                worktrees.insert(
121                    alias.clone(),
122                    WorktreeConfig {
123                        repo: alias.clone(),
124                        branch: worktree.branch,
125                    },
126                );
127            }
128            Err(e) => {
129                error!(alias = %alias, error = %e, "failed to create worktree");
130                failures.push((alias.clone(), e));
131            }
132        }
133    }
134
135    // --- write the ticket document, recording only what actually exists ---
136    let ticket = TicketConfig {
137        key: args.key.clone(),
138        description,
139        worktrees,
140    };
141    let mut ticket_document = TixDocument::empty();
142    ticket_document.set_section("ticket", &ticket)?;
143    ticket_document.save(&ticket_root.join(".tix").join("ticket.toml"))?;
144
145    println!("{}", ticket_root.display());
146
147    if !failures.is_empty() {
148        let names: Vec<&str> = failures.iter().map(|(alias, _)| alias.as_str()).collect();
149        return Err(SdkError::Message(format!(
150            "ticket created, but {} of {} worktrees failed: {} — fix and `tix ticket add` them",
151            failures.len(),
152            selected.len(),
153            names.join(", ")
154        )));
155    }
156    Ok(())
157}
