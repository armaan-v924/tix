1//! `tix ticket remove` — remove single worktrees from a ticket without
2//! destroying it.
3
4use crate::tix::ticket::{TicketSharedArgs, load_ticket_config, require_ticket_root};
5use tix_sdk::document::{TixDocument, with_write};
6use tix_sdk::{EngineConfig, SdkError, TixError};
7use tracing::info;
8
9/// Remove a worktree from a ticket without destroying it
10#[derive(clap::Args, Debug)]
11pub struct Args {
12    #[command(flatten)]
13    pub shared: TicketSharedArgs,
14
15    /// Worktree name(s) from the ticket's worktree map (a bare repo alias
16    /// matches its default worktree)
17    #[arg(required = true, value_name = "NAME")]
18    pub names: Vec<String>,
19}
20
21/// Prunes the named worktrees and deletes their entries from the ticket
22/// document.
23///
24/// `<name>` addresses the **worktree directory name** — the ticket map's key
25/// — not a repo alias: with #85 a repo can have several worktrees, so the
26/// alias alone is not an address. In the single-worktree case they coincide
27/// (`name == alias`), which is what makes bare-alias usage work today.
28///
29/// The ticket directory and every other worktree are untouched; removing
30/// the last worktree leaves a valid, empty ticket. Each successful prune is
31/// written back immediately through the format-preserving layer.
32pub fn run(app: &crate::tix::utils::App, args: Args) -> Result<(), SdkError> {
33    let root = require_ticket_root(&app.context, args.shared.ticket.as_ref())?;
34    let ticket = load_ticket_config(&root)?;
35
36    let document = TixDocument::load(&app.context.config_path)?;
37    let engine: EngineConfig = document.section_or_default("engine")?;
38
39    // Validate the batch up front: every name must be tracked, and every
40    // backing repo must still be registered (pruning goes through the repo).
41    let mut tracked: Vec<&str> = ticket.worktrees.keys().map(String::as_str).collect();
42    tracked.sort();
43    for name in &args.names {
44        let entry = ticket.worktrees.get(name).ok_or_else(|| {
45            TixError::WorktreeNotFound(format!(
46                "'{}' is not a tracked worktree of ticket '{}' (tracked: {})",
47                name,
48                ticket.key,
49                if tracked.is_empty() {
50                    "none".to_string()
51                } else {
52                    tracked.join(", ")
53                }
54            ))
55        })?;
56        if !engine.configured_repositories.contains_key(&entry.repo) {
57            return Err(SdkError::Engine(TixError::RepoNotFound(format!(
58                "worktree '{}' belongs to '{}', which is no longer a registered repository",
59                name, entry.repo
60            ))));
61        }
62    }
63
64    let ticket_document_path = root.join(".tix").join("ticket.toml");
65    for name in &args.names {
66        let entry = &ticket.worktrees[name];
67        let repo = engine.configured_repositories[&entry.repo]
68            .clone()
69            .resolve(&entry.repo)?;
70        repo.remove_worktree(&root.join(name), false)?;
71        info!(name = %name, repo = %entry.repo, "worktree removed");
72
73        // Drop the entry as soon as the prune lands, so an error later in
74        // the batch leaves the document agreeing with disk.
75        with_write(&ticket_document_path, |doc| {
76            if let Some(worktrees) = doc.doc_mut()["ticket"]["worktrees"].as_table_mut() {
77                worktrees.remove(name);
78            }
79            Ok(())
80        })?;
81        println!("removed {name}");
82    }
83    Ok(())
84}
