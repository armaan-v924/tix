1//! Plugin discovery for `tix --help` — the "Plugins" section
2//! (`design/spec.md` §5.6).
3//!
4//! Listing is best-effort by contract: a plugin whose handshake is missing,
5//! broken, slow, or malicious degrades to its bare name. **The listing
6//! itself never fails**, and handshake output is treated as untrusted input.
7
8use std::collections::BTreeMap;
9use std::io::Read;
10use std::path::{Path, PathBuf};
11use std::time::{Duration, Instant};
12use tracing::debug;
13
14/// How long one `print-cli-help` handshake may run before it is killed.
15const HANDSHAKE_DEADLINE: Duration = Duration::from_millis(300);
16
17/// Longest description rendered next to a plugin name.
18const DESCRIPTION_CAP: usize = 100;
19
20/// Builds the "Plugins:" help section, or `None` when no plugins are on
21/// `PATH` — zero plugins simply omits the section.
22pub fn plugins_help_section() -> Option<String> {
23    let plugins = discover_plugins();
24    if plugins.is_empty() {
25        return None;
26    }
27
28    let width = plugins.keys().map(String::len).max().unwrap_or(0);
29    let mut section = String::from("Plugins:\n");
30    for (name, path) in &plugins {
31        match handshake_description(path) {
32            Some(description) => {
33                section.push_str(&format!("  {name:width$}  {description}\n"));
34            }
35            None => section.push_str(&format!("  {name}\n")),
36        }
37    }
38    section.pop(); // trailing newline
39    Some(section)
40}

42/// Scans every `PATH` directory for executables named `tix-*`.
43///
44/// Deduplicated by name with the **first** match on `PATH` winning (shell
45/// semantics); the `tix-` prefix is stripped for display. Returned sorted by
46/// name (BTreeMap) for stable output.
47fn discover_plugins() -> BTreeMap<String, PathBuf> {
48    let mut plugins: BTreeMap<String, PathBuf> = BTreeMap::new();
49    let Some(path_var) = std::env::var_os("PATH") else {
50        return plugins;
51    };
52    for dir in std::env::split_paths(&path_var) {
53        let Ok(entries) = std::fs::read_dir(&dir) else {
54            continue;
55        };
56        for entry in entries.filter_map(Result::ok) {
57            let file_name = entry.file_name();
58            let Some(name) = file_name.to_str() else {
59                continue;
60            };
61            let Some(stripped) = name.strip_prefix("tix-") else {
62                continue;
63            };
64            if stripped.is_empty() || !is_executable_file(&entry.path()) {
65                continue;
66            }
67            // First match on PATH wins; later directories don't override.
68            plugins
69                .entry(stripped.to_string())
70                .or_insert_with(|| entry.path());
71        }
72    }
73    plugins
74}

76/// Runs `<plugin> print-cli-help` **bare** — no `--tix-*` flags — and
77/// returns a sanitized one-line description.
78///
79/// Any failure shape (spawn error, nonzero exit, timeout, empty or garbage
80/// output) degrades to `None`: the plugin still lists by name.
81fn handshake_description(plugin: &Path) -> Option<String> {
82    let mut child = std::process::Command::new(plugin)
83        .arg("print-cli-help")
84        .stdin(std::process::Stdio::null())
85        .stdout(std::process::Stdio::piped())
86        .stderr(std::process::Stdio::null())
87        .spawn()
88        .ok()?;
89
90    // Poll rather than block: a hung plugin must not hang `tix --help`.
91    let deadline = Instant::now() + HANDSHAKE_DEADLINE;
92    let status = loop {
93        match child.try_wait() {
94            Ok(Some(status)) => break status,
95            Ok(None) if Instant::now() < deadline => {
96                std::thread::sleep(Duration::from_millis(10));
97            }
98            _ => {
99                debug!(plugin = %plugin.display(), "handshake timed out or errored — killing");
100                let _ = child.kill();
101                let _ = child.wait();
102                return None;
103            }
104        }
105    };
106    if !status.success() {
107        return None;
108    }
109
110    // Bounded read: output is untrusted, so never slurp unbounded bytes.
111    let mut raw = Vec::with_capacity(512);
112    child.stdout.take()?.take(4096).read_to_end(&mut raw).ok()?;
113    sanitize(&raw)
114}
115
116/// Strips the handshake output down to one printable line, capped.
117fn sanitize(raw: &[u8]) -> Option<String> {
118    let text = String::from_utf8_lossy(raw);
119    let line = text.lines().find(|line| !line.trim().is_empty())?;
120    let clean: String = line
121        .chars()
122        .filter(|c| !c.is_control())
123        .take(DESCRIPTION_CAP)
124        .collect();
125    let clean = clean.trim().to_string();
126    (!clean.is_empty()).then_some(clean)
127}

129#[cfg(unix)]
130fn is_executable_file(path: &Path) -> bool {
131    use std::os::unix::fs::PermissionsExt;
132    path.metadata()
133        .map(|m| m.is_file() && m.permissions().mode() & 0o111 != 0)
134        .unwrap_or(false)
135}
136
137#[cfg(not(unix))]
138fn is_executable_file(path: &Path) -> bool {
139    path.is_file()
140}

142#[cfg(test)]
143mod tests {
144    use super::*;
145
146    /// One printable line survives; control characters and later lines don't.
147    #[test]
148    fn test_sanitize_strips_and_caps() {
149        assert_eq!(
150            sanitize(b"deploys things\nsecond line ignored").as_deref(),
151            Some("deploys things")
152        );
153        assert_eq!(
154            sanitize(b"\x1b[31mred\x07 alert\x1b[0m").as_deref(),
155            Some("[31mred alert[0m")
156        );
157        assert_eq!(sanitize(b"").as_deref(), None);
158        assert_eq!(sanitize(b"\n\n   \n").as_deref(), None);
159        let long = vec![b'a'; 500];
160        assert_eq!(sanitize(&long).unwrap().len(), DESCRIPTION_CAP);
161    }
