3use crate::tix::utils::App;
4use crate::tix::{Commands, cli, config, plugin, repo, ticket};
5use tix_sdk::context::Context;
6use tracing_subscriber::fmt;
7
8/// Prints a CLI error to stderr and exits nonzero.
9///
10/// The frontend owns process control and user-facing output — the engine
11/// only ever returns errors.
12fn fail(error: tix_sdk::SdkError) -> ! {
13    eprintln!("error: {error}");
14    std::process::exit(1);
15}
16
17fn main() {
18    let parsed = tix::TixParser::parse_with_plugin_help();
19    let log_level = parsed.resolve_log_level();
20
21    // Diagnostics go to stderr: stdout carries results only, so
22    // `tix ticket list --output json | jq` works even when warnings fire.
23    fmt()
24        .with_max_level(log_level)
25        .with_writer(std::io::stderr)
26        .init();
27
28    // Resolve the config path once (--config > TIX_CONFIG_PATH > platform
29    // default) and hand it to every subcommand. Existence is enforced here,
30    // centrally, for every command except the two that must run before a
31    // config exists (`tix config init`, `tix cli completions`).
32    let context = Context::resolve(parsed.config.clone()).unwrap_or_else(|e| fail(e));
33    if parsed.command.requires_config()
34        && let Err(e) = context.require_config_path()
35    {
36        fail(e);
37    }
38    let app = App {
39        context,
40        output: parsed.output.unwrap_or(tix::utils::OutputType::Default),
41        log_level,
42    };
43
44    let result = match parsed.command {
45        Commands::Cli(args) => match args.command {
46            cli::CliCommands::Completions(args) => cli::completions::run(&app, args),
47            cli::CliCommands::Update(args) => cli::update::run(&app, args),
48        },
49        Commands::Config(args) => match args.command {
50            config::ConfigCommands::Get(args) => config::get::run(&app, args),
51            config::ConfigCommands::Init(args) => config::init::run(&app, args),
52            config::ConfigCommands::Set(args) => config::set::run(&app, args),
53            config::ConfigCommands::Show(args) => config::show::run(&app, args),
54        },
55        Commands::Repo(args) => match args.command {
56            repo::RepoCommands::Add(args) => repo::add::run(&app, args),
57            repo::RepoCommands::Clone(args) => repo::clone::run(&app, args),
58        },
59        Commands::Ticket(args) => match args.command {
60            ticket::TicketCommands::Add(args) => ticket::add::run(&app, args),
61            ticket::TicketCommands::Destroy(args) => ticket::destroy::run(&app, args),
62            ticket::TicketCommands::Info(args) => ticket::info::run(&app, args),
63            ticket::TicketCommands::List(args) => ticket::list::run(&app, args),
64            ticket::TicketCommands::Remove(args) => ticket::remove::run(&app, args),
65            ticket::TicketCommands::Setup(args) => ticket::setup::run(&app, args),
66        },
67        Commands::Add(args) => ticket::add::run(&app, args),
68        Commands::Destroy(args) => ticket::destroy::run(&app, args),
69        Commands::Info(args) => ticket::info::run(&app, args),
70        Commands::List(args) => ticket::list::run(&app, args),
71        Commands::Remove(args) => ticket::remove::run(&app, args),
72        Commands::Setup(args) => ticket::setup::run(&app, args),
73        Commands::Completions(args) => cli::completions::run(&app, args),
74
75        Commands::External(args) => plugin::run(&app, args),
76    };
77
78    if let Err(e) = result {
79        fail(e);
80    }
81}
