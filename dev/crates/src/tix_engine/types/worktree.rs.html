4/// The recorded state of one worktree in a ticket document — the value type of
5/// [`TicketConfig::worktrees`](crate::TicketConfig::worktrees).
6///
7/// The map key (the worktree directory name under the ticket root) lives in
8/// [`TicketConfig`](crate::TicketConfig); this struct carries what that
9/// directory *is*: which repository it belongs to and which branch it has
10/// checked out. Repos in one ticket may sit on different branches — worktrees
11/// share a branch *prefix* rather than one branch.
12///
13/// # Examples
14///
15/// As TOML, an entry appears as a sub-table of `worktrees` named after its
16/// directory:
17///
18/// ```
19/// # use tix_engine::WorktreeConfig;
20/// let entry: WorktreeConfig = toml::from_str(
21///     r#"
22///     repo = "backend"
23///     branch = "feature/JIRA-123-fix-login"
24///     "#,
25/// )
26/// .unwrap();
27/// assert_eq!(entry.repo, "backend");
28/// assert_eq!(entry.branch, "feature/JIRA-123-fix-login");
29/// assert_eq!(entry.base, None);
30/// ```
31///
32/// A worktree tix cut a branch for also records the base it was cut from:
33///
34/// ```toml
35/// [ticket.worktrees.tix-part2]
36/// repo = "tix"
37/// branch = "stack/JIRA-1-part-2"
38/// base = "stack/JIRA-1-part-1"
39/// ```
40#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
41#[derive(Serialize, Deserialize, Clone, Debug, PartialEq)]
42#[serde(deny_unknown_fields)]
43pub struct WorktreeConfig {
44    /// The alias of the repository this worktree belongs to.
45    pub repo: String,
46    /// The branch checked out in this worktree.
47    pub branch: String,
48    /// The ref `branch` was cut from, as resolved at creation — a name such
49    /// as `origin/main` or a sibling's `stack/JIRA-1-part-1`, never a SHA.
50    ///
51    /// A name is what a pull request's base takes, and it is what makes
52    /// stack topology recoverable from disk: git itself only remembers the
53    /// merge-base. Absent when tix did not cut the branch (it already
54    /// existed) or the entry predates the field.
55    #[serde(default, skip_serializing_if = "Option::is_none")]
56    pub base: Option<String>,
57}

59/// A live git worktree associated with a repository and ticket branch.
60///
61/// Produced by [`TicketConfig::resolve`](crate::TicketConfig::resolve) (from
62/// recorded state verified against disk) or by
63/// [`Repository::create_worktree`](crate::Repository::create_worktree) (from a
64/// worktree it just created). Holding one means the worktree existed at the
65/// time it was produced.
66#[derive(Serialize, Debug, Clone, PartialEq)]
67pub struct Worktree {
68    /// The worktree directory name under the ticket root — the key of this
69    /// worktree's entry in [`TicketConfig::worktrees`](crate::TicketConfig::worktrees).
70    pub name: String,
71    /// The alias of the repository.
72    pub repo_alias: String,
73    /// The path to the worktree directory.
74    pub path: PathBuf,
75    /// The branch of the worktree.
76    pub branch: String,
77    /// The ref the branch was cut from — see [`WorktreeConfig::base`].
78    pub base: Option<String>,
79}
