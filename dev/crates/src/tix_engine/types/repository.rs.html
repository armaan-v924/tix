7/// The configuration for a repository.
8#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
9#[derive(Serialize, Deserialize, Clone, Debug, PartialEq)]
10pub struct RepositoryConfig {

12    pub remote: String,
13    /// The path to the code directory of the repository.
14    pub code_path: PathBuf,
15}

17impl RepositoryConfig {
18    /// Creates a new `RepositoryConfig`.
19    ///
20    /// # Arguments
21    ///
22    /// * `remote` - The remote URL of the repository.
23    /// * `code_path` - The path to the code directory of the repository.
24    ///
25    /// # Examples
26    ///
27    /// ```
28    /// # use std::path::PathBuf;
29    /// # use tix_engine::RepositoryConfig;
30    /// let config = RepositoryConfig::new("https://github.com/user/repo.git".to_string(), PathBuf::from("~/code/repo"));
31    /// ```
32    pub fn new(remote: String, code_path: PathBuf) -> Self {
33        Self { remote, code_path }
34    }
35
36    /// Opens the already-cloned repository at [`Self::code_path`] and returns a live [`Repository`].
37    ///
38    /// Use [`Self::clone_remote`] if the repository has not been cloned yet.
39    ///
40    /// # Errors
41    ///
42    /// Returns [`TixError::GitError`] if the path is not a valid git repository.
43    ///
44    /// # Examples
45    ///
46    /// ```no_run
47    /// # use tix_engine::RepositoryConfig;
48    /// # use std::path::PathBuf;
49    /// let config = RepositoryConfig::new(
50    ///     "https://github.com/owner/repo.git".to_string(),
51    ///     PathBuf::from("/home/user/code/repo"),
52    /// );
53    /// let repo = config.resolve("my-repo");
54    /// ```
55    pub fn resolve(self, alias: &str) -> Result<Repository, TixError> {
56        debug!(alias = %alias, path = %self.code_path.display(), "opening repository");
57        let repo = git2::Repository::open(&self.code_path).map_err(|e| {
58            // debug, not error: failing to open is an expected probe on the
59            // ensure() path (resolve-then-clone); the Err carries the story
60            // for callers that treat it as fatal.
61            debug!(alias = %alias, error = %e, "failed to open repository");
62            TixError::GitError(e)
63        })?;
64        debug!(alias = %alias, "repository opened");
65        Ok(Repository {
66            alias: alias.to_string(),
67            config: self,
68            repo,
69        })
70    }

72    /// Clones [`Self::remote`] into [`Self::code_path`] and returns a live [`Repository`].
73    ///
74    /// Use [`Self::resolve`] if the repository is already cloned locally.
75    ///
76    /// # Errors
77    ///
78    /// Returns [`TixError::GitError`] if the clone fails.
79    ///
80    /// # Examples
81    ///
82    /// ```no_run
83    /// # use tix_engine::RepositoryConfig;
84    /// # use std::path::PathBuf;
85    /// let config = RepositoryConfig::new(
86    ///     "https://github.com/owner/repo.git".to_string(),
87    ///     PathBuf::from("/home/user/code/repo"),
88    /// );
89    /// let repo = config.clone_remote("my-repo");
90    /// ```
91    pub fn clone_remote(self, alias: &str) -> Result<Repository, TixError> {
92        info!(alias = %alias, remote = %self.remote, "cloning repository");
93        let repo = git2::build::RepoBuilder::new()
94            .fetch_options(crate::auth::fetch_options())
95            .clone(&self.remote, &self.code_path)
96            .map_err(|e| {
97                error!(alias = %alias, error = %e, "clone failed");
98                TixError::GitError(e)
99            })?;
100        info!(alias = %alias, path = %self.code_path.display(), "clone complete");
101        Ok(Repository {
102            alias: alias.to_string(),
103            config: self,
104            repo,
105        })
106    }

108    /// Opens the repository if already cloned, or clones it first if not.
109    ///
110    /// Tries [`Self::resolve`] first. If that fails with a [`TixError::GitError`], falls back to
111    /// [`Self::clone_remote`]. Other errors propagate without triggering a clone.
112    ///
113    /// # Errors
114    ///
115    /// Returns [`TixError::GitError`] if the clone fails.
116    ///
117    /// # Examples
118    ///
119    /// ```no_run
120    /// # use tix_engine::RepositoryConfig;
121    /// # use std::path::PathBuf;
122    /// let config = RepositoryConfig::new(
123    ///     "https://github.com/owner/repo.git".to_string(),
124    ///     PathBuf::from("/home/user/code/repo"),
125    /// );
126    /// let repo = config.ensure("my-repo");
127    /// ```
128    pub fn ensure(self, alias: &str) -> Result<Repository, TixError> {
129        let config = self.clone();
130        match self.resolve(alias) {
131            Ok(repo) => Ok(repo),
132            Err(TixError::GitError(_)) => config.clone_remote(alias),
133            Err(e) => Err(e),
134        }
135    }

148    /// Creates a git worktree named `name` for `branch` at `path`.
149    ///
150    /// `name` is the worktree directory name under the ticket root — the key
151    /// of the worktree's entry in
152    /// [`TicketConfig::worktrees`](crate::TicketConfig::worktrees) — and
153    /// `path` is the full, already-resolved directory the worktree should
154    /// appear at. The engine takes both as given; deriving them from a ticket
155    /// is the frontend's job.
156    ///
157    /// Syncs the repository first (fetches and fast-forwards). Pass `force: true` to discard
158    /// local changes before syncing. If `branch` exists locally the worktree
159    /// checks it out; otherwise the branch is created at the synced HEAD and
160    /// the worktree opens on it — the branch name is always `branch`, never
161    /// the worktree name.
162    ///
163    /// # Errors
164    ///
165    /// Returns [`TixError::GitError`] if the sync or worktree creation fails.
166    ///
167    /// # Examples
168    ///
169    /// ```no_run
170    /// # use tix_engine::RepositoryConfig;
171    /// # use std::path::PathBuf;
172    /// let repo = RepositoryConfig::new(
173    ///     "https://github.com/owner/repo.git".to_string(),
174    ///     PathBuf::from("/home/user/code/repo"),
175    /// ).resolve("my-repo").unwrap();
176    /// let worktree = repo.create_worktree(
177    ///     "my-repo",
178    ///     "feature/JIRA-123-fix-login",
179    ///     &PathBuf::from("/home/user/tickets/JIRA-123/my-repo"),
180    ///     false,
181    /// );
182    /// ```
183    pub fn create_worktree(
184        &self,
185        name: &str,
186        branch: &str,
187        path: &Path,
188        force: bool,
189    ) -> Result<Worktree, TixError> {
190        info!(alias = %self.alias, name, branch, "creating worktree");
191        self.sync(force)?;
192
193        // The worktree must open on `branch` by that exact name. Without an
194        // explicit reference, git2 would create a branch named after the
195        // *worktree* — wrong the moment name and branch differ (per-worktree
196        // branches).
197        let local_branch = match self.repo.find_branch(branch, git2::BranchType::Local) {
198            Ok(existing) => existing,
199            Err(_) => {
200                debug!(alias = %self.alias, branch, "branch not found locally, creating at synced HEAD");
201                let head = self
202                    .repo
203                    .head()
204                    .and_then(|h| h.peel_to_commit())
205                    .map_err(TixError::GitError)?;
206                self.repo
207                    .branch(branch, &head, false)
208                    .map_err(TixError::GitError)?
209            }
210        };
211        let reference = Some(local_branch.into_reference());
212
213        if let Some(parent) = path.parent() {
214            std::fs::create_dir_all(parent).map_err(TixError::IoError)?;
215        }
216
217        let mut options = WorktreeAddOptions::new();
218        options.reference(reference.as_ref());
219
220        // Git registers worktrees under a repo-unique name. The config-level
221        // `name` (directory under the ticket root) collides the moment two
222        // tickets include the same repo, so registration uses the sanitized
223        // branch instead — branches are unique per repo by git's own rules
224        // (v2 parity: repo_worktrees held sanitized names).
225        let registration = registration_name(branch);
226        let worktree = self
227            .repo
228            .worktree(&registration, path, Some(&options))
229            .map(|_| Worktree {
230                name: name.to_string(),
231                branch: branch.to_string(),
232                repo_alias: self.alias.clone(),
233                path: path.to_path_buf(),
234            })
235            .map_err(|e| {
236                error!(alias = %self.alias, name, error = %e, "failed to create worktree");
237                TixError::GitError(e)
238            })?;
239        info!(alias = %self.alias, name, path = %path.display(), "worktree created");
240        Ok(worktree)
241    }

243    /// Finds the registered worktree whose recorded path is `path`, if any.
244    ///
245    /// Paths are normalized before comparison — git records canonicalized
246    /// paths (on macOS, `/var/...` becomes `/private/var/...`), so a raw
247    /// equality check would miss legitimate matches.
248    fn find_worktree_by_path(&self, path: &Path) -> Result<Option<git2::Worktree>, TixError> {
249        let target = normalize_path(path);
250        let names = self.repo.worktrees().map_err(TixError::GitError)?;
251        for name in names.iter().filter_map(|n| n.ok().flatten()) {
252            let worktree = self.repo.find_worktree(name).map_err(TixError::GitError)?;
253            if normalize_path(worktree.path()) == target {
254                return Ok(Some(worktree));
255            }
256        }
257        Ok(None)
258    }

260    /// Prunes the git worktree living at `path` from this repository.
261    ///
262    /// Lookup is by the worktree's recorded path rather than its registration
263    /// name — the path is what the ticket document knows
264    /// (`<ticket_root>/<name>`), and it stays unambiguous even for orphaned
265    /// worktrees whose directory is already gone.
266    ///
267    /// Pass `force: true` to remove even if the worktree is in a dirty state. Without `force`,
268    /// returns an error if the worktree fails validation (e.g. has uncommitted changes).
269    ///
270    /// # Errors
271    ///
272    /// - [`TixError::WorktreeNotFound`] if no registered worktree records `path`
273    /// - [`TixError::Message`] if the worktree is dirty without `force`
274    /// - [`TixError::GitError`] if pruning fails
275    ///
276    /// # Examples
277    ///
278    /// ```no_run
279    /// # use tix_engine::RepositoryConfig;
280    /// # use std::path::PathBuf;
281    /// let repo = RepositoryConfig::new(
282    ///     "https://github.com/owner/repo.git".to_string(),
283    ///     PathBuf::from("/home/user/code/repo"),
284    /// ).resolve("my-repo").unwrap();
285    /// repo.remove_worktree(&PathBuf::from("/home/user/tickets/JIRA-123/my-repo"), false);
286    /// ```
287    pub fn remove_worktree(&self, path: &Path, force: bool) -> Result<(), TixError> {
288        info!(alias = %self.alias, path = %path.display(), "removing worktree");
289
290        let worktree = self
291            .find_worktree_by_path(path)?
292            .ok_or_else(|| TixError::WorktreeNotFound(path.display().to_string()))?;
293
294        if !force {
295            // validate() catches structural breakage (e.g. the directory is
296            // gone). It says nothing about cleanliness — that takes opening
297            // the worktree and asking for its status.
298            if worktree.validate().is_err() {
299                error!(alias = %self.alias, path = %path.display(), "worktree failed validation, use force to remove");
300                return Err(TixError::from(
301                    "worktree is not in a clean state, use force to remove",
302                ));
303            }
304            let worktree_repo =
305                git2::Repository::open(worktree.path()).map_err(TixError::GitError)?;
306            let count = uncommitted_changes(&worktree_repo)?;
307            if count > 0 {
308                error!(alias = %self.alias, path = %path.display(), count, "worktree has uncommitted changes, use force to remove");
309                return Err(TixError::from(
310                    "worktree has uncommitted changes, use force to remove",
311                ));
312            }
313        }
314
315        let mut opts = git2::WorktreePruneOptions::new();
316        opts.working_tree(true).valid(true);
317
318        worktree.prune(Some(&mut opts)).map_err(|e| {
319            error!(alias = %self.alias, path = %path.display(), "failed to remove worktree: {e}");
320            TixError::GitError(e)
321        })?;
322        info!(alias = %self.alias, path = %path.display(), "worktree removed");
323        Ok(())
324    }

343    /// Fetches and fast-forwards `branch` from `origin`.
344    ///
345    /// Errors if the repository is mid-operation (merge, rebase, etc.) or if the local branch has
346    /// diverged from remote without `force`. With `force`, discards all local changes and resets
347    /// to the remote state. Also errors if the branch is already checked out in another worktree.
348    ///
349    /// # Errors
350    ///
351    /// - [`TixError::Message`] for dirty state, diverged branch, or worktree conflicts
352    /// - [`TixError::GitError`] for underlying git failures
353    ///
354    /// # Examples
355    ///
356    /// ```no_run
357    /// # use tix_engine::RepositoryConfig;
358    /// # use std::path::PathBuf;
359    /// let repo = RepositoryConfig::new(
360    ///     "https://github.com/owner/repo.git".to_string(),
361    ///     PathBuf::from("/home/user/code/repo"),
362    /// ).resolve("my-repo").unwrap();
363    /// repo.sync_base("main", false);
364    /// ```
365    pub fn sync_base(&self, branch: &str, force: bool) -> Result<(), TixError> {
366        debug!(alias = %self.alias, branch, force, "syncing repository");
367
368        // RepositoryState::Clean means no git operation is in progress.
369        // force cannot recover from this — the user must resolve it manually.
370        if self.repo.state() != RepositoryState::Clean {
371            error!(alias = %self.alias, "repository is mid-operation, cannot sync");
372            return Err(TixError::from(
373                "repository is mid-operation (merge, rebase, etc), resolve it before syncing",
374            ));
375        }
376
377        // if anything is dirty and force is not set, bail rather than silently discarding work.
378        if !force {
379            let count = uncommitted_changes(&self.repo)?;
380            if count > 0 {
381                error!(alias = %self.alias, count, "repository has uncommitted changes");
382                return Err(TixError::from(
383                    "repository has uncommitted changes, use force to discard them",
384                ));
385            }
386        }
387
388        // fetch so we have up-to-date remote refs before checking branch existence.
389        // this also means if the branch only exists on origin and not locally,
390        // we'll have it available to create a local tracking branch from.
391        debug!(alias = %self.alias, branch, "fetching from origin");
392        let mut remote = self
393            .repo
394            .find_remote("origin")
395            .map_err(TixError::GitError)?;
396        remote
397            .fetch(&[branch], Some(&mut crate::auth::fetch_options()), None)
398            .map_err(TixError::GitError)?;
399
400        // check if the branch exists locally. if not, try to create it from
401        // the remote ref we just fetched — if origin doesn't have it either, error.
402        if self
403            .repo
404            .find_branch(branch, git2::BranchType::Local)
405            .is_err()
406        {
407            debug!(alias = %self.alias, branch, "branch not found locally, creating from origin");
408            let remote_ref = format!("refs/remotes/origin/{}", branch);
409            let remote_commit = self
410                .repo
411                .find_reference(&remote_ref)
412                .map_err(|_| TixError::from(format!("branch '{}' not found on origin", branch)))?;
413            let commit = self
414                .repo
415                .reference_to_annotated_commit(&remote_commit)
416                .map_err(TixError::GitError)?;
417            let target = self
418                .repo
419                .find_commit(commit.id())
420                .map_err(TixError::GitError)?;
421            // false = don't force-overwrite if it somehow already exists
422            self.repo
423                .branch(branch, &target, false)
424                .map_err(TixError::GitError)?;
425            debug!(alias = %self.alias, branch, "local branch created from origin");
426        }
427
428        // check divergence before touching anything — if not fast-forwardable and
429        // not force, we'd rather error now than after nuking local changes.
430        let ref_name = format!("refs/heads/{}", branch);
431        let fetch_head = self
432            .repo
433            .find_reference("FETCH_HEAD")
434            .map_err(TixError::GitError)?;
435        let fetch_commit = self
436            .repo
437            .reference_to_annotated_commit(&fetch_head)
438            .map_err(TixError::GitError)?;
439        if !force {
440            // merge_analysis checks whether applying the remote commit is a
441            // fast-forward — i.e. our local branch is a direct ancestor of the
442            // remote, so we can move the pointer forward without losing anything.
443            let (analysis, _) = self
444                .repo
445                .merge_analysis(&[&fetch_commit])
446                .map_err(TixError::GitError)?;
447            if !analysis.is_fast_forward() && !analysis.is_up_to_date() {
448                error!(alias = %self.alias, branch, "branch has diverged from remote");
449                return Err(TixError::from(format!(
450                    "'{}' has diverged from remote, use force to reset",
451                    branch
452                )));
453            }
454        }
455
456        // git won't let you switch to a branch that's already checked out in another
457        // worktree — each worktree must have a unique branch. we check all registered
458        // worktrees and error with which one has the branch so the user knows where to look.
459        let worktrees = self.repo.worktrees().map_err(TixError::GitError)?;
460        for name in worktrees.iter().filter_map(|n| n.ok().flatten()) {
461            // open the worktree as its own git2::Repository so we can inspect its HEAD
462            let wt = self.repo.find_worktree(name).map_err(TixError::GitError)?;
463            let wt_repo = git2::Repository::open(wt.path()).map_err(TixError::GitError)?;
464            if let Ok(head) = wt_repo.head() {
465                // shorthand() gives the branch name (e.g. "main") rather than the
466                // full ref path (e.g. "refs/heads/main")
467                if head.shorthand() == Ok(branch) {
468                    error!(alias = %self.alias, branch, worktree = name, "branch is checked out in another worktree");
469                    return Err(TixError::from(format!(
470                        "branch '{}' is checked out in worktree '{}'",
471                        branch, name
472                    )));
473                }
474            }
475        }
476
477        // --- destructive work starts here ---
478
479        if force {
480            // checkout_head rewrites the working tree to match HEAD.
481            // force() discards modified tracked files and remove_untracked removes
482            // files git doesn't know about — the same set uncommitted_changes
483            // counts. Ignored files (build caches, `.env`) are deliberately kept:
484            // they were never "changes", so forcing past a dirty check must not
485            // delete them (#175).
486            warn!(alias = %self.alias, "force flag set — discarding all local changes");
487            let mut clean_opts = git2::build::CheckoutBuilder::new();
488            clean_opts.force().remove_untracked(true);
489            self.repo
490                .checkout_head(Some(&mut clean_opts))
491                .map_err(TixError::GitError)?;
492        }
493
494        // set_head points HEAD at the branch ref without touching the working tree.
495        // checkout_head then updates the working tree to match.
496        debug!(alias = %self.alias, branch, "switching HEAD");
497        self.repo.set_head(&ref_name).map_err(TixError::GitError)?;
498        self.repo
499            .checkout_head(Some(git2::build::CheckoutBuilder::new().force()))
500            .map_err(TixError::GitError)?;
501
502        // move the local branch pointer to the fetched remote commit
503        let mut branch_ref = self
504            .repo
505            .find_reference(&ref_name)
506            .map_err(TixError::GitError)?;
507        if force {
508            // discard any local commits that aren't on the remote
509            debug!(alias = %self.alias, branch, "force resetting to remote");
510            branch_ref
511                .set_target(fetch_commit.id(), "sync_base: force reset")
512                .map_err(TixError::GitError)?;
513        } else {
514            debug!(alias = %self.alias, branch, "fast-forwarding to remote");
515            branch_ref
516                .set_target(fetch_commit.id(), "sync_base: fast-forward")
517                .map_err(TixError::GitError)?;
518        }
519
520        // final checkout to update the working tree to the new branch pointer
521        self.repo
522            .checkout_head(Some(git2::build::CheckoutBuilder::new().force()))
523            .map_err(TixError::GitError)?;
524
525        debug!(alias = %self.alias, branch, "sync complete");
526        Ok(())
527    }
528}

530/// Counts the working-tree entries that would be lost by discarding local
531/// changes: modified, staged, and untracked files.
532///
533/// Ignored files are excluded on purpose. Build output, dependency caches, and
534/// `.env` files are routinely present in a clean checkout, and `git status`
535/// does not report them either. Passing explicit options matters: with
536/// `statuses(None)`, libgit2 falls back to defaults that include ignored files.
537fn uncommitted_changes(repo: &git2::Repository) -> Result<usize, TixError> {
538    let mut options = git2::StatusOptions::new();
539    options
540        .include_untracked(true)
541        .recurse_untracked_dirs(true)
542        .include_ignored(false);
543    let statuses = repo
544        .statuses(Some(&mut options))
545        .map_err(TixError::GitError)?;
546    Ok(statuses.len())
547}

549/// The repo-unique name a worktree is registered under: the branch with path
550/// separators flattened. Branch uniqueness per repo (enforced by git) makes
551/// this collision-free in practice; a rare sanitization collision surfaces as
552/// a clear git error at creation.
553fn registration_name(branch: &str) -> String {
554    branch.replace(['/', '\\'], "-")
555}
556
557/// Canonicalizes `path` for comparison, tolerating paths that no longer
558/// exist (an orphaned worktree's directory) by canonicalizing the parent and
559/// re-joining the final component.
560fn normalize_path(path: &Path) -> PathBuf {
561    if let Ok(canonical) = path.canonicalize() {
562        return canonical;
563    }
564    match (path.parent(), path.file_name()) {
565        (Some(parent), Some(name)) => parent
566            .canonicalize()
567            .map(|parent| parent.join(name))
568            .unwrap_or_else(|_| path.to_path_buf()),
569        _ => path.to_path_buf(),
570    }
571}

1080    /// Initialises an empty bare repository at `path`. Has no commits — call
1081    /// [`add_commit_to_bare`] immediately after to establish a branch.
1082    pub fn init_bare_repo(path: &Path) -> git2::Repository {
1083        let repo = git2::Repository::init_bare(path).unwrap();
1084        repo.set_head("refs/heads/main").unwrap();
1085        repo
1086    }
1087
1088    /// Clones the bare repository at `remote` into `dest`, setting `origin` automatically.
1089    pub fn clone_repo(remote: &Path, dest: &Path) -> git2::Repository {
1090        git2::Repository::clone(remote.to_str().unwrap(), dest).unwrap()
1091    }
1092
1093    /// Wraps a `git2::Repository` in a [`Repository`], pointing `origin` at `remote`.
1094    pub fn tix_repo(remote: &Path, local: git2::Repository) -> Repository {
1095        let code_path = local.path().parent().unwrap().to_path_buf();
1096        Repository {
1097            alias: "default".into(),
1098            config: RepositoryConfig::new(remote.to_str().unwrap().into(), code_path),
1099            repo: local,
1100        }
1101    }
1102
1103    /// Commits the current index state to `branch` in a working-directory repository.
1104    /// Chains onto the existing branch tip so history is preserved across multiple calls.
1105    pub fn add_commit<'a>(
1106        repo: &'a git2::Repository,
1107        branch: &str,
1108        message: &str,
1109    ) -> git2::Commit<'a> {
1110        let branch_ref = repo.find_branch(branch, git2::BranchType::Local).unwrap();
1111        let sig = git2::Signature::now("test", "test@test.com").unwrap();
1112        let tree_id = repo.index().unwrap().write_tree().unwrap();
1113        let tree = repo.find_tree(tree_id).unwrap();
1114        // resolve the current branch tip as parent; None on the first commit (root).
1115        let parent = branch_ref
1116            .get()
1117            .target()
1118            .and_then(|oid| repo.find_commit(oid).ok());
