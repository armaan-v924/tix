1//! Plugin dispatch: exec `tix-<name>` with the invocation contract
2//! ([contract](https://tix.armaanv.dev/latest/plugins/specification/#the-exec-contract)).
3//!
4//! Unknown subcommands land here via the clap catch-all — builtins always
5//! win, so dispatch is only ever reached for non-builtin names. There is no
6//! plugin registry: `PATH` is the registry.
7

12use tix_sdk::delta::{Delta, DeltaTarget};
13use tix_sdk::document::with_write;

18/// Recursion guard: `tix-foo` shelling `tix foo` forever is the one failure
19/// mode that damages more than tix.
20const DEPTH_ENV: &str = "TIX_DEPTH";
21const DEPTH_CAP: u32 = 10;
22
23/// Dispatches `tix <name> <args…>` to the `tix-<name>` binary on `PATH`.
24///
25/// The host resolves its own globals **before** forwarding — the raw args
26/// are pre-scanned (the same SDK code plugins build on) and the
27/// plugin receives settled `--tix-*` values, never raw flags. stdin/stdout/
28/// stderr are inherited: the plugin owns the terminal. The child's exit code
29/// propagates unmodified, except **125**, which is reserved for protocol
30/// mismatch and reported as a versioning problem instead.
31///
32/// On exit 0 the `--tix-delta` file is handed to the diff-back apply path;
33/// a nonzero exit discards it.
34pub fn run(app: &App, args: Vec<String>) -> Result<(), SdkError> {
35    let mut args = args.into_iter();
36    let name = args
37        .next()
38        .ok_or_else(|| SdkError::Message("no plugin name in external args".to_string()))?;
39    let raw_args: Vec<String> = args.collect();
40
41    let Some(binary) = find_plugin(&name) else {
42        return Err(SdkError::Message(format!(
43            "unknown command '{name}': not a builtin, and no 'tix-{name}' executable on PATH"
44        )));
45    };
46
47    // Fork-bomb guard: increment TIX_DEPTH, hard-fail past the cap.
48    let depth: u32 = std::env::var(DEPTH_ENV)
49        .ok()
50        .and_then(|value| value.parse().ok())
51        .unwrap_or(0);
52    if depth >= DEPTH_CAP {
53        return Err(SdkError::Message(format!(
54            "tix invocation depth {depth} exceeds the cap of {DEPTH_CAP} — \
55             a plugin is probably re-invoking itself"
56        )));
57    }
58
59    // Pre-scan the raw args for tix's own globals (external_subcommand
60    // captured them raw), then settle values for forwarding.
61    let (globals, user_args) = prescan_globals(&raw_args);
62    let log_level = globals
63        .log_level
64        .or_else(|| globals.verbose.then(|| "trace".to_string()))
65        .or_else(|| globals.quiet.then(|| "warn".to_string()))
66        .unwrap_or_else(|| app.log_level.to_string().to_lowercase());
67    let output = globals
68        .output
69        .unwrap_or_else(|| app.output.as_str().to_string());
70    // Per-process color resolution: the plugin's stdout is ours (inherited).
71    let color = std::env::var_os("NO_COLOR").is_none() && std::io::stdout().is_terminal();
72
73    // Ticket context from the discovery walk; its absence is forwarded as an
74    // absent flag, which is load-bearing (setup-shaped plugins run without).
75    let ticket_root = tix_sdk::discovery::discover_ticket_root()?;
76    let repo_context = ticket_root
77        .as_deref()
78        .and_then(|root| detect_current_repo(root).transpose())
79        .transpose()?;
80
81    // Host-created temp file for the outbound delta. Kept alive
82    // until after the wait; deleted on drop.
83    let delta_file = tempfile::NamedTempFile::new().map_err(SdkError::from)?;
84
85    let mut command = std::process::Command::new(&binary);
86    command
87        .arg("--tix-protocol")
88        .arg(PROTOCOL.to_string())
89        .arg("--tix-config")
90        .arg(&app.context.config_path)
91        .arg("--tix-delta")
92        .arg(delta_file.path())
93        .arg("--tix-log-level")
94        .arg(&log_level)
95        .arg("--tix-output")
96        .arg(&output)
97        .arg("--tix-color")
98        .arg(color.to_string())
99        .env(DEPTH_ENV, (depth + 1).to_string());
100    if let Some(root) = &ticket_root {
101        command.arg("--tix-ticket").arg(root);
102    }
103    if let Some((alias, path)) = &repo_context {
104        command.arg("--tix-repo").arg(alias);
105        command.arg("--tix-repo-dir").arg(path);
106    }
107    command.args(&user_args);
108
109    debug!(plugin = %binary.display(), "dispatching");
110    let status = command
111        .status()
112        .map_err(|e| SdkError::Message(format!("could not exec '{}': {e}", binary.display())))?;
113
114    match status.code() {
115        // The delta is applied only on exit 0; a failed plugin's
116        // half-intended writes are discarded with the temp file.
117        Some(0) => apply_delta_if_present(app, delta_file.path(), ticket_root.as_deref()),
118        Some(PROTOCOL_MISMATCH_EXIT) => {
119            // Reserved: report a versioning problem, not a plugin crash, and
120            // keep 125 out of the propagated range.
121            Err(SdkError::Message(format!(
122                "plugin 'tix-{name}' speaks a different tix protocol (host speaks {PROTOCOL}) — \
123                 rebuild or update the plugin"
124            )))
125        }
126        Some(code) => {
127            // The plugin's error is the plugin's to report; tix adds nothing.
128            std::process::exit(code);
129        }
130        None => Err(SdkError::Message(format!(
131            "plugin 'tix-{name}' was terminated by a signal"
132        ))),
133    }
134}

136/// First `tix-<name>` executable on `PATH`, shell semantics.
137fn find_plugin(name: &str) -> Option<PathBuf> {
138    let binary_name = format!("tix-{name}");
139    std::env::split_paths(&std::env::var_os("PATH")?)
140        .map(|dir| dir.join(&binary_name))
141        .find(|candidate| is_executable(candidate))
142}

144/// Which repo worktree the cwd is inside, when it is inside one: the
145/// longest-prefix match of the logical cwd against `ticket_root/<name>` for
146/// every worktree the ticket tracks (v2's `detect_current_repo`).
147fn detect_current_repo(
148    ticket_root: &std::path::Path,
149) -> Result<Option<(String, PathBuf)>, SdkError> {
150    let document =
151        tix_sdk::document::TixDocument::load(&ticket_root.join(".tix").join("ticket.toml"))?;
152    let Some(ticket): Option<TicketConfig> = document.section("ticket")? else {
153        return Ok(None);
154    };
155    let cwd = tix_sdk::discovery::logical_cwd()?;
156
157    let mut best: Option<(String, PathBuf)> = None;
158    for name in ticket.worktrees.keys() {
159        let worktree_path = ticket_root.join(name);
160        let is_better = cwd.starts_with(&worktree_path)
161            && best.as_ref().is_none_or(|(_, current)| {
162                worktree_path.components().count() > current.components().count()
163            });
164        if is_better {
165            best = Some((ticket.worktrees[name].repo.clone(), worktree_path));
166        }
167    }
168    Ok(best)
169}

171#[cfg(unix)]
172fn is_executable(path: &std::path::Path) -> bool {
173    use std::os::unix::fs::PermissionsExt;
174    path.metadata()
175        .map(|m| m.is_file() && m.permissions().mode() & 0o111 != 0)
176        .unwrap_or(false)
177}
178
179#[cfg(not(unix))]
180fn is_executable(path: &std::path::Path) -> bool {
181    path.is_file()
182}

184/// Applies the plugin's outbound delta, if it wrote one.
185///
186/// The apply happens against a **fresh parse at apply time** under the
187/// exclusive lock — never the host's startup snapshot — so it merges with
188/// whatever a nested `tix` or a second terminal wrote meanwhile. After the
189/// ops land, the host re-deserializes the sections it has types for:
190/// a delta that breaks a host-owned section ([engine]/[cli]/[defaults] or
191/// [ticket]) is rejected wholesale and nothing is written. Plugin sections
192/// pass unvalidated — the host has no schema for them, and the plugin
193/// validated its own on the way in. Writes outside the plugin's own section
194/// are allowed, unsupported: applied if revalidation passes.
195fn apply_delta_if_present(
196    app: &App,
197    delta_path: &Path,
198    ticket_root: Option<&Path>,
199) -> Result<(), SdkError> {
200    // No file written (or nothing in it) means no changes.
201    let bytes = std::fs::read(delta_path).unwrap_or_default();
202    if bytes.is_empty() {
203        return Ok(());
204    }
205
206    let delta = Delta::parse(&bytes)?;
207    let target_path = match delta.target {
208        DeltaTarget::Global => app.context.config_path.clone(),
209        DeltaTarget::Ticket => ticket_root
210            .ok_or_else(|| {
211                SdkError::PluginImplementation(
212                    "delta targets the ticket document, but the plugin ran outside a ticket"
213                        .to_string(),
214                )
215            })?
216            .join(".tix")
217            .join("ticket.toml"),
218    };
219
220    debug!(target = ?delta.target, ops = delta.ops.len(), "applying plugin delta");
221    with_write(&target_path, |document| {
222        delta.apply_ops(document)?;
223
224        // Revalidate host-owned sections; reject the whole delta if any no
225        // longer parse. with_write discards on Err, so nothing is written.
226        let reject = |e: SdkError| {
227            SdkError::PluginImplementation(format!("delta breaks a host-owned section: {e}"))
228        };
229        match delta.target {
230            DeltaTarget::Global => {
231                document.section::<EngineConfig>("engine").map_err(reject)?;
232                document.section::<CliConfig>("cli").map_err(reject)?;
233                document.section::<Defaults>("defaults").map_err(reject)?;
234            }
235            DeltaTarget::Ticket => {
236                document.section::<TicketConfig>("ticket").map_err(reject)?;
237            }
238        }
239        Ok(())
240    })
