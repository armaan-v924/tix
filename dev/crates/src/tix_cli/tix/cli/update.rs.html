1//! `tix update` — self-update from GitHub releases.
2//!
3//! Lists the published releases, picks the newest one in the running
4//! binary's major, and replaces the running binary via temp file + rename.
5//! A newer major is reported but only installed with consent: under spec
6//! §2.3 a major means an on-disk migration, so crossing one silently would
7//! leave a binary that cannot read the user's files until `tix migrate`.
8
9use semver::Version;
10use serde::Deserialize;
11use sha2::Digest;
12use std::io::{IsTerminal, Read};
13use std::path::{Path, PathBuf};
14use tix_sdk::SdkError;
15use tracing::{debug, info, warn};

17/// GitHub repository the updater checks, overridable for testing.
18const DEFAULT_OWNER: &str = "armaan-v924";
19const DEFAULT_REPO: &str = "tix";
20const USER_AGENT: &str = concat!("tix-updater/", env!("CARGO_PKG_VERSION"));
21/// GitHub's maximum page size for the releases listing.
22const PER_PAGE: usize = 100;

24/// Update tix to the newest release in its major version
25#[derive(clap::Args, Debug)]
26pub struct Args {

28    #[arg(long)]
29    pub dry_run: bool,
30
31    /// Install a newer major version if one exists. A new major needs
32    /// `tix migrate` before it can use your config and tickets
33    #[arg(long)]

35}

62/// What the release list offers a binary at `current`.
63#[derive(Debug)]
64struct Selection<'a> {
65    /// Newest release in the running major, if newer than `current`.
66    same_major: Option<(Version, &'a Release)>,
67    /// Newest release in any later major.
68    newer_major: Option<(Version, &'a Release)>,
69}
70
71/// Picks candidates from a release list. Drafts and prereleases are skipped,
72/// as `releases/latest` skips them, and so are tags that are not semver.
73fn select<'a>(releases: &'a [Release], current: &Version) -> Selection<'a> {
74    let mut selection = Selection {
75        same_major: None,
76        newer_major: None,
77    };
78    for release in releases.iter().filter(|r| !r.draft && !r.prerelease) {
79        let Ok(version) = parse_tag(&release.tag_name) else {
80            debug!(tag = %release.tag_name, "skipping release with a non-semver tag");
81            continue;
82        };
83        let slot = if version.major == current.major && version > *current {
84            &mut selection.same_major
85        } else if version.major > current.major {
86            &mut selection.newer_major
87        } else {
88            continue;
89        };
90        if slot.as_ref().is_none_or(|(best, _)| version > *best) {
91            *slot = Some((version, release));
92        }
93    }
94    selection
95}

97/// The answer to "cross into the newer major?".
98#[derive(Debug, PartialEq, Eq)]
99enum Consent {
100    Cross,
101    Stay,
102    /// Nobody could be asked and no flag answered up front.
103    Refused,
104}
105
106/// The one decision point for crossing a major. The flag answers up front;
107/// otherwise only a terminal can be asked. Anything else is refused, never
108/// treated as a quiet "no": [`run`] turns that into a non-zero exit.
109fn consent(
110    flag: bool,
111    interactive: bool,
112    ask: impl FnOnce() -> Result<bool, SdkError>,
113) -> Result<Consent, SdkError> {
114    Ok(match (flag, interactive) {
115        (true, _) => Consent::Cross,
116        (false, true) if ask()? => Consent::Cross,
117        (false, true) => Consent::Stay,
118        (false, false) => Consent::Refused,
119    })
120}

122/// Lists releases, installs the newest in the running major, and asks
123/// before crossing into a newer one.
124///
125/// Asset naming follows the release workflow's convention:
126/// `tix-v<version>-<os>-<arch>.<tar.gz|zip>`, and every archive ships a
127/// `<asset>.sha256` sibling that the download is verified against. A release
128/// missing that sibling aborts the update rather than installing unverified:
129/// the workflow always publishes one, so its absence means either a
130/// half-uploaded release or an attacker withholding the file to steer the
131/// updater onto an unchecked path. The binary is replaced via temp file +
132/// rename in its own directory.
133///
134/// Without a terminal and without `--major`, a newer major is refused: the
135/// in-major update (if any) still installs, then the command fails so a
136/// script sees that a migration-bearing release is waiting.
137pub fn run(_app: &crate::tix::utils::App, args: Args) -> Result<(), SdkError> {
138    let target = detect_target()?;
139    let owner = std::env::var("TIX_UPDATE_OWNER").unwrap_or_else(|_| DEFAULT_OWNER.into());
140    let repo = std::env::var("TIX_UPDATE_REPO").unwrap_or_else(|_| DEFAULT_REPO.into());
141
142    let releases = fetch_releases(&owner, &repo)?;
143    let current = Version::parse(env!("CARGO_PKG_VERSION"))
144        .map_err(|e| SdkError::Message(format!("could not parse own version: {e}")))?;
145    let Selection {
146        same_major,
147        newer_major,
148    } = select(&releases, &current);
149
150    if args.dry_run {
151        if let Some((version, release)) = &newer_major {
152            let action = if args.major {
153                "would install"
154            } else {
155                "would skip (pass --major to install)"
156            };
157            println!(
158                "{action} tix {version}: a new major version, needs `tix migrate` ({})",
159                release.html_url
160            );
161        }
162        let chosen = if args.major && newer_major.is_some() {
163            &newer_major
164        } else {
165            &same_major
166        };
167        match chosen {
168            Some((version, release)) => {
169                print_plan(&target, &current, version, release)?;
170            }
171            None => println!("tix {current} is the newest {}.x release", current.major),
172        }
173        return Ok(());
174    }
175
176    let mut refused = None;
177    let chosen = match newer_major {
178        Some((version, release)) => {
179            let ask = || {
180                crate::tix::utils::confirm(&format!(
181                    "tix {version} is a new major version and needs `tix migrate` before \
182                     it can use your config and tickets ({}). Install it",
183                    release.html_url
184                ))
185            };
186            match consent(args.major, std::io::stdin().is_terminal(), ask)? {
187                Consent::Cross => Some((version, release)),
188                Consent::Stay => same_major,
189                Consent::Refused => {
190                    refused = Some((version, release));
191                    same_major
192                }
193            }
194        }
195        None => same_major,
196    };
197
198    match chosen {
199        Some((version, release)) => install(&target, &current, &version, release)?,
200        None => info!("tix {current} is the newest {}.x release", current.major),
201    }
202
203    if let Some((version, release)) = refused {
204        return Err(SdkError::Message(format!(
205            "tix {version} is a new major version and was not installed: it needs \
206             `tix migrate` before it can use your config and tickets ({}). Rerun \
207             `tix cli update --major` to install it",
208            release.html_url
209        )));
210    }
211    Ok(())
212}

214/// The asset and checksum names `release` must publish for this platform.
215fn assets<'a>(
216    target: &Target,
217    version: &Version,
218    release: &'a Release,
219) -> Result<(&'a ReleaseAsset, &'a ReleaseAsset), SdkError> {
220    let asset_name = format!(
221        "tix-v{version}-{}.{}",
222        target.asset_suffix, target.archive_ext
223    );
224    let asset = release
225        .assets
226        .iter()
227        .find(|asset| asset.name == asset_name)
228        .ok_or_else(|| {
229            SdkError::Message(format!(
230                "release {} has no asset '{asset_name}' for this platform",
231                release.tag_name
232            ))
233        })?;
234    let checksum_name = format!("{asset_name}.sha256");
235    let checksum_asset = release
236        .assets
237        .iter()
238        .find(|asset| asset.name == checksum_name)
239        .ok_or_else(|| {
240            SdkError::Message(format!(
241                "release {} publishes no '{checksum_name}' — refusing to install \
242                 an unverified binary",
243                release.tag_name
244            ))
245        })?;
246    Ok((asset, checksum_asset))
247}

249fn print_plan(
250    target: &Target,
251    current: &Version,
252    version: &Version,
253    release: &Release,
254) -> Result<(), SdkError> {
255    let (asset, checksum) = assets(target, version, release)?;
256    let destination = install_destination(target)?;
257    println!("would update tix {current} -> {version}");
258    println!("  asset:       {}", asset.name);
259    println!("  checksum:    {}", checksum.name);
260    println!("  destination: {}", destination.display());
261    Ok(())
262}

264fn install(
265    target: &Target,
266    current: &Version,
267    version: &Version,
268    release: &Release,
269) -> Result<(), SdkError> {
270    let (asset, checksum) = assets(target, version, release)?;
271    let destination = install_destination(target)?;
272
273    info!("updating tix {current} -> {version} ({})", asset.name);
274    let staging = tempfile::tempdir().map_err(SdkError::from)?;
275    let archive_path = staging.path().join(&asset.name);
276    download(&asset.browser_download_url, &archive_path)?;
277
278    verify_checksum(&archive_path, &checksum.browser_download_url)?;
279
280    let extracted = extract_archive(&archive_path, target)?;
281    install_binary(&extracted, &destination)?;
282
283    info!(
284        "updated tix {current} -> {version} ({})",
285        destination.display()
286    );
287    Ok(())
288}

290/// Every release, newest first as GitHub returns them, across all pages.
291fn fetch_releases(owner: &str, repo: &str) -> Result<Vec<Release>, SdkError> {
292    let mut releases = Vec::new();
293    for page in 1.. {
294        let url = format!(
295            "https://api.github.com/repos/{owner}/{repo}/releases?per_page={PER_PAGE}&page={page}"
296        );
297        let batch = ureq::get(&url)
298            .header("User-Agent", USER_AGENT)
299            .header("Accept", "application/vnd.github+json")
300            .call()
301            .map_err(|e| SdkError::Message(format!("could not list releases: {e}")))?
302            .body_mut()
303            .read_json::<Vec<Release>>()
304            .map_err(|e| SdkError::Message(format!("could not parse release JSON: {e}")))?;
305        let last = batch.len() < PER_PAGE;
306        releases.extend(batch);
307        if last {
308            break;
309        }
310    }
311    Ok(releases)
312}

319fn detect_target() -> Result<Target, SdkError> {
320    match (std::env::consts::OS, std::env::consts::ARCH) {
321        ("linux", "x86_64") => Ok(Target {
322            asset_suffix: "linux-x86_64",
323            archive_ext: "tar.gz",
324            exe_name: "tix",
325        }),
326        ("macos", "aarch64") => Ok(Target {
327            asset_suffix: "macos-aarch64",
328            archive_ext: "tar.gz",
329            exe_name: "tix",
330        }),
331        ("windows", "x86_64") => Ok(Target {
332            asset_suffix: "windows-x86_64",
333            archive_ext: "zip",
334            exe_name: "tix.exe",
335        }),
336        (os, arch) => Err(SdkError::Message(format!(
337            "self-update is not supported on {os}-{arch}"
338        ))),
339    }
340}

342fn download(url: &str, dest: &Path) -> Result<(), SdkError> {
343    let mut response = ureq::get(url)
344        .header("User-Agent", USER_AGENT)
345        .call()
346        .map_err(|e| SdkError::Message(format!("download failed: {e}")))?;
347    let mut file = std::fs::File::create(dest).map_err(SdkError::from)?;
348    std::io::copy(&mut response.body_mut().as_reader(), &mut file).map_err(SdkError::from)?;
349    Ok(())
350}

352/// Verifies the downloaded archive against the release's `.sha256` asset.
353fn verify_checksum(archive_path: &Path, checksum_url: &str) -> Result<(), SdkError> {
354    let mut response = ureq::get(checksum_url)
355        .header("User-Agent", USER_AGENT)
356        .call()
357        .map_err(|e| SdkError::Message(format!("checksum download failed: {e}")))?;
358    let mut text = String::new();
359    response
360        .body_mut()
361        .as_reader()
362        .take(4096)
363        .read_to_string(&mut text)
364        .map_err(SdkError::from)?;
365    verify_digest(archive_path, &parse_expected_digest(&text)?)
366}

368/// Pulls the digest out of a sha256sum-style checksum file: `<hex digest>`
369/// optionally followed by the filename it covers.
370///
371/// The shape is checked here rather than deferred to the comparison, so a body
372/// that is not a checksum at all — a CDN error page, a truncated upload —
373/// reports what it actually is instead of a mismatch against gibberish.
374fn parse_expected_digest(text: &str) -> Result<String, SdkError> {
375    let token = text
376        .split_whitespace()
377        .next()
378        .ok_or_else(|| SdkError::Message("empty checksum file".to_string()))?;
379    let digest = token.to_lowercase();
380    if digest.len() != 64 || !digest.bytes().all(|b| b.is_ascii_hexdigit()) {
381        return Err(SdkError::Message(format!(
382            "malformed checksum file: expected a 64-character hex digest, got '{}'",
383            token.chars().take(64).collect::<String>()
384        )));
385    }
386    Ok(digest)
387}

389/// Hashes the archive on disk and refuses anything but an exact match.
390fn verify_digest(archive_path: &Path, expected: &str) -> Result<(), SdkError> {
391    let bytes = std::fs::read(archive_path).map_err(SdkError::from)?;
392    let actual = hex(&sha2::Sha256::digest(&bytes));
393    if actual != expected {
394        return Err(SdkError::Message(format!(
395            "checksum mismatch: expected {expected}, got {actual} — not installing"
396        )));
397    }
398    debug!("checksum verified");
399    Ok(())
400}

402fn hex(bytes: &[u8]) -> String {
403    bytes.iter().map(|b| format!("{b:02x}")).collect()
404}

406/// Pulls the target executable out of the archive into its parent dir.
407///
408/// Entry paths are only matched on their file name, never joined onto the
409/// output directory, so a `../` name cannot write outside it. The match
410/// must be a regular file: unpacking a symlink named like the executable
411/// would leave a link that `install_binary` then follows.
412fn extract_archive(archive_path: &Path, target: &Target) -> Result<PathBuf, SdkError> {
413    let out_dir = archive_path
414        .parent()
415        .expect("staging archive always has a parent")
416        .join("extract");
417    std::fs::create_dir_all(&out_dir).map_err(SdkError::from)?;
418
419    match target.archive_ext {
420        "tar.gz" => extract_tar_gz(archive_path, &out_dir, target.exe_name),
421        "zip" => extract_zip(archive_path, &out_dir, target.exe_name),
422        other => Err(SdkError::Message(format!(
423            "unsupported archive format '{other}'"
424        ))),
425    }
426}

428fn extract_tar_gz(archive_path: &Path, out_dir: &Path, exe: &str) -> Result<PathBuf, SdkError> {
429    let file = std::fs::File::open(archive_path).map_err(SdkError::from)?;
430    let mut archive = tar::Archive::new(flate2::read::GzDecoder::new(file));
431    for entry in archive.entries().map_err(SdkError::from)? {
432        let mut entry = entry.map_err(SdkError::from)?;
433        let is_exe = entry
434            .path()
435            .map_err(SdkError::from)?
436            .file_name()
437            .is_some_and(|name| name == exe);
438        if is_exe {
439            if !entry.header().entry_type().is_file() {
440                return Err(not_a_regular_file(exe));
441            }
442            let dest = out_dir.join(exe);
443            entry.unpack(&dest).map_err(SdkError::from)?;
444            return Ok(dest);
445        }
446    }
447    Err(SdkError::Message(format!(
448        "executable '{exe}' not found in archive"
449    )))
450}

452fn extract_zip(archive_path: &Path, out_dir: &Path, exe: &str) -> Result<PathBuf, SdkError> {
453    let file = std::fs::File::open(archive_path).map_err(SdkError::from)?;
454    let mut archive = zip::ZipArchive::new(file)
455        .map_err(|e| SdkError::Message(format!("could not read zip archive: {e}")))?;
456    for index in 0..archive.len() {
457        let mut entry = archive
458            .by_index(index)
459            .map_err(|e| SdkError::Message(format!("could not read zip entry: {e}")))?;
460        let name = entry
461            .name()
462            .map_err(|e| SdkError::Message(format!("could not read zip entry name: {e}")))?;
463        let is_exe = Path::new(name.as_ref())
464            .file_name()
465            .is_some_and(|name| name == exe);
466        if is_exe {
467            if !entry.is_file() {
468                return Err(not_a_regular_file(exe));
469            }
470            let dest = out_dir.join(exe);
471            let mut out = std::fs::File::create(&dest).map_err(SdkError::from)?;
472            std::io::copy(&mut entry, &mut out).map_err(SdkError::from)?;
473            return Ok(dest);
474        }
475    }
476    Err(SdkError::Message(format!(
477        "executable '{exe}' not found in archive"
478    )))
479}

481fn not_a_regular_file(exe: &str) -> SdkError {
482    SdkError::Message(format!(
483        "'{exe}' in the archive is not a regular file; refusing to install it"
484    ))
485}

487/// Where the new binary lands: `TIX_INSTALL_PATH` override, else next to the
488/// currently running executable.
489fn install_destination(target: &Target) -> Result<PathBuf, SdkError> {
490    if let Ok(path) = std::env::var("TIX_INSTALL_PATH") {
491        return Ok(PathBuf::from(path));
492    }
493    let current = std::env::current_exe().map_err(SdkError::from)?;
494    let parent = current
495        .parent()
496        .ok_or_else(|| SdkError::Message("running executable has no parent directory".into()))?;
497    Ok(parent.join(target.exe_name))
498}

500/// Replaces the binary: rename when possible (same filesystem — atomic),
501/// copy as the cross-device fallback; executable bit restored on unix.
502fn install_binary(src: &Path, dest: &Path) -> Result<(), SdkError> {
503    if let Some(parent) = dest.parent() {
504        std::fs::create_dir_all(parent).map_err(SdkError::from)?;
505    }
506    if dest.exists() {
507        warn!(dest = %dest.display(), "replacing existing binary");
508    }
509    std::fs::rename(src, dest)
510        .or_else(|_| std::fs::copy(src, dest).map(|_| ()).map_err(SdkError::from))?;
511
512    #[cfg(unix)]
513    {
514        use std::os::unix::fs::PermissionsExt;
515        std::fs::set_permissions(dest, std::fs::Permissions::from_mode(0o755))
516            .map_err(SdkError::from)?;
517    }
518    Ok(())
519}

578    /// Writes a zip of regular files at `path`.
579    fn write_zip(path: &Path, files: &[(&str, &str)]) {
580        let mut zip = zip::ZipWriter::new(std::fs::File::create(path).unwrap());
581        for (name, contents) in files {
582            zip.start_file(*name, zip::write::SimpleFileOptions::default())
583                .unwrap();
584            std::io::Write::write_all(&mut zip, contents.as_bytes()).unwrap();
585        }
586        zip.finish().unwrap();
587    }

589    /// Appends one tar entry, writing the name straight into the header so
590    /// a test can use names `Header::set_path` would refuse, like `../x`.
591    fn append_tar(
592        builder: &mut tar::Builder<impl std::io::Write>,
593        name: &str,
594        kind: tar::EntryType,
595        link: Option<&str>,
596        contents: &str,
597    ) {
598        let mut header = tar::Header::new_gnu();
599        header.as_gnu_mut().unwrap().name[..name.len()].copy_from_slice(name.as_bytes());
600        header.set_entry_type(kind);
601        if let Some(link) = link {
602            header.set_link_name(link).unwrap();
603        }
604        header.set_size(contents.len() as u64);
605        header.set_mode(0o755);
606        header.set_cksum();
607        builder.append(&header, contents.as_bytes()).unwrap();
608    }

809    /// End to end over HTTP: the updater fetches a real `.sha256` body, and
810    /// the same archive with one byte flipped is refused.
811    #[test]
812    fn test_verify_checksum_over_http() {
813        let dir = tempfile::tempdir().unwrap();
814        let archive = dir.path().join("tix-v9.9.9-linux-x86_64.tar.gz");
815        std::fs::write(&archive, b"pretend this is a release archive").unwrap();
816        let digest = hex(&sha2::Sha256::digest(std::fs::read(&archive).unwrap()));
817        let body = format!("{digest}  tix-v9.9.9-linux-x86_64.tar.gz\n");
818
819        let url = serve_once(&body);
820        verify_checksum(&archive, &url).expect("untampered archive verifies");
821
822        // A download corrupted in flight — or swapped for someone else's
823        // binary — no longer matches the digest the release published.
824        std::fs::write(&archive, b"pretend this is a release archiveX").unwrap();
825        let url = serve_once(&body);
826        let err = verify_checksum(&archive, &url).expect_err("corrupted archive is refused");
827        assert!(
828            err.to_string().contains("checksum mismatch"),
829            "unexpected error: {err}"
830        );
831    }

833    /// Serves `body` to exactly one request and returns its URL. A whole HTTP
834    /// mock crate would be a heavier dependency than the two lines of protocol
835    /// this needs.
836    fn serve_once(body: &str) -> String {
837        let listener = std::net::TcpListener::bind("127.0.0.1:0").unwrap();
838        let url = format!("http://{}/checksum", listener.local_addr().unwrap());
839        let response = format!(
840            "HTTP/1.1 200 OK\r\nContent-Length: {}\r\nContent-Type: text/plain\r\n\r\n{body}",
841            body.len()
842        );
843        std::thread::spawn(move || {
844            use std::io::Write;
845            let (mut stream, _) = listener.accept().unwrap();
846            // Drain the request line/headers so the client is not writing into
847            // a socket we have already closed.
848            let mut request = [0u8; 1024];
849            let _ = std::io::Read::read(&mut stream, &mut request);
850            stream.write_all(response.as_bytes()).unwrap();
851            stream.flush().unwrap();
852        });
853        url
854    }
