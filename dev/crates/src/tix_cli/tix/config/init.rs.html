24/// Creates the global config interactively.
25///
26/// Prompts for each required `[cli]` field with a pre-filled default and
27/// then for the `[defaults]` seeds, which are optional and skipped by an
28/// empty answer. Writes to the resolved config path — the same resolution
29/// every command uses (`--config` > `TIX_CONFIG_PATH` > platform default),
30/// which is why this command is exempt from the config-must-exist check.
31/// Parent directories are created; an existing file is refused without
32/// `--force`.
33///
34/// The seeds are prompted here rather than left to `tix config set` because
35/// they are stamped into a ticket at creation and never re-read (#137): a
36/// `branch_prefix` set after the first `tix ticket setup` is a prefix that
37/// ticket will never have.
38///
39/// `--dry-run` runs every prompt and check — an existing file is still
40/// refused without `--force` — and prints the document instead of saving
41/// it, so `--dry-run --force` shows what an overwrite would leave. With
42/// `--output json` or `toml` it prints a [`Plan`] whose action carries the
43/// document's data.
44pub fn run(app: &crate::tix::utils::App, args: Args) -> Result<(), SdkError> {
45    let path = &app.context.config_path;
46    let exists = path.exists();
47    if exists && !args.force {
48        let refusal = SdkError::Message(format!(
49            "config already exists at {} — pass --force to overwrite",
50            path.display()
51        ));
52        return match args.dry_run {
53            true => Plan::new("config init").finish(app.output, Err(refusal)),
54            false => Err(refusal),
55        };
56    }
57
58    let home = dirs::home_dir();
59    let default_tickets = home.as_ref().map(|h| h.join("tickets"));
60    let tickets_directory = prompt(
61        "Tickets directory",
62        default_tickets.as_deref().and_then(|p| p.to_str()),
63    )?;
64    let default_code = home.as_ref().map(|h| h.join("code"));
65    let code_directory = prompt(
66        "Code directory (where repos are cloned)",
67        default_code.as_deref().and_then(|p| p.to_str()),
68    )?;
69
70    let mut document = TixDocument::empty();
71    // An explicit [cli] table, not the inline `cli = { … }` form index-based
72    // auto-creation would produce.
73    let mut cli = toml_edit::Table::new();
74    cli["tickets_directory"] = toml_edit::value(tickets_directory);
75    cli["code_directory"] = toml_edit::value(code_directory);
76    document.doc_mut()["cli"] = toml_edit::Item::Table(cli);
77
78    let defaults = prompt_defaults()?;
79    // No section at all rather than an empty one: absent is what every read
80    // of `[defaults]` already treats as "nothing seeded".
81    if !defaults.is_empty() {
82        document.doc_mut()["defaults"] = toml_edit::Item::Table(defaults);
83    }
84
85    // Cheap guard against this command and the section types drifting apart:
86    // what it scaffolds must be what they parse.
87    validate_section(&document, "cli")?;
88    validate_section(&document, "defaults")?;
89    if args.dry_run {
90        if app.output == OutputType::Default {
91            print!("{document}");
92            return Ok(());
93        }
94        let content: toml::Table = toml::from_str(&document.to_string())?;
95        let mut plan = Plan::new("config init");
96        plan.push(
97            Action::new(
98                if exists { "overwrite" } else { "write" },
99                format!(
100                    "would {} {}",
101                    if exists { "overwrite" } else { "write" },
102                    path.display()
103                ),
104            )
105            .with_path("path", path)
106            .with("content", content),
107        );
108        return plan.finish(app.output, Ok(()));
109    }
110    document.save(path)?;
111
112    tracing::info!("wrote {}", path.display());
113    Ok(())
114}

116/// Prompts for the `[defaults]` seeds, returning the table they populate —
117/// empty when every answer was skipped.
118///
119/// `repositories` is prompted as a comma-separated list rather than the
120/// TOML array literal `tix config set` takes: this is a prompt, not a TOML
121/// value slot, and the seed names registered repositories, which is why it
122/// comes last — on a first run there are none yet, and `tix repo add` is
123/// still ahead.
124fn prompt_defaults() -> Result<toml_edit::Table, SdkError> {
125    eprintln!("\n[defaults] — stamped into each new ticket at creation; blank to skip.");
126
127    let mut defaults = toml_edit::Table::new();
128    let scalars = [
129        ("branch_prefix", "Branch prefix"),
130        ("github_base_url", "GitHub base URL"),
131        ("default_repository_owner", "Default repository owner"),
132    ];
133    for (key, label) in scalars {
134        if let Some(answer) = prompt_optional(label)? {
135            defaults[key] = toml_edit::value(answer);
136        }
137    }
138
139    let repositories = prompt_optional("Repositories a new ticket includes, comma-separated")?
140        .map(|answer| {
141            answer
142                .split(',')
143                .map(str::trim)
144                .filter(|name| !name.is_empty())
145                .collect::<toml_edit::Array>()
146        })
147        .filter(|names| !names.is_empty());
148    if let Some(names) = repositories {
149        defaults["repositories"] = toml_edit::value(names);
150    }
151
152    Ok(defaults)
153}
