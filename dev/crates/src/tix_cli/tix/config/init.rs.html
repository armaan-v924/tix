11/// Create the global config with prompts or explicit options
12#[derive(clap::Args, Debug)]
13pub struct Args {
14    /// Overwrite an existing config file
15    #[arg(long)]
16    pub force: bool,
17
18    /// Print the config that would be written, using the same inputs as
19    /// initialization (with --output json or toml, a plan carrying it)
20    #[arg(long)]
21    pub dry_run: bool,
22
23    /// Directory for tickets. Supply both directories to skip all prompts
24    #[arg(long, value_name = "PATH", value_hint = clap::ValueHint::DirPath, requires = "code_directory",
25        value_parser = clap::builder::NonEmptyStringValueParser::new())]
26    tickets_directory: Option<String>,
27
28    /// Directory where repositories are cloned
29    #[arg(long, value_name = "PATH", value_hint = clap::ValueHint::DirPath, requires = "tickets_directory",
30        value_parser = clap::builder::NonEmptyStringValueParser::new())]
31    code_directory: Option<String>,
32
33    #[command(flatten)]
34    seeds: SeedInputs,
35}

61/// Creates the global config from prompts or explicit arguments.
62///
63/// Prompts for each required `[cli]` field with a pre-filled default and
64/// then for the `[defaults]` seeds, which are optional and skipped by an
65/// empty answer. Writes to the resolved config path — the same resolution
66/// every command uses (`--config` > `TIX_CONFIG_PATH` > platform default),
67/// which is why this command is exempt from the config-must-exist check.
68/// Parent directories are created; an existing file is refused without
69/// `--force`.
70///
71/// The seeds are prompted here rather than left to `tix config set` because
72/// they are stamped into a ticket at creation and never re-read (#137): a
73/// `branch_prefix` set after the first `tix ticket setup` is a prefix that
74/// ticket will never have.
75///
76/// `--dry-run` uses the same input mode and checks. An existing file is
77/// refused without `--force`. A successful preview prints the document instead of saving
78/// it, so `--dry-run --force` shows what an overwrite would leave. With
79/// `--output json` or `toml` it prints a [`Plan`] whose action carries the
80/// document's data.
81pub fn run(app: &crate::tix::utils::App, args: Args) -> Result<(), SdkError> {
82    let path = &app.context.config_path;
83    let exists = path.exists();
84    if exists && !args.force {
85        let refusal = already_exists(path);
86        return match args.dry_run {
87            true => Plan::new("config init").finish(app.output, Err(refusal)),
88            false => Err(refusal),
89        };
90    }
91
92    let (tickets_directory, code_directory, seeds) =
93        match (args.tickets_directory, args.code_directory) {
94            (Some(tickets), Some(code)) => (tickets, code, args.seeds),
95            (None, None) => prompt_inputs()?,
96            _ => {
97                return Err(SdkError::Message(
98                    "supply both --tickets-directory and --code-directory".into(),
99                ));
100            }
101        };
102
103    let mut document = TixDocument::empty();
104    // An explicit [cli] table, not the inline `cli = { … }` form index-based
105    // auto-creation would produce.
106    let mut cli = toml_edit::Table::new();
107    cli["tickets_directory"] = toml_edit::value(tickets_directory);
108    cli["code_directory"] = toml_edit::value(code_directory);
109    document.doc_mut()["cli"] = toml_edit::Item::Table(cli);
110
111    let defaults = seed_table(seeds);
112    // No section at all rather than an empty one: absent is what every read
113    // of `[defaults]` already treats as "nothing seeded".
114    if !defaults.is_empty() {
115        document.doc_mut()["defaults"] = toml_edit::Item::Table(defaults);
116    }
117
118    // Cheap guard against this command and the section types drifting apart:
119    // what it scaffolds must be what they parse.
120    validate_section(&document, "cli")?;
121    validate_section(&document, "defaults")?;
122    if args.dry_run {
123        if app.output == OutputType::Default {
124            print!("{document}");
125            return Ok(());
126        }
127        let content: toml::Table = toml::from_str(&document.to_string())?;
128        let mut plan = Plan::new("config init");
129        plan.push(
130            Action::new(
131                if exists { "overwrite" } else { "write" },
132                format!(
133                    "would {} {}",
134                    if exists { "overwrite" } else { "write" },
135                    path.display()
136                ),
137            )
138            .with_path("path", path)
139            .with("content", content),
140        );
141        return plan.finish(app.output, Ok(()));
142    }
143    if args.force {
144        document.save(path)?;
145    } else {
146        document.save_new(path).map_err(|error| match &error {
147            SdkError::Engine(TixError::IoError(io))
148                if io.kind() == std::io::ErrorKind::AlreadyExists =>
149            {
150                already_exists(path)
151            }
152            _ => error,
153        })?;
154    }
155
156    tracing::info!("wrote {}", path.display());
157    Ok(())
158}
159
160fn already_exists(path: &Path) -> SdkError {
161    SdkError::Message(format!(
162        "config already exists at {} — pass --force to overwrite",
163        path.display()
164    ))
165}

167fn prompt_inputs() -> Result<(String, String, SeedInputs), SdkError> {
168    let home = dirs::home_dir();
169    let default_tickets = home.as_ref().map(|h| h.join("tickets"));
170    let tickets = prompt(
171        "Tickets directory",
172        default_tickets.as_deref().and_then(|p| p.to_str()),
173    )?;
174    let default_code = home.as_ref().map(|h| h.join("code"));
175    let code = prompt(
176        "Code directory (where repos are cloned)",
177        default_code.as_deref().and_then(|p| p.to_str()),
178    )?;
179    eprintln!("\n[defaults] — stamped into each new ticket at creation; blank to skip.");
180    let seeds = SeedInputs {
181        branch_prefix: prompt_optional("Branch prefix")?,
182        github_base_url: prompt_optional("GitHub base URL")?,
183        default_repository_owner: prompt_optional("Default repository owner")?,
184        repositories: prompt_optional("Repositories a new ticket includes, comma-separated")?
185            .map(|answer| answer.split(',').map(str::to_owned).collect())
186            .unwrap_or_default(),
187    };
188    Ok((tickets, code, seeds))
189}
190
191fn seed_table(seeds: SeedInputs) -> toml_edit::Table {
192    let mut defaults = toml_edit::Table::new();
193    for (key, value) in [
194        ("branch_prefix", seeds.branch_prefix),
195        ("github_base_url", seeds.github_base_url),
196        ("default_repository_owner", seeds.default_repository_owner),
197    ] {
198        if let Some(value) = value {
199            defaults[key] = toml_edit::value(value);
200        }
201    }
202    let names: toml_edit::Array = seeds
203        .repositories
204        .iter()
205        .map(|name| name.trim())
206        .filter(|name| !name.is_empty())
207        .collect();
208    if !names.is_empty() {
209        defaults["repositories"] = toml_edit::value(names);
210    }
211    defaults
212}
