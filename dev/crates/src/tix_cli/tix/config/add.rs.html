1//! `tix config add` — append one element to a list key.
2
3use crate::tix::config::{ConfigKeyPath, edit, validate_section};
4use crate::tix::utils::confirm;
5use tix_sdk::SdkError;
6use tix_sdk::document::{TixDocument, with_write};
7
8/// Append a value to a list in the config document
9#[derive(clap::Args, Debug)]
10pub struct Args {
11    /// The `<section>.<key>` path of the list (e.g. `defaults.repositories`)
12    pub key: ConfigKeyPath,
13
14    /// The value to append (parsed as a TOML value when possible, else a
15    /// string)
16    pub value: String,
17
18    /// Append without confirming, even if the value is already in the list
19    #[arg(short, long)]
20    pub force: bool,
21}
22
23/// Appends to the list at `<section>.<key>`, creating it when unset.
24///
25/// Duplicates are allowed — a list in the config is the user's, and tix has
26/// no business deciding that two identical entries are a mistake — but
27/// appending one is nearly always a slip, so it is confirmed first.
28/// `--force` skips the question, which is also what a script wants.
29///
30/// The append matches the layout already in the file: one element per line
31/// in a multi-line array, inline in an inline one, comments where they were
32/// (`edit::push_in_layout`). As with `tix config set`, the edited section
33/// is re-deserialized before the write lands, so an element of the wrong
34/// type fails without touching the file.
35pub fn run(app: &crate::tix::utils::App, args: Args) -> Result<(), SdkError> {
36    let value = edit::parse_value(&args.value);
37
38    // Read once outside the write cycle so the question is not asked while
39    // holding the exclusive lock. Nothing is riding on the answer staying
40    // true: duplicates are permitted, so the worst a concurrent write can
41    // do is add one the user was not asked about.
42    if !args.force {
43        let document = TixDocument::load(&app.context.config_path)?;
44        let mut item = Some(document.doc().as_item());
45        for segment in args.key.segments() {
46            item = item.and_then(|item| item.get(segment));
47        }
48        let present = item
49            .and_then(|item| item.as_array())
50            .is_some_and(|array| edit::position(array, &value).is_some());
51        if present
52            && !confirm(&format!(
53                "'{}' is already in {}. Add it again?",
54                args.value, args.key
55            ))?
56        {
57            println!("aborted");
58            return Ok(());
59        }
60    }
61
62    let list = with_write(&app.context.config_path, |document| {
63        let array = edit::array_for_add(document, &args.key)?;
64        edit::push_in_layout(array, value);
65        let rendered = edit::render(array);
66        validate_section(document, args.key.section())?;
67        Ok(rendered)
68    })?;
69
70    println!(
71        "{} = {list} ({})",
72        args.key,
73        app.context.config_path.display()
74    );
75    Ok(())
76}
