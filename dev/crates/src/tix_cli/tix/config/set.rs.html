1//! `tix config set` — write one `[cli]` key through the format-preserving
2//! document layer.
3
4use crate::tix::config::{CliConfig, ConfigKey};
5use tix_sdk::SdkError;
6use tix_sdk::document::with_write;
7
8/// Set a value in the `[cli]` section
9#[derive(clap::Args, Debug)]
10pub struct Args {
11    /// The `[cli]` key to set
12    pub key: ConfigKey,
13
14    /// The new value (parsed as a TOML value when possible, else a string)
15    pub value: String,
16}
17
18/// Sets `[cli].<key>` and persists atomically.
19///
20/// The edit is a targeted path traversal over the format-preserving DOM —
21/// plugin tables, comments, and formatting elsewhere in the file survive
22/// byte-identical (applies to the CLI itself, not just plugin
23/// deltas). Before writing, the resulting `[cli]` section is re-deserialized
24/// into [`CliConfig`]; a value the type rejects fails the whole write, and
25/// nothing lands on disk ([`with_write`] discards on error).
26///
27/// The whole cycle runs under the exclusive advisory lock (#67), so
28/// concurrent `tix config set` invocations serialize rather than clobber.
29pub fn run(app: &crate::tix::utils::App, args: Args) -> Result<(), SdkError> {
30    with_write(&app.context.config_path, |document| {
31        // `1` stays an integer, `true` a bool; anything that isn't valid
32        // TOML (a bare path, say) is a string.
33        let value: toml_edit::Value = args
34            .value
35            .parse()
36            .unwrap_or_else(|_| toml_edit::Value::from(args.value.as_str()));
37        // entry().or_insert with an explicit table so a missing [cli]
38        // materializes as a real section, not an inline `cli = { … }`.
39        document
40            .doc_mut()
41            .entry("cli")
42            .or_insert(toml_edit::table())[args.key.toml_key()] = toml_edit::Item::Value(value);
43
44        // Revalidate: the edited section must still parse as CliConfig
45        // (deny_unknown_fields; type-incompatible values fail here).
46        let _valid: CliConfig = document.section("cli")?.ok_or_else(|| {
47            SdkError::Message("edit produced no [cli] section — this is a bug".to_string())
48        })?;
49        Ok(())
50    })?;
51    println!(
52        "{} = {} ({})",
53        args.key.toml_key(),
54        args.value,
55        app.context.config_path.display()
56    );
57    Ok(())
58}
