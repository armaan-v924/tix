44/// Prompts on stderr and reads one line from stdin; an empty answer takes
45/// `default` when one is offered.
46///
47/// The prompt goes to stderr so interactive commands stay pipeable — stdout
48/// carries only results.
49pub fn prompt(label: &str, default: Option<&str>) -> Result<String, tix_sdk::SdkError> {
50    let answer = ask(&match default {
51        Some(default) => format!("{label} [{default}]"),
52        None => label.to_string(),
53    })?;
54
55    match (answer.is_empty(), default) {
56        (false, _) => Ok(answer),
57        (true, Some(default)) => Ok(default.to_string()),
58        (true, None) => Err(tix_sdk::SdkError::Message(format!("{label} is required"))),
59    }
60}

62/// Prompts for a value that may be left unset: an empty answer is `None`,
63/// not an error.
64///
65/// The optional counterpart to [`prompt`], for keys whose absence is a
66/// meaningful state rather than a missing answer — every `[defaults]` seed.
67pub fn prompt_optional(label: &str) -> Result<Option<String>, tix_sdk::SdkError> {
68    let answer = ask(&format!("{label} (optional)"))?;
69    Ok((!answer.is_empty()).then_some(answer))
70}

72/// Asks a yes/no question, defaulting to no.
73///
74/// One place for the CLI's confirmation convention: `[y/N]`, an empty
75/// answer declines, and only `y`/`yes` accepts. A command that offers a way
76/// past the question spells it `--force`.
77///
78/// Reads through [`ask`] rather than [`prompt`], whose own `[default]`
79/// suffix would print a second bracket after the `[y/N]`.
80pub fn confirm(question: &str) -> Result<bool, tix_sdk::SdkError> {
81    let answer = ask(&format!("{question} [y/N]"))?;
82    Ok(matches!(answer.to_lowercase().as_str(), "y" | "yes"))
83}

85/// Writes `label` to stderr and reads one trimmed line from stdin.
86fn ask(label: &str) -> Result<String, tix_sdk::SdkError> {
87    use std::io::{BufRead, Write};
88
89    eprint!("{label}: ");
90    std::io::stderr().flush().map_err(tix_sdk::SdkError::from)?;
91
92    let mut answer = String::new();
93    std::io::stdin()
94        .lock()
95        .read_line(&mut answer)
96        .map_err(tix_sdk::SdkError::from)?;
97    Ok(answer.trim().to_string())
98}
