1//! The plan a `--dry-run` prints, in whichever `--output` format was asked
2//! for.
3//!
4//! Every dry run builds a [`Plan`] — one [`Action`] per thing the real run
5//! would do — and hands it to [`Plan::finish`] along with whether the real
6//! run would have succeeded. The default format prints each action's
7//! one-line `summary`; JSON and TOML print the whole plan, so a script reads
8//! the same shape from every command:
9//!
10//! ```text
11//! {
12//!   "dry_run": true,
13//!   "command": "destroy",
14//!   "would_succeed": false,
15//!   "actions": [
16//!     { "action": "prune", "summary": "would prune worktree api (…)", "worktree": "api", … }
17//!   ]
18//! }
19//! ```
20//!
21//! A plan the real run would not finish is still printed, up to the action
22//! it would stop at, and then the error is returned as usual — stderr and a
23//! non-zero exit — so `would_succeed` and the exit code always agree.
24
25use crate::tix::utils::OutputType;
26use serde::Serialize;
27use tix_sdk::SdkError;
28
29/// Everything a dry run found the real run would do.
30#[derive(Debug, Serialize)]
31pub struct Plan {
32    /// Always `true`: marks the output as a plan rather than a result.
33    dry_run: bool,
34    /// The command planned, as typed (`destroy`, `config set`, …).
35    command: &'static str,
36    /// Whether the real run would get through every action and succeed.
37    would_succeed: bool,
38    /// What the real run would do, in order.
39    actions: Vec<Action>,
40}
41
42/// One step of a [`Plan`]: what would happen, said once for people
43/// (`summary`) and once for scripts (`action` plus its details).
44#[derive(Debug, Serialize)]
45pub struct Action {
46    /// The kind of step, kebab-case (`prune`, `set`, `install`, …).
47    action: &'static str,
48    /// The line the default output prints.
49    summary: String,
50    /// What the step acts on: worktree, key, path, value, and so on.
51    #[serde(flatten)]
52    details: toml::Table,
53}
54
55impl Plan {
56    /// An empty plan for `command`.
57    pub fn new(command: &'static str) -> Self {
58        Self {
59            dry_run: true,
60            command,
61            would_succeed: true,
62            actions: Vec::new(),
63        }
64    }
65
66    /// Appends `action`.
67    pub fn push(&mut self, action: Action) {
68        self.actions.push(action);
69    }
70
71    /// Prints the plan in `output`, then returns `outcome`: what the real
72    /// run would have returned once it got this far. An `Err` marks the
73    /// plan as one that would not succeed.
74    pub fn finish(
75        mut self,
76        output: OutputType,
77        outcome: Result<(), SdkError>,
78    ) -> Result<(), SdkError> {
79        self.would_succeed = outcome.is_ok();
80        self.print(output)?;
81        outcome
82    }
83
84    fn print(&self, output: OutputType) -> Result<(), SdkError> {
85        match output {
86            OutputType::Default => {
87                for action in &self.actions {
88                    println!("{}", action.summary);
89                }
90            }
91            OutputType::Json => {
92                let json = serde_json::to_string_pretty(self)
93                    .map_err(|e| SdkError::Message(format!("json conversion failed: {e}")))?;
94                println!("{json}");
95            }
96            OutputType::Toml => print!("{}", toml::to_string(self)?),
97        }
98        Ok(())
99    }
100}
101
102impl Action {
103    /// A step of kind `action`, printed as `summary`.
104    pub fn new(action: &'static str, summary: impl Into<String>) -> Self {
105        Self {
106            action,
107            summary: summary.into(),
108            details: toml::Table::new(),
109        }
110    }
111
112    /// Adds the detail `name`.
113    pub fn with(mut self, name: &str, value: impl Into<toml::Value>) -> Self {
114        self.details.insert(name.to_string(), value.into());
115        self
116    }
117
118    /// Adds the detail `name` when there is one to add.
119    pub fn with_some(self, name: &str, value: Option<impl Into<toml::Value>>) -> Self {
120        match value {
121            Some(value) => self.with(name, value),
122            None => self,
123        }
124    }
125
126    /// Adds the path detail `name`, as the string the default output shows.
127    pub fn with_path(self, name: &str, path: &std::path::Path) -> Self {
128        self.with(name, path.display().to_string())
129    }
130}

132#[cfg(test)]
133mod tests {
134    use super::*;
135
136    fn sample() -> Plan {
137        let mut plan = Plan::new("config set");
138        plan.push(
139            Action::new("set", "would set defaults.repositories = [\"api\"]")
140                .with("key", "defaults.repositories")
141                .with("value", toml::Value::Array(vec![toml::Value::from("api")]))
142                .with_some("previous", None::<String>)
143                .with_path("path", std::path::Path::new("/c/config.toml")),
144        );
145        plan
146    }
147
148    /// Details sit beside `action` and `summary`, absent ones are left out,
149    /// and values keep their type.
150    #[test]
151    fn test_json_shape() {
152        let value = serde_json::to_value(sample()).unwrap();
153        assert_eq!(value["dry_run"], true);
154        assert_eq!(value["command"], "config set");
155        let action = &value["actions"][0];
156        assert_eq!(action["action"], "set");
157        assert_eq!(action["key"], "defaults.repositories");
158        assert_eq!(action["value"], serde_json::json!(["api"]));
159        assert!(action.get("previous").is_none());
160    }
161
162    /// The same plan renders as TOML that parses back to the same data.
163    #[test]
164    fn test_toml_round_trips() {
165        let text = toml::to_string(&sample()).unwrap();
166        let parsed: toml::Table = toml::from_str(&text).unwrap();
167        assert_eq!(parsed["would_succeed"].as_bool(), Some(true));
168        let action = &parsed["actions"].as_array().unwrap()[0];
169        assert_eq!(action["path"].as_str(), Some("/c/config.toml"));
170    }
171
172    /// An error outcome is passed through and marks the plan.
173    #[test]
174    fn test_finish_reports_failure() {
175        let plan = sample();
176        let outcome = plan.finish(OutputType::Json, Err(SdkError::Message("no".into())));
177        assert!(outcome.is_err());
178    }
179}
