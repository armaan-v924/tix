44/// Destroys a ticket workspace: every tracked worktree pruned, then the
45/// directory removed.
46///
47/// **Default path** — checks each worktree before confirming or pruning.
48/// The directory is deleted only after every worktree prunes cleanly.
49/// Each successful prune deletes its entry
50/// from the ticket document immediately, so an abort partway leaves a
51/// resolvable ticket for a retry.
52///
53/// A worktree whose directory was deleted by hand is dropped with a warning
54/// rather than failing. One that is there but no longer a git checkout
55/// cannot be checked for uncommitted work, so the default path refuses up
56/// front, before confirming; `--force` deletes it with the rest.
57///
58/// **`--force`** — no confirmation, worktrees force-pruned (uncommitted
59/// changes discarded), and the directory is deleted **even if pruning
60/// fails** — forced deletion means deleted. Whatever had to be left behind
61/// (e.g. stale registrations in the source repos) is warned about, with a
62/// pointer at `git worktree prune`.
63///
64/// **`--dry-run`** — the same checks, nothing done: no question is asked,
65/// and each worktree is run through [`plan_prune`] instead of pruned. The
66/// result is a [`Plan`] printed in the `--output` format. Without `--force`
67/// it stops where the real run would stop, on the same refusal; with it, it
68/// says what force would get past (how many uncommitted changes it would
69/// discard), and a worktree that would fail to prune is reported and the
70/// plan carries on, as the real forced run would.
71pub fn run(app: &crate::tix::utils::App, args: Args) -> Result<(), SdkError> {
72    let mut output = TicketOutput::new("destroy", args.dry_run);
73    let result = execute(app, args, &mut output);
74    output.finish(app.output, result)
75}

77fn execute(
78    app: &crate::tix::utils::App,
79    args: Args,
80    output: &mut TicketOutput,
81) -> Result<(), SdkError> {
82    let selector = args.target.as_ref().or(args.shared.ticket.as_ref());
83    let root = require_ticket_root(&app.context, selector, args.shared.project.as_deref())?;
84    output.ticket(&root)?;
85    let ticket = load_ticket_config(&root)?;
86
87    let engine = load_engine_config(&app.context)?;
88
89    let mut entries: Vec<_> = ticket.worktrees.iter().collect();
90    entries.sort_by(|a, b| a.0.cmp(b.0));
91    if let Some(plan) = &mut output.plan {
92        return dry_run(plan, &root, &ticket, &engine, &entries, args.force);
93    }
94
95    if !args.force {
96        for (name, entry) in &entries {
97            check_worktree(&root, name, entry, &engine, false).map_err(|error| {
98                output.worktree_failed(name, &root.join(name), &entry.repo, &error);
99                output.set_error(&error);
100                refusal(name, error, false)
101            })?;
102        }
103    }
104
105    if !args.force
106        && !args.yes
107        && !crate::tix::utils::confirm(&format!(
108            "Destroy ticket '{}' at {}?",
109            ticket.key,
110            root.display()
111        ))?
112    {
113        info!("aborted");
114        output.declined();
115        return Ok(());
116    }
117
118    let mut leftovers: Vec<String> = Vec::new();
119    for (name, entry) in entries {
120        let pruned = prune_worktree(&root, name, entry, &engine, args.force);
121        let path = root.join(name);
122        match (pruned, args.force) {
123            (Ok(status), _) => {
124                output.pruned(name, &path, &entry.repo, status);
125                if status == WorktreeStatus::Ok {
126                    info!("pruned worktree {name}");
127                }
128            }
129            // Forced deletion means deleted: pruning failures are warnings.
130            (Err(e), true) => {
131                output.worktree_failed(name, &path, &entry.repo, &e);
132                warn!(name = %name, error = %e, "could not prune worktree, continuing (--force)");
133                leftovers.push(format!("{} ({})", name, entry.repo));
134            }
135            // The default path aborts before deleting anything further.
136            (Err(e), false) => {
137                output.worktree_failed(name, &path, &entry.repo, &e);
138                output.set_error(&e);
139                return Err(refusal(name, e, false));
140            }
141        }
142    }
143
144    std::fs::remove_dir_all(&root).map_err(SdkError::from)?;
145    output.changed();
146    info!("destroyed {} ({})", ticket.key, root.display());
147
148    if !leftovers.is_empty() {
149        warn!(
150            "left dangling: {} — run `git worktree prune` in the affected source repos",
151            leftovers.join(", ")
152        );
153    }
154    Ok(())
155}

157/// Checks deletion of a single worktree before any worktree is pruned.
158fn check_worktree(
159    root: &Path,
160    name: &str,
161    entry: &WorktreeConfig,
162    engine: &EngineConfig,
163    force: bool,
164) -> Result<PrunePlan, SdkError> {
165    let path = root.join(name);
166    if !force && WorktreeStatus::of(&path) == WorktreeStatus::NotAGitRepository {
167        return Err(SdkError::Message(format!(
168            "worktree '{name}' at {} is not a git repository, so tix cannot check it for \
169             uncommitted work — move anything you need out of it, then pass --force",
170            path.display()
171        )));
172    }
173    plan_prune(root, name, entry, engine, force)
174}

176/// `--dry-run`: what [`run`] would do to the ticket at `root`, planned and
177/// printed in the requested output format, with nothing done.
178fn dry_run(
179    plan: &mut Plan,
180    root: &Path,
181    ticket: &TicketConfig,
182    engine: &EngineConfig,
183    entries: &[(&String, &WorktreeConfig)],
184    force: bool,
185) -> Result<(), SdkError> {
186    for (name, entry) in entries {
187        let path = root.join(name);
188        let found = match check_worktree(root, name, entry, engine, force) {
189            Ok(found) => found,
190            Err(e) if force => {
191                plan.push(
192                    Action::new(
193                        "prune-fails",
194                        format!(
195                            "would fail to prune worktree {name} ({e}) and delete it anyway \
196                             (--force), leaving its registration in {} dangling",
197                            entry.repo
198                        ),
199                    )
200                    .with("worktree", name.as_str())
201                    .with("repo", entry.repo.as_str())
202                    .with_path("path", &path)
203                    .with("forced", true)
204                    .with("reason", e.to_string()),
205                );
206                continue;
207            }
208            Err(e) => {
209                plan.clear_actions();
210                plan.set_error(&e);
211                return Err(refusal(name, e, true));
212            }
213        };
214        let (kind, summary) = match &found {
215            PrunePlan::Prune => (
216                "prune",
217                format!("would prune worktree {name} ({})", path.display()),
218            ),
219            PrunePlan::Discard { changes } => (
220                "prune",
221                format!(
222                    "would prune worktree {name} ({}), discarding {} (--force)",
223                    path.display(),
224                    uncommitted(*changes)
225                ),
226            ),
227            PrunePlan::ForcePast { reason } => (
228                "prune",
229                format!(
230                    "would prune worktree {name} ({}) past: {reason} (--force)",
231                    path.display()
232                ),
233            ),
234            PrunePlan::Missing => (
235                "drop",
236                format!("would drop worktree {name}: missing on disk"),
237            ),
238            // Only reachable with --force: the pre-flight refused it otherwise.
239            PrunePlan::NotAGitRepository => (
240                "delete-unchecked",
241                format!(
242                    "would delete {} unchecked: not a git repository (--force)",
243                    path.display()
244                ),
245            ),
246        };
247        plan.push(prune_action(
248            kind,
249            summary,
250            name,
251            &entry.repo,
252            &path,
253            &found,
254        ));
255    }
256
257    let unmanaged = match unmanaged_entries(root, ticket) {
258        Ok(entries) => entries,
259        Err(e) => {
260            plan.clear_actions();
261            return Err(e.into());
262        }
263    };
264    let mut summary = format!("would destroy {} ({})", ticket.key, root.display());
265    if !unmanaged.is_empty() {
266        summary.push_str(&format!(
267            ", including unmanaged ticket entries: {unmanaged:?}"
268        ));
269    }
270    plan.push(
271        Action::new("delete-ticket", summary)
272            .with("ticket", ticket.key.as_str())
273            .with_path("path", root)
274            .with("unmanaged_entries", unmanaged),
275    );
276    Ok(())
277}

279/// Direct ticket-root entries outside the tix metadata directory and
280/// tracked worktree directories.
281fn unmanaged_entries(root: &Path, ticket: &TicketConfig) -> std::io::Result<Vec<String>> {
282    let mut entries = Vec::new();
283    for entry in std::fs::read_dir(root)? {
284        let entry = entry?;
285        let name = entry.file_name();
286        if name == TIX_DIR {
287            continue;
288        }
289        if name
290            .to_str()
291            .is_some_and(|name| ticket.worktrees.contains_key(name))
292            && entry.file_type()?.is_dir()
293        {
294            continue;
295        }
296        entries.push(name.to_string_lossy().into_owned());
297    }
298    entries.sort();
299    Ok(entries)
300}

302/// The error a non-forced destroy stops with when `name` cannot be pruned —
303/// or, for `dry_run`, would stop with.
304fn refusal(name: &str, error: SdkError, dry_run: bool) -> SdkError {
305    let fix = match error {
306        SdkError::Message(_) => return error,
307        SdkError::Engine(TixError::Dirty { .. }) => "commit or stash the changes",
308        _ => "fix it",
309    };
310    SdkError::Message(if dry_run {
311        format!("destroy would stop at worktree '{name}': {error}. {fix} (or pass --force) first")
312    } else {
313        format!(
314            "could not prune worktree '{name}': {error} — nothing further was deleted; \
315             {fix} (or pass --force) and retry"
316        )
317    })
