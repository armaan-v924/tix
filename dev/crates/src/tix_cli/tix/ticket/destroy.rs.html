14/// Destroy a ticket: prune its worktrees, delete its directory
15#[derive(clap::Args, Debug)]
16pub struct Args {
17    #[command(flatten)]
18    pub shared: TicketSharedArgs,
19
20    /// The ticket to destroy (id under the tickets directory, or a path);

38/// Destroys a ticket workspace: every tracked worktree pruned, then the
39/// directory removed.
40///
41/// **Default path** — confirms first, prunes each worktree *without* force
42/// (dirty worktrees abort), and deletes the directory only after every
43/// tracked worktree pruned cleanly. Each successful prune deletes its entry
44/// from the ticket document immediately, so an abort partway leaves a
45/// resolvable ticket for a retry.
46///
47/// A worktree whose directory was deleted by hand is dropped with a warning
48/// rather than failing. One that is there but no longer a git checkout
49/// cannot be checked for uncommitted work, so the default path refuses up
50/// front, before confirming; `--force` deletes it with the rest.
51///
52/// **`--force`** — no confirmation, worktrees force-pruned (uncommitted
53/// changes discarded), and the directory is deleted **even if pruning
54/// fails** — forced deletion means deleted. Whatever had to be left behind
55/// (e.g. stale registrations in the source repos) is warned about, with a
56/// pointer at `git worktree prune`.
57///
58/// **`--dry-run`** — the same checks, nothing done: no question is asked,
59/// and each worktree is run through [`plan_prune`] instead of pruned. The
60/// result is a [`Plan`] printed in the `--output` format. Without `--force`
61/// it stops where the real run would stop, on the same refusal; with it, it
62/// says what force would get past (how many uncommitted changes it would
63/// discard), and a worktree that would fail to prune is reported and the
64/// plan carries on, as the real forced run would.
65pub fn run(app: &crate::tix::utils::App, args: Args) -> Result<(), SdkError> {
66    let selector = args.target.as_ref().or(args.shared.ticket.as_ref());
67    let root = require_ticket_root(&app.context, selector, args.shared.project.as_deref())?;
68    let ticket = load_ticket_config(&root)?;
69
70    let engine = load_engine_config(&app.context)?;
71
72    if args.dry_run {
73        return dry_run(app, &root, &ticket, &engine, args.force);
74    }
75
76    if !args.force {
77        preflight(&root, &ticket)?;
78    }
79
80    if !args.force
81        && !crate::tix::utils::confirm(&format!(
82            "Destroy ticket '{}' at {}?",
83            ticket.key,
84            root.display()
85        ))?
86    {
87        info!("aborted");
88        return Ok(());
89    }
90
91    let mut entries: Vec<(&String, &WorktreeConfig)> = ticket.worktrees.iter().collect();
92    entries.sort_by(|a, b| a.0.cmp(b.0));
93
94    let mut leftovers: Vec<String> = Vec::new();
95    for (name, entry) in entries {
96        let pruned = prune_worktree(&root, name, entry, &engine, args.force);
97        match (pruned, args.force) {
98            (Ok(WorktreeStatus::Ok), _) => info!("pruned worktree {name}"),
99            // Already gone or no checkout: prune_worktree warned.
100            (Ok(_), _) => {}
101            // Forced deletion means deleted: pruning failures are warnings.
102            (Err(e), true) => {
103                warn!(name = %name, error = %e, "could not prune worktree, continuing (--force)");
104                leftovers.push(format!("{} ({})", name, entry.repo));
105            }
106            // The default path aborts before deleting anything further.
107            (Err(e), false) => return Err(refusal(name, e, false)),
108        }
109    }
110
111    std::fs::remove_dir_all(&root).map_err(SdkError::from)?;
112    info!("destroyed {} ({})", ticket.key, root.display());
113
114    if !leftovers.is_empty() {
115        warn!(
116            "left dangling: {} — run `git worktree prune` in the affected source repos",
117            leftovers.join(", ")
118        );
119    }
120    Ok(())
121}

123/// Refuses a ticket holding a worktree directory that is no longer a
124/// checkout: deleting the ticket deletes it too, and nothing can check it
125/// for uncommitted work first. `--force` skips this.
126fn preflight(root: &Path, ticket: &tix_sdk::TicketConfig) -> Result<(), SdkError> {
127    let mut names: Vec<&String> = ticket.worktrees.keys().collect();
128    names.sort();
129    match names
130        .into_iter()
131        .find(|name| WorktreeStatus::of(&root.join(name)) == WorktreeStatus::NotAGitRepository)
132    {
133        Some(name) => Err(SdkError::Message(format!(
134            "worktree '{name}' at {} is not a git repository, so tix cannot check it for \
135             uncommitted work — move anything you need out of it, then pass --force",
136            root.join(name).display()
137        ))),
138        None => Ok(()),
139    }
140}

142/// `--dry-run`: what [`run`] would do to the ticket at `root`, planned and
143/// printed in the requested output format, with nothing done.
144fn dry_run(
145    app: &crate::tix::utils::App,
146    root: &Path,
147    ticket: &tix_sdk::TicketConfig,
148    engine: &tix_sdk::EngineConfig,
149    force: bool,
150) -> Result<(), SdkError> {
151    let mut plan = Plan::new("destroy");
152    if !force && let Err(e) = preflight(root, ticket) {
153        return plan.finish(app.output, Err(e));
154    }
155
156    let mut entries: Vec<(&String, &WorktreeConfig)> = ticket.worktrees.iter().collect();
157    entries.sort_by(|a, b| a.0.cmp(b.0));
158
159    for (name, entry) in entries {
160        let path = root.join(name);
161        let found = match plan_prune(root, name, entry, engine, force) {
162            Ok(found) => found,
163            Err(e) if force => {
164                plan.push(
165                    Action::new(
166                        "prune-fails",
167                        format!(
168                            "would fail to prune worktree {name} ({e}) and delete it anyway \
169                             (--force), leaving its registration in {} dangling",
170                            entry.repo
171                        ),
172                    )
173                    .with("worktree", name.as_str())
174                    .with("repo", entry.repo.as_str())
175                    .with_path("path", &path)
176                    .with("forced", true)
177                    .with("reason", e.to_string()),
178                );
179                continue;
180            }
181            Err(e) => return plan.finish(app.output, Err(refusal(name, e, true))),
182        };
183        let (kind, summary) = match &found {
184            PrunePlan::Prune => (
185                "prune",
186                format!("would prune worktree {name} ({})", path.display()),
187            ),
188            PrunePlan::Discard { changes } => (
189                "prune",
190                format!(
191                    "would prune worktree {name} ({}), discarding {} (--force)",
192                    path.display(),
193                    uncommitted(*changes)
194                ),
195            ),
196            PrunePlan::ForcePast { reason } => (
197                "prune",
198                format!(
199                    "would prune worktree {name} ({}) past: {reason} (--force)",
200                    path.display()
201                ),
202            ),
203            PrunePlan::Missing => (
204                "drop",
205                format!("would drop worktree {name}: missing on disk"),
206            ),
207            // Only reachable with --force: the pre-flight refused it otherwise.
208            PrunePlan::NotAGitRepository => (
209                "delete-unchecked",
210                format!(
211                    "would delete {} unchecked: not a git repository (--force)",
212                    path.display()
213                ),
214            ),
215        };
216        plan.push(prune_action(
217            kind,
218            summary,
219            name,
220            &entry.repo,
221            &path,
222            &found,
223        ));
224    }
225
226    plan.push(
227        Action::new(
228            "delete-ticket",
229            format!("would destroy {} ({})", ticket.key, root.display()),
230        )
231        .with("ticket", ticket.key.as_str())
232        .with_path("path", root),
233    );
234    plan.finish(app.output, Ok(()))
235}

237/// The error a non-forced destroy stops with when `name` cannot be pruned —
238/// or, for `dry_run`, would stop with.
239fn refusal(name: &str, error: SdkError, dry_run: bool) -> SdkError {
240    let fix = match error {
241        SdkError::Engine(TixError::Dirty { .. }) => "commit or stash the changes",
242        _ => "fix it",
243    };
244    SdkError::Message(if dry_run {
245        format!(
246            "destroy would stop at worktree '{name}': {error} — nothing would be deleted \
247             past it; {fix} (or pass --force) first"
248        )
249    } else {
250        format!(
251            "could not prune worktree '{name}': {error} — nothing further was deleted; \
252             {fix} (or pass --force) and retry"
253        )
254    })
