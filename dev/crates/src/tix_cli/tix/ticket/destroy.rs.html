1//! `tix ticket destroy` — prune every worktree, then delete the ticket.
2
3use crate::tix::config::load_engine_config;
4use crate::tix::ticket::{
5    TicketRef, TicketSharedArgs, WorktreeStatus, load_ticket_config, prune_worktree,
6    require_ticket_root,
7};
8use tix_sdk::{SdkError, TixError, WorktreeConfig};
9use tracing::{info, warn};
10
11/// Destroy a ticket: prune its worktrees, delete its directory
12#[derive(clap::Args, Debug)]
13pub struct Args {
14    #[command(flatten)]
15    pub shared: TicketSharedArgs,
16
17    /// The ticket to destroy (id under the tickets directory, or a path);
18    /// discovered from cwd when omitted
19    #[arg(value_name = "TICKET", conflicts_with = "ticket")]
20    pub target: Option<TicketRef>,
21
22    /// Skip confirmation, force-prune (discarding uncommitted changes), and
23    /// delete the ticket directory even if pruning fails
24    #[arg(short, long)]
25    pub force: bool,
26}
27
28/// Destroys a ticket workspace: every tracked worktree pruned, then the
29/// directory removed.
30///
31/// **Default path** — confirms first, prunes each worktree *without* force
32/// (dirty worktrees abort), and deletes the directory only after every
33/// tracked worktree pruned cleanly. Each successful prune deletes its entry
34/// from the ticket document immediately, so an abort partway leaves a
35/// resolvable ticket for a retry.
36///
37/// A worktree whose directory was deleted by hand is dropped with a warning
38/// rather than failing. One that is there but no longer a git checkout
39/// cannot be checked for uncommitted work, so the default path refuses up
40/// front, before confirming; `--force` deletes it with the rest.
41///
42/// **`--force`** — no confirmation, worktrees force-pruned (uncommitted
43/// changes discarded), and the directory is deleted **even if pruning
44/// fails** — forced deletion means deleted. Whatever had to be left behind
45/// (e.g. stale registrations in the source repos) is warned about, with a
46/// pointer at `git worktree prune`.
47pub fn run(app: &crate::tix::utils::App, args: Args) -> Result<(), SdkError> {
48    let selector = args.target.as_ref().or(args.shared.ticket.as_ref());
49    let root = require_ticket_root(&app.context, selector, args.shared.project.as_deref())?;
50    let ticket = load_ticket_config(&root)?;
51
52    let engine = load_engine_config(&app.context)?;
53
54    // Deleting the ticket deletes a worktree directory that is no longer a
55    // checkout too, and nothing can check it for uncommitted work first.
56    if !args.force {
57        let mut names: Vec<&String> = ticket.worktrees.keys().collect();
58        names.sort();
59        if let Some(name) = names
60            .into_iter()
61            .find(|name| WorktreeStatus::of(&root.join(name)) == WorktreeStatus::NotAGitRepository)
62        {
63            return Err(SdkError::Message(format!(
64                "worktree '{name}' at {} is not a git repository, so tix cannot check it for \
65                 uncommitted work — move anything you need out of it, then pass --force",
66                root.join(name).display()
67            )));
68        }
69    }
70
71    if !args.force
72        && !crate::tix::utils::confirm(&format!(
73            "Destroy ticket '{}' at {}?",
74            ticket.key,
75            root.display()
76        ))?
77    {
78        info!("aborted");
79        return Ok(());
80    }
81
82    let mut entries: Vec<(&String, &WorktreeConfig)> = ticket.worktrees.iter().collect();
83    entries.sort_by(|a, b| a.0.cmp(b.0));
84
85    let mut leftovers: Vec<String> = Vec::new();
86    for (name, entry) in entries {
87        let pruned = prune_worktree(&root, name, entry, &engine, args.force);
88        match (pruned, args.force) {
89            (Ok(WorktreeStatus::Ok), _) => info!("pruned worktree {name}"),
90            // Already gone or no checkout: prune_worktree warned.
91            (Ok(_), _) => {}
92            // Forced deletion means deleted: pruning failures are warnings.
93            (Err(e), true) => {
94                warn!(name = %name, error = %e, "could not prune worktree, continuing (--force)");
95                leftovers.push(format!("{} ({})", name, entry.repo));
96            }
97            // The default path aborts before deleting anything further.
98            (Err(e), false) => {
99                let fix = match e {
100                    SdkError::Engine(TixError::Dirty { .. }) => "commit or stash the changes",
101                    _ => "fix it",
102                };
103                return Err(SdkError::Message(format!(
104                    "could not prune worktree '{name}': {e} — nothing further was deleted; \
105                     {fix} (or pass --force) and retry"
106                )));
107            }
108        }
109    }
110
111    std::fs::remove_dir_all(&root).map_err(SdkError::from)?;
112    info!("destroyed {} ({})", ticket.key, root.display());
113
114    if !leftovers.is_empty() {
115        warn!(
116            "left dangling: {} — run `git worktree prune` in the affected source repos",
117            leftovers.join(", ")
118        );
119    }
120    Ok(())
121}
