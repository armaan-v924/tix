11/// Add repository worktrees to a ticket
12#[derive(clap::Args, Debug)]
13pub struct Args {
14    #[command(flatten)]
15    pub shared: TicketSharedArgs,
16
17    /// Branch for the new worktree(s); derived from `[defaults]` + ticket
18    /// context when omitted, and required when the derived branch is taken
19    #[arg(short, long)]
20    pub branch: Option<String>,
21
22    /// Directory name for the new worktree, defaulting to the repository
23    /// alias; required when a worktree already holds that name
24    // `--as` mirrors `tix repo add --as`: one word, one meaning — "the name
25    // this lands under".
26    #[arg(long = "as", value_name = "NAME")]
27    pub name: Option<String>,
28
29    /// Registered repository aliases to add
30    #[arg(required = true)]
31    pub repo_aliases: Vec<RepoAlias>,
32}

34/// One worktree this invocation intends to create: the directory it will
35/// occupy under the ticket root, and the repository behind it.
36#[derive(Debug, PartialEq)]
37struct Planned {
38    /// The worktree directory name — the ticket map's key.
39    name: String,
40    /// The alias of the repository the worktree comes from.
41    alias: String,
42}
43
44/// The branch the new worktrees check out, and where it came from.
45///
46/// A branch already in use is a different error depending on the source:
47/// `--branch` is the remedy to point at only when it wasn't given.
48#[derive(Clone, Copy)]
49enum Branch<'a> {
50    /// Given explicitly as `--branch`.
51    Explicit(&'a str),
52    /// Derived from `[defaults]` and the ticket's key and description.
53    Derived(&'a str),
54}
55
56impl Branch<'_> {
57    fn name(&self) -> &str {
58        match self {
59            Branch::Explicit(branch) | Branch::Derived(branch) => branch,
60        }
61    }
62}

64/// Adds a worktree per given alias to the current (or `--ticket`) ticket.
65///
66/// The worktree lands in `<ticket root>/<name>`, where `<name>` is `--as`
67/// when given and the repo alias otherwise — the map key is the directory,
68/// so the single-worktree case reads exactly as it always has, and a repo
69/// can appear more than once under distinct names.
70///
71/// The branch is `--branch` when given (created if it doesn't exist), else
72/// derived exactly as `tix ticket setup` derives it —
73/// `<branch_prefix>/<key>-<sanitized-description>` — but evaluated **at add
74/// time** against the *current* `[defaults].branch_prefix` and the ticket's
75/// recorded key/description.
76///
77/// The ticket document is updated per successful worktree through the
78/// format-preserving layer, so plugin tables and comments in
79/// `.tix/ticket.toml` survive.
80pub fn run(app: &crate::tix::utils::App, args: Args) -> Result<(), SdkError> {
81    let root = require_ticket_root(&app.context, args.shared.ticket.as_ref())?;
82    let ticket = load_ticket_config(&root)?;
83
84    let document = TixDocument::load(&app.context.config_path)?;
85    let engine: EngineConfig = document.section_or_default("engine")?;
86    let defaults: Defaults = document.section_or_default("defaults")?;
87
88    // Validate the whole batch up front: an unknown alias, a taken name, and
89    // a branch already checked out are user errors, not partial-failure
90    // material.
91    for alias in &args.repo_aliases {
92        if !engine.configured_repositories.contains_key(&alias.0) {
93            let mut known: Vec<&str> = engine
94                .configured_repositories
95                .keys()
96                .map(String::as_str)
97                .collect();
98            known.sort();
99            return Err(SdkError::Engine(TixError::RepoNotFound(format!(
100                "'{}' is not a registered repository (registered: {})",
101                alias.0,
102                if known.is_empty() {
103                    "none".to_string()
104                } else {
105                    known.join(", ")
106                }
107            ))));
108        }
109    }
110
111    let branch_name = args.branch.clone().unwrap_or_else(|| {
112        derive_branch_name(
113            defaults.branch_prefix.as_deref(),
114            &ticket.key,
115            (!ticket.description.is_empty()).then_some(ticket.description.as_str()),
116        )
117    });
118    let branch = match args.branch {
119        Some(_) => Branch::Explicit(&branch_name),
120        None => Branch::Derived(&branch_name),
121    };
122    let aliases: Vec<String> = args
123        .repo_aliases
124        .iter()
125        .map(|alias| alias.0.clone())
126        .collect();
127    let planned = plan(&aliases, args.name.as_deref(), branch, &ticket)?;
128
129    let ticket_document_path = root.join(".tix").join("ticket.toml");
130    let mut failures: Vec<(String, TixError)> = Vec::new();
131    for worktree in &planned {
132        let repo_config = engine.configured_repositories[&worktree.alias].clone();
133        let path = root.join(&worktree.name);
134        let created = repo_config
135            .ensure(&worktree.alias)
136            .and_then(|repo| repo.create_worktree(&worktree.name, &branch_name, &path, false));
137        match created {
138            Ok(created) => {
139                info!(name = %worktree.name, alias = %worktree.alias, branch = %created.branch, "worktree added");
140                // Record each success immediately — a later failure must not
141                // lose the worktrees that already exist on disk.
142                with_write(&ticket_document_path, |doc| {
143                    // An explicit table reached through `table_at`, so the
144                    // entry renders as [ticket.worktrees.<name>], matching
145                    // setup's output — indexing the path directly would
146                    // collapse the worktrees into one inline line (#146).
147                    let mut entry = toml_edit::Table::new();
148                    entry["repo"] = toml_edit::value(worktree.alias.as_str());
149                    entry["branch"] = toml_edit::value(created.branch.as_str());
150                    doc.table_at(&["ticket", "worktrees"])?
151                        .insert(&worktree.name, toml_edit::Item::Table(entry));
152                    Ok(())
153                })?;
154                println!("{}", path.display());
155            }
156            Err(e) => {
157                error!(name = %worktree.name, error = %e, "failed to add worktree");
158                failures.push((worktree.name.clone(), e));
159            }
160        }
161    }
162
163    if !failures.is_empty() {
164        let names: Vec<&str> = failures.iter().map(|(name, _)| name.as_str()).collect();
165        return Err(SdkError::Message(format!(
166            "{} of {} worktrees failed: {}",
167            failures.len(),
168            planned.len(),
169            names.join(", ")
170        )));
171    }
172    Ok(())
173}

175/// Resolves the aliases and `--as` into the worktree directories to create.
176///
177/// Both defaults stay in force for a repo the ticket already tracks; what
178/// forces an explicit flag is a *collision*, not the repo's presence. The
179/// name defaults to the alias until a worktree holds that directory, and the
180/// branch derives until a worktree of the same repo has it checked out — the
181/// two conditions that make the defaults unusable, and exactly the pair a
182/// second worktree of a repo sitting at its default name trips. So
183/// `tix add api` after `tix remove api` still just works, while the second
184/// worktree of a live `api` is named and branched by hand rather than
185/// invented.
186fn plan(
187    aliases: &[String],
188    explicit_name: Option<&str>,
189    branch: Branch<'_>,
190    ticket: &TicketConfig,
191) -> Result<Vec<Planned>, SdkError> {
192    if let Some(name) = explicit_name {
193        if aliases.len() != 1 {
194            return Err(SdkError::Message(format!(
195                "--as names a single worktree, but {} repositories were given — \
196                 add them one command at a time",
197                aliases.len()
198            )));
199        }
200        validate_name(name)?;
201    }
202
203    let mut planned: Vec<Planned> = Vec::with_capacity(aliases.len());
204    for alias in aliases {
205        let name = explicit_name.unwrap_or(alias.as_str()).to_string();
206
207        if let Some(entry) = ticket.worktrees.get(&name) {
208            // Same repo under its default name is the ordinary "add it
209            // twice" slip, and it needs a branch as well as a name — say so
210            // once instead of erroring twice in a row.
211            return Err(SdkError::Message(if &entry.repo == alias {
212                format!(
213                    "ticket '{}' already has a worktree of '{alias}' at '{name}' (on {}) — \
214                     add a second with `--as <name> --branch <branch>`",
215                    ticket.key, entry.branch
216                )
217            } else {
218                format!(
219                    "ticket '{}' already has a worktree named '{name}' (of '{}') — \
220                     pick another name",
221                    ticket.key, entry.repo
222                )
223            }));
224        }
225
226        if let Some(holder) = branch_holder(ticket, alias, branch.name()) {
227            return Err(SdkError::Message(match branch {
228                Branch::Explicit(branch) => format!(
229                    "'{alias}' already has {branch} checked out in worktree '{holder}' — \
230                     git allows a branch in one worktree at a time"
231                ),
232                Branch::Derived(branch) => format!(
233                    "the branch this derives, {branch}, is already checked out in worktree \
234                     '{holder}' — pass --branch <branch> for the new worktree"
235                ),
236            }));
237        }
238
239        if planned.iter().any(|other| other.name == name) {
240            return Err(SdkError::Message(format!(
241                "'{alias}' was given twice — a second worktree of one repository needs \
242                 `--as <name> --branch <branch>`, so add it in its own command"
243            )));
244        }
245        planned.push(Planned {
246            name,
247            alias: alias.clone(),
248        });
249    }
250    Ok(planned)
251}

253/// The worktree of `alias` that already has `branch` checked out, if any.
254///
255/// Scoped to the repository, because a branch is: the same name under a
256/// different repo is a different branch and no conflict at all. Only this
257/// ticket is visible here — the same branch in another ticket's worktree, or
258/// in the source checkout itself, is git's to reject at creation.
259fn branch_holder<'a>(ticket: &'a TicketConfig, alias: &str, branch: &str) -> Option<&'a str> {
260    // Sorted so a tie names the same worktree run to run — HashMap iteration
261    // order is not stable.
262    let mut holders: Vec<&str> = ticket
263        .worktrees
264        .iter()
265        .filter(|(_, entry)| entry.repo == alias && entry.branch == branch)
266        .map(|(name, _)| name.as_str())
267        .collect();
268    holders.sort_unstable();
269    holders.into_iter().next()
270}
271
272/// Rejects `--as` values that cannot be a directory under the ticket root.
273///
274/// The worktree name *is* that directory — which is what makes the
275/// filesystem enforce uniqueness for free — so separators, the relative
276/// names, and the ticket's own `.tix` metadata directory are all out.
277fn validate_name(name: &str) -> Result<(), SdkError> {
278    if name.is_empty() || name.contains(['/', '\\']) || matches!(name, "." | ".." | ".tix") {
279        return Err(SdkError::Message(format!(
280            "'{name}' is not a valid worktree name — it becomes a directory under the ticket \
281             root, so '/', '\\', '.', '..', and '.tix' are rejected"
282        )));
283    }
284    Ok(())
285}

292    /// A ticket tracking the given `(name, repo, branch)` worktrees.
293    fn ticket(worktrees: &[(&str, &str, &str)]) -> TicketConfig {
294        TicketConfig {
295            key: "JIRA-123".to_string(),
296            description: "Fix the login bug".to_string(),
297            worktrees: worktrees
298                .iter()
299                .map(|(name, repo, branch)| {
300                    (
301                        name.to_string(),
302                        WorktreeConfig {
303                            repo: repo.to_string(),
304                            branch: branch.to_string(),
305                        },
306                    )
307                })
308                .collect(),
309        }
310    }
