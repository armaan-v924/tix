1//! `tix ticket remove` — remove single worktrees from a ticket without
2//! destroying it.
3
4use crate::tix::config::load_engine_config;

10};
11use tix_sdk::{SdkError, TixError, ignored_worktree_paths};

14/// Remove a worktree from a ticket without destroying it
15#[derive(clap::Args, Debug)]
16pub struct Args {
17    #[command(flatten)]
18    pub shared: TicketSharedArgs,
19
20    /// Worktree name(s) from the ticket's worktree map (a bare repo alias
21    /// matches its default worktree)
22    #[arg(required = true, value_name = "NAME")]
23    pub names: Vec<String>,
24
25    /// Remove worktrees with uncommitted changes, discarding them
26    #[arg(short, long)]
27    pub force: bool,
28
29    /// Print what would be removed, and what would be refused, without
30    /// removing anything; exits non-zero where the real run would fail
31    #[arg(long)]
32    pub dry_run: bool,
33}
34
35/// Prunes the named worktrees and deletes their entries from the ticket
36/// document.
37///
38/// `<name>` addresses the **worktree directory name** — the ticket map's key
39/// — not a repo alias: with #85 a repo can have several worktrees, so the
40/// alias alone is not an address. In the single-worktree case they coincide
41/// (`name == alias`), which is what makes bare-alias usage work today.
42///
43/// A worktree whose directory was deleted by hand, or is no longer a git
44/// checkout, is dropped from the document with a warning (and git's stale
45/// registration pruned) rather than failing; a directory that is still
46/// there is left in place.
47///
48/// A worktree with uncommitted changes (untracked files included) is
49/// refused unless `--force`, which discards them. The batch stops at the
50/// first refusal; worktrees already removed stay removed.
51///
52/// The ticket directory and every other worktree are untouched; removing
53/// the last worktree leaves a valid, empty ticket. Each successful prune is
54/// written back immediately through the format-preserving layer.
55///
56/// `--dry-run` walks the same batch through the same checks
57/// ([`plan_prune`]) and prints the result as a [`Plan`](crate::tix::plan::Plan) in the `--output`
58/// format. It stops where the real run would stop, on the same refusal
59/// (worded as a prediction), so its exit code answers "would this work?";
60/// with `--force` it says how many uncommitted changes would be discarded.
61pub fn run(app: &crate::tix::utils::App, args: Args) -> Result<(), SdkError> {
62    let mut output = TicketOutput::new("remove", args.dry_run);
63    let result = execute(app, args, &mut output);
64    output.finish(app.output, result)
65}

67fn execute(
68    app: &crate::tix::utils::App,
69    args: Args,
70    output: &mut TicketOutput,
71) -> Result<(), SdkError> {
72    let root = require_ticket_root(
73        &app.context,
74        args.shared.ticket.as_ref(),
75        args.shared.project.as_deref(),
76    )?;
77    output.ticket(&root)?;
78    let ticket = load_ticket_config(&root)?;
79
80    let engine = load_engine_config(&app.context)?;
81
82    // Validate the batch up front: every name must be tracked, and every
83    // backing repo must still be registered (pruning goes through the repo).
84    let mut tracked: Vec<&str> = ticket.worktrees.keys().map(String::as_str).collect();
85    tracked.sort();
86    for name in &args.names {
87        let entry = ticket.worktrees.get(name).ok_or_else(|| {
88            TixError::WorktreeNotFound(format!(
89                "'{}' is not a tracked worktree of ticket '{}' (tracked: {})",
90                name,
91                ticket.key,
92                if tracked.is_empty() {
93                    "none".to_string()
94                } else {
95                    tracked.join(", ")
96                }
97            ))
98        })?;
99        backing_repository(name, entry, &engine)?;
100    }
101
102    if let Some(plan) = &mut output.plan {
103        for name in &args.names {
104            let entry = &ticket.worktrees[name];
105            let path = root.join(name);
106            let found = match plan_prune(&root, name, entry, &engine, args.force) {
107                Ok(found) => found,
108                Err(e) => {
109                    plan.set_error(&e);
110                    return Err(refusal(name, e, true));
111                }
112            };
113            let summary = match &found {
114                PrunePlan::Prune => format!("would remove worktree {name} ({})", path.display()),
115                PrunePlan::Discard { changes } => format!(
116                    "would remove worktree {name} ({}), discarding {} (--force)",
117                    path.display(),
118                    uncommitted(*changes)
119                ),
120                PrunePlan::ForcePast { reason } => format!(
121                    "would remove worktree {name} ({}) past: {reason} (--force)",
122                    path.display()
123                ),
124                PrunePlan::Missing => {
125                    format!("would drop worktree {name} from the ticket: missing on disk")
126                }
127                PrunePlan::NotAGitRepository => format!(
128                    "would drop worktree {name} from the ticket: not a git repository \
129                     ({} would be left in place)",
130                    path.display()
131                ),
132            };
133            plan.push(prune_action(
134                match found {
135                    PrunePlan::Missing | PrunePlan::NotAGitRepository => "drop",
136                    _ => "remove",
137                },
138                summary,
139                name,
140                &entry.repo,
141                &path,
142                &found,
143            ));
144        }
145        return Ok(());
146    }
147
148    for name in &args.names {
149        let entry = &ticket.worktrees[name];
150        let path = root.join(name);
151        let status = prune_worktree(&root, name, entry, &engine, args.force).map_err(|error| {
152            output.worktree_failed(name, &path, &entry.repo, &error);
153            output.set_error(&error);
154            refusal(name, error, false)
155        })?;
156        output.pruned(name, &path, &entry.repo, status);
157        match status {
158            WorktreeStatus::NotAGitRepository => warn!(
159                "left {} in place: it is not a git checkout, so tix cannot tell whether it holds work",
160                root.join(name).display()
161            ),
162            _ => info!("removed worktree {name}"),
163        }
164    }
165    Ok(())
166}

168/// `error` from pruning `name` — or, for `dry_run`, from checking it —
169/// worded with the way past a dirty worktree.
170fn refusal(name: &str, error: SdkError, dry_run: bool) -> SdkError {
171    let lead = if dry_run {
172        "would refuse to remove"
173    } else {
174        "could not remove"
175    };
176    match error {
177        SdkError::Engine(dirty @ TixError::Dirty { .. }) => SdkError::Message(format!(
178            "{lead} worktree '{name}': {dirty} — commit or stash the changes, or pass --force to discard them"
179        )),
180        e => e,
181    }
182}

184/// The plan entry for pruning worktree `name`, with what [`plan_prune`]
185/// found recorded alongside it.
186pub(crate) fn prune_action(
187    kind: &'static str,
188    mut summary: String,
189    name: &str,
190    repo: &str,
191    path: &std::path::Path,
192    found: &PrunePlan,
193) -> Action {
194    let inventory = match found {
195        PrunePlan::Prune | PrunePlan::Discard { .. } | PrunePlan::ForcePast { .. } => {
196            match ignored_worktree_paths(path) {
197                Ok(paths) => {
198                    if !paths.is_empty() {
199                        summary.push_str(&format!(", including ignored entries: {paths:?}"));
200                    }
201                    Some(("ignored_paths", toml::Value::from(paths)))
202                }
203                Err(error) => {
204                    summary.push_str(&format!(", ignored entries could not be listed ({error})"));
205                    Some(("ignored_paths_error", toml::Value::from(error.to_string())))
206                }
207            }
208        }
209        PrunePlan::Missing | PrunePlan::NotAGitRepository => None,
210    };
211    let mut action = Action::new(kind, summary)
212        .with("worktree", name)
213        .with("repo", repo)
214        .with_path("path", path);
215    if let Some((key, value)) = inventory {
216        action = action.with(key, value);
217    }
218    match found {
219        PrunePlan::Prune => action,
220        PrunePlan::Discard { changes } => action
221            .with("forced", true)
222            .with("uncommitted_changes", *changes as i64),
223        PrunePlan::ForcePast { reason } => {
224            action.with("forced", true).with("reason", reason.as_str())
225        }
226        PrunePlan::Missing => action.with("reason", "missing on disk"),
227        PrunePlan::NotAGitRepository => action.with("reason", "not a git repository"),
228    }
229}
