1//! Projects: named namespaces one level above tickets.
2//!
3//! A project is a directory under the tickets directory that groups tickets:
4//! loose tickets live at `<tickets_directory>/<KEY>/`, project tickets at
5//! `<tickets_directory>/<project>/<KEY>/`. What makes a directory a project
6//! is the **file** `.tix/project.toml`, created empty
7//! ([projects](https://tix.armaanv.dev/latest/concepts/projects/)).
8//!
9//! The marker is an assertion, not metadata. It is what lets the discovery
10//! walk ([`crate::discovery::discover_from`]) recognise a project without
11//! reading config, and it is what the listing uses too, so there is one
12//! definition of "project" rather than two that can disagree — a stray
13//! directory under the tickets directory is not a project. It is empty on
14//! purpose: adding fields to a document that exists is additive, adding a
15//! file where none existed is not, so the file is settled now and its
16//! contents wait for a reason to exist.
17//!
18//! Projects are created implicitly on first use ([`Project::ensure`]);
19//! there is no separate "create a project" step.
20
21use crate::discovery::{TIX_DIR, discover_from, is_ticket_root, same_directory};

27/// Returns true if `path` is a project root: a directory containing the file
28/// `.tix/project.toml`.
29///
30/// The predicate is the file, not the `.tix/` directory, for the same reason
31/// [`is_ticket_root`] tests its own file: tickets have a `.tix/` too.
32pub fn is_project_root(path: &Path) -> bool {
33    marker_path(path).is_file()
34}
35
36/// The marker file of the project rooted at `root`: `<root>/.tix/project.toml`.
37pub fn marker_path(root: &Path) -> PathBuf {
38    root.join(TIX_DIR).join("project.toml")
39}

41/// A project: its name, and the directory its tickets live in.
42#[derive(Debug, Clone, PartialEq, Eq)]
43pub struct Project {
44    /// The project name — the directory name.
45    pub name: String,
46    /// The project directory. Its tickets are its immediate children.
47    pub root: PathBuf,
48}
49
50impl Project {
51    /// The project called `name` under `tickets_directory`, whether or not
52    /// it exists yet.
53    ///
54    /// # Errors
55    ///
56    /// [`SdkError::Message`] when `name` cannot be a directory name: empty,
57    /// `.`/`..`, or containing `/` or `\`.
58    pub fn named(tickets_directory: &Path, name: &str) -> Result<Self, SdkError> {
59        if name.is_empty() {
60            return Err(SdkError::Message(
61                "the project name is empty — it becomes a directory name".to_string(),
62            ));
63        }
64        if name == "." || name == ".." || name.contains(['/', '\\']) {
65            return Err(SdkError::Message(format!(
66                "'{name}' is not a valid project name — it becomes a directory name, so '/', \
67                 '\\', '.' and '..' are rejected"
68            )));
69        }
70        Ok(Self {
71            name: name.to_string(),
72            root: tickets_directory.join(name),
73        })
74    }
75
76    /// The project rooted at `root`, named for its directory. `None` when the
77    /// path has no final component (e.g. `/`).
78    ///
79    /// Does not check the marker; see [`is_project_root`].
80    pub fn at(root: &Path) -> Option<Self> {
81        let name = root.file_name()?.to_string_lossy().into_owned();
82        Some(Self {
83            name,
84            root: root.to_path_buf(),
85        })
86    }
87
88    /// Whether this project exists: its root carries the marker.
89    pub fn exists(&self) -> bool {
90        is_project_root(&self.root)
91    }
92
93    /// Creates the project if it does not exist yet: the directory, and the
94    /// empty marker inside it. A no-op for an existing project.
95    ///
96    /// A directory of that name that is *not* yet a project — created by
97    /// hand, say — is adopted by writing the marker into it.
98    ///
99    /// # Errors
100    ///
101    /// [`SdkError::Message`] when the directory is a ticket root: a loose
102    /// ticket already has that name, and a ticket cannot also be a project.
103    /// IO errors propagate.
104    pub fn ensure(&self) -> Result<(), SdkError> {
105        if is_ticket_root(&self.root) {
106            return Err(SdkError::Message(format!(
107                "cannot use '{}' as a project — {} is already a ticket",
108                self.name,
109                self.root.display()
110            )));
111        }
112        let marker = marker_path(&self.root);
113        if let Some(parent) = marker.parent() {
114            std::fs::create_dir_all(parent)?;
115        }
116        // `create_new` so a concurrent creator never truncates a marker that
117        // has since grown contents.
118        match std::fs::OpenOptions::new()
119            .write(true)
120            .create_new(true)
121            .open(&marker)
122        {
123            Ok(_) => {
124                debug!(path = %self.root.display(), "project created");
125                Ok(())
126            }
127            Err(e) if e.kind() == ErrorKind::AlreadyExists => Ok(()),
128            Err(e) => Err(e.into()),
129        }
130    }
131
132    /// Whether the ticket at `ticket_root` belongs to this project — its
133    /// nearest enclosing project is this one.
134    ///
135    /// Compared as directories, not as path strings, so a project reached
136    /// through a symlink still matches.
137    pub fn contains(&self, ticket_root: &Path) -> bool {
138        project_of(ticket_root).is_some_and(|project| same_directory(&project.root, &self.root))
139    }
140}

142/// The project the ticket at `ticket_root` belongs to, if any: the nearest
143/// project root above it.
144///
145/// Walks from the ticket rather than reading config, so a ticket outside the
146/// tickets directory still reports its project.
147pub fn project_of(ticket_root: &Path) -> Option<Project> {
148    discover_from(ticket_root)
149        .project
150        .as_deref()
151        .and_then(Project::at)
152}
153
154/// Resolves the project a command is scoped to.
155///
156/// In order:
157///
158/// 1. **Flag** — `flag` names a project under `tickets_directory`. It need
159///    not exist yet; commands that create tickets call [`Project::ensure`].
160/// 2. **Inferred** — the nearest project above `cwd`, found by the discovery
161///    walk ([`discover_from`]). Like ticket discovery, this is not bounded by
162///    `tickets_directory`.
163/// 3. **Loose** — `None`: tickets live directly in `tickets_directory`.
164///
165/// Pass the **logical** cwd ([`crate::discovery::logical_cwd`]).
166///
167/// # Errors
168///
169/// [`SdkError::Message`] when `flag` is not a valid project name.
170pub fn resolve_project(
171    flag: Option<&str>,
172    tickets_directory: &Path,
173    cwd: &Path,
174) -> Result<Option<Project>, SdkError> {
175    if let Some(name) = flag {
176        return Project::named(tickets_directory, name).map(Some);
177    }
178    Ok(discover_from(cwd).project.as_deref().and_then(Project::at))
179}

181/// The directory tickets of `project` live in: the project root, or
182/// `tickets_directory` itself for loose tickets.
183pub fn tickets_home<'a>(project: Option<&'a Project>, tickets_directory: &'a Path) -> &'a Path {
184    project.map_or(tickets_directory, |project| &project.root)
185}
186
187/// Every project directly under `tickets_directory`, sorted by name.
188///
189/// A project is a child directory carrying the marker; anything else — a
190/// loose ticket, a stray directory, a file — is not one. A missing tickets
191/// directory means no projects.
192///
193/// # Errors
194///
195/// IO errors other than the directory not existing.
196pub fn list_projects(tickets_directory: &Path) -> Result<Vec<Project>, SdkError> {
197    let entries = match std::fs::read_dir(tickets_directory) {
198        Ok(entries) => entries,
199        Err(e) if e.kind() == ErrorKind::NotFound => return Ok(Vec::new()),
200        Err(e) => return Err(e.into()),
201    };
202    let mut projects: Vec<Project> = entries
203        .filter_map(Result::ok)
204        .map(|entry| entry.path())
205        .filter(|path| is_project_root(path) && !is_ticket_root(path))
206        .filter_map(|path| Project::at(&path))
207        .collect();
208    projects.sort_by(|a, b| a.name.cmp(&b.name));
209    Ok(projects)
210}

217    fn make_ticket_root(root: &Path) {
218        std::fs::create_dir_all(root.join(".tix")).unwrap();
219        std::fs::write(root.join(".tix/ticket.toml"), "[ticket]\nkey = \"K\"\n").unwrap();
220    }

224    /// A project name becomes a directory name, so it is held to the same

228        let tickets = Path::new("/t");
229        for name in ["", ".", "..", "a/b", "a\\b"] {

238    /// First use creates the directory and the empty marker; a second use is
239    /// a no-op that leaves the marker alone.
240    #[test]
241    fn test_ensure_creates_marker_once() {
242        let dir = tempdir().unwrap();
243        let project = Project::named(&dir.path().join("tickets"), "acme").unwrap();
244        assert!(!project.exists());
245
246        project.ensure().unwrap();
247        assert!(project.exists());
248        let marker = marker_path(&project.root);
249        assert_eq!(std::fs::read_to_string(&marker).unwrap(), "");
250
251        std::fs::write(&marker, "# hand-written\n").unwrap();
252        project.ensure().unwrap();
253        assert_eq!(
254            std::fs::read_to_string(&marker).unwrap(),
255            "# hand-written\n"
256        );
257    }
