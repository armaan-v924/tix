import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid>
  <Card title="Ticket-scoped workspaces" icon="open-book">
    `tix setup JIRA-123` creates a directory for the ticket and a worktree
    inside it for each repository the work touches, each on its own branch.
  </Card>
  <Card title="Your checkouts stay put" icon="approve-check">
    Worktrees hang off the repositories you already have. Nothing is cloned
    twice, and nothing switches branches under you.
  </Card>
  <Card title="Destroy without residue" icon="trash">
    `tix destroy` prunes every worktree and removes the directory — and
    refuses outright while any of them still has uncommitted work.
  </Card>
  <Card title="Extensible" icon="puzzle">
    Any executable named `tix-<name>` on your `PATH` becomes `tix <name>`,
    invoked with the workspace context already resolved.
  </Card>
</CardGrid>
