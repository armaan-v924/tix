1//! Credential wiring for authenticated remotes.
2//!
3//! libgit2 does not consult git's credential helpers on its own. Without an
4//! explicit callback, any remote that asks for authentication fails with
5//! "remote authentication required but no callback set" — including every
6//! private repository and every SSH remote. This module supplies the
7//! callback and the [`git2`] option structs that carry it.
8//!
9//! Credentials themselves are never read, stored, or logged by tix: the
10//! callback delegates to the ssh-agent and to whatever credential helper the
11//! user's git config already declares, so tix inherits the machine's
12//! existing setup rather than defining one of its own.
13
14use git2::{Cred, CredentialType, Error, FetchOptions, RemoteCallbacks};
15use tracing::debug;
16
17/// How many times a single remote may be asked for credentials.
18///
19/// libgit2 re-invokes the callback when the server rejects what it was
20/// given. A helper that keeps returning the same wrong answer would loop
21/// forever, so refuse once the plausible options are exhausted.
22const MAX_ATTEMPTS: usize = 3;
23
24/// Remote callbacks that answer credential challenges.
25///
26/// Resolution order follows what the server offers: an ssh-agent key for SSH
27/// remotes, then the git credential helper for HTTPS, then the default
28/// (which covers negotiated schemes and unauthenticated remotes that still
29/// raise a challenge).
30pub(crate) fn remote_callbacks<'cb>() -> RemoteCallbacks<'cb> {
31    let mut callbacks = RemoteCallbacks::new();
32    let mut attempts = 0usize;
33
34    callbacks.credentials(move |url, username_from_url, allowed| {
35        attempts += 1;
36        if attempts > MAX_ATTEMPTS {
37            return Err(Error::from_str(
38                "authentication failed: exhausted available credentials",
39            ));
40        }
41        debug!(attempt = attempts, ?allowed, "credential challenge");
42
43        if allowed.contains(CredentialType::SSH_KEY) {
44            return Cred::ssh_key_from_agent(username_from_url.unwrap_or("git"));
45        }
46        if allowed.contains(CredentialType::USER_PASS_PLAINTEXT) {
47            let config = git2::Config::open_default()?;
48            return Cred::credential_helper(&config, url, username_from_url);
49        }
50        if allowed.contains(CredentialType::DEFAULT) {
51            return Cred::default();
52        }
53        Err(Error::from_str(
54            "authentication failed: no supported credential type offered",
55        ))
56    });
57
58    callbacks
59}
60
61/// Fetch options carrying the credential callbacks.
62///
63/// Use for every network operation — a bare `None` skips authentication
64/// entirely and fails on any remote that challenges.
65pub(crate) fn fetch_options<'cb>() -> FetchOptions<'cb> {
66    let mut options = FetchOptions::new();
67    options.remote_callbacks(remote_callbacks());
68    options
69}
