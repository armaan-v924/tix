13/// Create a new ticket workspace with worktrees
14#[derive(clap::Args, Debug)]
15pub struct Args {
16    /// Where the ticket goes: a bare key (e.g. JIRA-123) created under the
17    /// tickets directory, or a path (anything containing a separator, or
18    /// `.`/`..`) created at that location
19    #[arg(value_hint = clap::ValueHint::DirPath)]
20    pub target: TicketRef,

28    /// from the current directory when omitted; outside any project the

31    pub project: Option<String>,

33    /// Human-readable description; feeds branch name derivation

37    /// What each new branch is cut from: a branch name (local or origin's,
38    /// whichever is ahead), or any ref, tag, or SHA. Defaults to each
39    /// repository's own default branch on origin

41    pub base: Option<String>,

52/// The project a new ticket goes in: `--project`, else the project the cwd
53/// is inside, else none (a loose ticket, directly in the tickets directory).
54///
55/// Only the key form has a project to choose. A path names its location
56/// outright, so `--project` alongside one could only contradict it, and no
57/// project is inferred for it — a path into a project directory lands in that
58/// project by being there.
59///
60/// # Errors
61///
62/// [`SdkError::Message`] when `--project` accompanies the path form, or is
63/// not a valid project name.
64fn project_for(
65    target: &TicketRef,
66    flag: Option<&str>,
67    tickets_directory: &Path,
68    cwd: &Path,
69) -> Result<Option<Project>, SdkError> {
70    match target {
71        TicketRef::Path(path) => match flag {
72            Some(name) => Err(SdkError::Message(format!(
73                "--project '{name}' conflicts with the path '{}', which already says where the \
74                 ticket goes — pass a bare key to create it in the project",
75                path.display()
76            ))),
77            None => Ok(None),
78        },
79        TicketRef::Id(_) => resolve_project(flag, tickets_directory, cwd),
80    }
81}

83/// Which shape the target argument took.
84///
85/// The two forms agree on everything except where missing directories may be
86/// invented — see [`create_ticket_root`].
87#[derive(Debug, Clone, Copy, PartialEq, Eq)]
88enum Form {
89    /// A bare key, placed under the configured tickets directory.
90    Key,
91    /// An explicit path, placed exactly where it points.
92    Path,
93}
94
95/// Where a ticket will be created, and the key its document will record.
96#[derive(Debug, PartialEq, Eq)]
97struct Target {
98    /// The directory to create: `<tickets_directory>/<key>` for the key
99    /// form, the resolved argument for the path form.
100    root: PathBuf,
101    /// The ticket key — the argument itself, the directory name, or `--key`.
102    key: String,
103    /// The shape the argument took.
104    form: Form,
105}
106
107/// Resolves the creation target: where the ticket directory goes, and what
108/// key it records.
109///
110/// Shape decides, exactly as `--ticket` is disambiguated ([`TicketRef`],
111/// `design/spec.md` §4) — one rule for finding tickets and creating them:
112///
113/// - **Key form** (a bare name): created at `home.join(key)`, and the key is
114///   the argument. `home` is the tickets directory for a loose ticket (v2
115///   parity) or the project directory ([`tickets_home`]).
116/// - **Path form** (contains a separator, is absolute, or is `.`/`..`):
117///   created at that path, wherever it points. The key defaults to the
118///   directory name, which is the only thing the key form could ever have
119///   given it; `--key` overrides when the directory should be named for
120///   humans and the ticket for the tracker.
121///
122/// Relative paths resolve against `cwd`, which callers pass as the
123/// **logical** cwd: never canonicalized, so a ticket created through a
124/// symlink keeps the identity the user sees (the discovery rule, applied at
125/// creation).
126///
127/// # Errors
128///
129/// [`SdkError::Message`] when the key is empty or unusable as a directory
130/// name, when a path names no directory to create (`/`, `..` — the argument
131/// must name the ticket directory itself, not the place it goes), or when
132/// `--key` accompanies the key form, where it could only contradict the
133/// argument that already names the ticket.
134fn resolve_target(
135    target: &TicketRef,
136    key_override: Option<&str>,
137    home: &Path,
138    cwd: &Path,
139) -> Result<Target, SdkError> {
140    match target {
141        TicketRef::Id(key) => {
142            if let Some(override_key) = key_override {
143                return Err(SdkError::Message(format!(
144                    "--key '{override_key}' conflicts with '{key}', which already names the ticket — \
145                     pass a path (e.g. './{key}') to give the directory a name of its own"
146                )));
147            }
148            let key = validated_key(key)?;
149            Ok(Target {
150                root: home.join(&key),
151                key,
152                form: Form::Key,
153            })
154        }
155        TicketRef::Path(path) => {
156            let root = if path.is_absolute() {
157                drop_current_dir_components(path)
158            } else {
159                drop_current_dir_components(&cwd.join(path))
160            };
161            let key = match key_override {
162                Some(key) => validated_key(key)?,
163                None => {
164                    let name = root.file_name().ok_or_else(|| {
165                        SdkError::Message(format!(
166                            "'{}' does not name a directory to create — give the path the ticket \
167                             itself should occupy, e.g. './client-work/JIRA-123'",
168                            path.display()
169                        ))
170                    })?;
171                    validated_key(&name.to_string_lossy())?
172                }
173            };
174            Ok(Target {
175                root,
176                key,
177                form: Form::Path,
178            })
179        }
180    }
181}

183/// Drops `.` components, so `./client-work/JIRA-1` prints and is recorded as
184/// the location it actually names.
185///
186/// Only `.` — it resolves to the directory it sits in whatever that directory
187/// is, so removing it cannot change which path is meant. `..` is deliberately
188/// left in place: collapsing it lexically would resolve *through* a symlink
189/// and pick a different directory than the shell would (`design/spec.md` §4).
190fn drop_current_dir_components(path: &Path) -> PathBuf {
191    path.components()
192        .filter(|component| !matches!(component, std::path::Component::CurDir))
193        .collect()
194}
195
196/// Returns `key` if it can serve as both a ticket key and a directory name.
197///
198/// The key form cannot produce most of these (a name with a separator parses
199/// as a path, not a key), but `--key` takes arbitrary text and the check is
200/// what keeps it from writing a ticket that no directory can hold.
201fn validated_key(key: &str) -> Result<String, SdkError> {
202    if key.is_empty() {
203        return Err(SdkError::Message(
204            "the ticket key is empty — it becomes a directory name".to_string(),
205        ));
206    }
207    if key.contains(['/', '\\']) {
208        return Err(SdkError::Message(format!(
209            "'{key}' is not a valid ticket key — it becomes a directory name, so '/' and '\\' are rejected"
210        )));
211    }
212    if key == "." || key == ".." {
213        return Err(SdkError::Message(format!(
214            "'{key}' is not a valid ticket key — it becomes a directory name"
215        )));
216    }
217    Ok(key.to_string())
218}

220/// Creates the ticket directory, refusing to overwrite one and — in the path
221/// form — refusing to invent its parent.
222///
223/// The forms differ deliberately. `tickets_directory` is configured and
224/// intentional, so the key form creates it when it is missing. An arbitrary
225/// path is neither: a typo (`./cleint-work/JIRA-1`) would otherwise grow a
226/// tree of worktrees somewhere the user never meant, and unwinding that costs
227/// more than the `mkdir -p` the error asks for. Creating parents is the
228/// loosening we could still make later; requiring them is not.
229fn create_ticket_root(target: &Target) -> Result<(), SdkError> {
230    if target.root.exists() {
231        return Err(SdkError::Message(format!(
232            "ticket '{}' already exists at {}",
233            target.key,
234            target.root.display()
235        )));
236    }
237    match target.form {
238        Form::Key => std::fs::create_dir_all(&target.root).map_err(SdkError::from),
239        Form::Path => {
240            let parent = target.root.parent().ok_or_else(|| {
241                SdkError::Message(format!(
242                    "{} has no parent directory to create the ticket in",
243                    target.root.display()
244                ))
245            })?;
246            if !parent.is_dir() {
247                return Err(SdkError::Message(format!(
248                    "{} does not exist — create it first (`mkdir -p`), or pass a bare key to \
249                     create the ticket under the tickets directory",
250                    parent.display()
251                )));
252            }
253            std::fs::create_dir(&target.root).map_err(SdkError::from)
254        }
255    }
256}

258/// Creates a ticket workspace with a worktree per selected repo.
259///
260/// The target argument is disambiguated by shape (`resolve_target`): a bare
261/// key lands under `tickets_directory` — inside its project when there is one
262/// (`project_for`) — and a path lands where it points. Both forms are
263/// otherwise identical — same seeds, same derivation, same document.
264///
265/// A project named for the first time is created here, just before the
266/// ticket inside it: the directory and its empty `.tix/project.toml`.
267///
268/// Runs **without ticket context** by design — this is the command that
269/// creates it. `[defaults]` is read **once**, here, and the derived values
270/// are written into the ticket document; later changes to `[defaults]` never
271/// touch this ticket (see
272/// [creation-time seeds](https://tix.armaanv.dev/latest/concepts/seeds/)).
273///
274/// Repo selection: explicit aliases win, `--all` takes every registered
275/// repo, and neither falls back to the `defaults.repositories` seed list.
276///
277/// Each branch is cut from `--base`, or from its own repository's default
278/// branch on origin — so one ticket can span a `main` repo and a `master`
279/// repo without flags. The resolved base is recorded per worktree, the same
280/// stamp-once discipline the branch name gets; source checkouts are never
281/// touched.
282///
283/// Partial failure leaves successfully created worktrees in place and
284/// records only them in `.tix/ticket.toml`; the command then errors naming
285/// the repos that failed.
286pub fn run(app: &crate::tix::utils::App, args: Args) -> Result<(), SdkError> {
287    let document = TixDocument::load(&app.context.config_path)?;
288    let cli: CliConfig = document.section("cli")?.ok_or_else(|| {
289        SdkError::Message("global config has no [cli] section — run `tix config init`".to_string())
290    })?;
291    let engine: EngineConfig = document.section_or_default("engine")?;
292    let defaults: Defaults = document.section_or_default("defaults")?;
293
294    let cwd = tix_sdk::discovery::logical_cwd()?;
295    let project = project_for(
296        &args.target,
297        args.project.as_deref(),
298        &cli.tickets_directory,
299        &cwd,
300    )?;
301    let target = resolve_target(
302        &args.target,
303        args.key.as_deref(),
304        tickets_home(project.as_ref(), &cli.tickets_directory),
305        &cwd,
306    )?;
307
308    // --- select repositories ---
309    let selected: Vec<String> = if args.all {
310        let mut aliases: Vec<String> = engine.configured_repositories.keys().cloned().collect();
311        aliases.sort();
312        aliases
313    } else if !args.repo_aliases.is_empty() {
314        args.repo_aliases
315            .iter()
316            .map(|alias| alias.0.clone())
317            .collect()
318    } else {
319        defaults.repositories.clone()
320    };
321    for alias in &selected {
322        if !engine.configured_repositories.contains_key(alias) {
323            let mut known: Vec<&str> = engine
324                .configured_repositories
325                .keys()
326                .map(String::as_str)
327                .collect();
328            known.sort();
329            return Err(SdkError::Engine(TixError::RepoNotFound(format!(
330                "'{alias}' is not a registered repository (registered: {})",
331                if known.is_empty() {
332                    "none".to_string()
333                } else {
334                    known.join(", ")
335                }
336            ))));
337        }
338    }
339
340    // --- seed reads: derivation happens once, now ---
341    let branch = derive_branch_name(
342        defaults.branch_prefix.as_deref(),
343        &target.key,
344        args.description.as_deref(),
345    );
346    let description = args.description.clone().unwrap_or_default();
347
348    if let Some(project) = &project {
349        project.ensure()?;
350    }
351    create_ticket_root(&target)?;
352    let ticket_root = &target.root;
353
354    // --- create worktrees; collect successes and failures ---
355    let mut worktrees: HashMap<String, WorktreeConfig> = HashMap::new();
356    let mut failures: Vec<(String, TixError)> = Vec::new();
357    for alias in &selected {
358        let repo_config = engine.configured_repositories[alias].clone();
359        let result = repo_config.ensure(alias).and_then(|repo| {
360            repo.create_worktree(
361                alias,
362                &branch,
363                &ticket_root.join(alias),
364                args.base.as_deref(),
365            )
366        });
367        match result {
368            Ok(worktree) => {
369                info!("created worktree {alias} on {}", worktree.branch);
370                worktrees.insert(
371                    alias.clone(),
372                    WorktreeConfig {
373                        repo: alias.clone(),
374                        branch: worktree.branch,
375                        base: worktree.base,
376                    },
377                );
378            }
379            Err(e) => {
380                error!("{alias}: could not create worktree: {e}");
381                failures.push((alias.clone(), e));
382            }
383        }
384    }
385
386    // --- write the ticket document, recording only what actually exists ---
387    let ticket = TicketConfig {
388        key: target.key.clone(),
389        description,
390        worktrees,
391    };
392    let mut ticket_document = TixDocument::empty();
393    ticket_document.set_section("ticket", &ticket)?;
394    ticket_document.save(&ticket_root.join(".tix").join("ticket.toml"))?;
395
396    println!("{}", ticket_root.display());
397
398    if !failures.is_empty() {
399        let names: Vec<&str> = failures.iter().map(|(alias, _)| alias.as_str()).collect();
400        return Err(SdkError::Message(format!(
401            "ticket created, but {} of {} worktrees failed: {} — fix and `tix ticket add` them",
402            failures.len(),
403            selected.len(),
404            names.join(", ")
405        )));
406    }
407    Ok(())
408}
